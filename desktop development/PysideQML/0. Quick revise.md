
#### Text

```JavaScript
Text{
	text: "I can do all things through Christ who strengthens me"
	anchors.centerIn: parent
	font.pixelSize: 20 //use fontSizeMode or minimumPixelSize or... instead
	font.overline: true
	font.strikeout: true
	font.underline: true
	font.bold: true
	font.italic: true
	font.letterSpacing: 20
	font.wordSpacing: 10
	wrapMode: Text.WordWrap //WrapAnywhere
	elide: Text.ElideRight
	fontSizeMode: Text.Fit // key: will shrink if no space
	font.minimumPixelSize: 14 
	font.pixelSize: parent.height * 0.4
}

Label {
    text: "Styled Label"
    font.pixelSize: 24
    color: "blue"
    horizontalAlignment: Text.AlignHCenter
}
```

#### Button

```JavaScript
Button{
	id: button
	contentItem: Text {
		text: "Click Me"
		color: root.textColor
		horizontalAlignment: Text.AlignHCenter
		verticalAlignment: Text.AlignVCenter
		font.bold: true
	}
	anchors.centerIn: parent
	background: Rectangle{
		color: root.buttonColor
		radius: 10
		implicitWidth: 100
		implicitHeight: 40
	}
}
```

#### Input Field

```JavaScript
//for single line field
TextField {
	Layout.preferredHeight: 44
	Layout.fillWidth: true
	leftPadding: 10
	rightPadding: 10
	placeholderText: "Email"
	color: root.whiteText //indicator + text color
	background: Rectangle { 
		border.color: root.borderColor
		radius: 8             
		color: parent.activeFocus ? root.fieldColor : root.fieldColor
		border.width: parent.activeFocus ? 2 : 1
	}
}

//for multi line field
TextArea {
    id: notes
    placeholderText: "Write your notes here..."
    width: parent.width * 0.8
    height: 100
    wrapMode: Text.Wrap
}
```
