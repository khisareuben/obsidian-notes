## 📝 Step 1: Python Backend with QThread

We’ll create a worker that runs in a background thread and reports progress back to QML.

python

```
# main.py
import sys
import time
from PySide6.QtCore import QObject, Slot, Signal, QThread
from PySide6.QtGui import QGuiApplication
from PySide6.QtQml import QQmlApplicationEngine

class Worker(QObject):
    progressChanged = Signal(int)
    finished = Signal()

    @Slot()
    def run_task(self):
        # Simulate a long task
        for i in range(1, 101):
            time.sleep(0.05)  # simulate work
            self.progressChanged.emit(i)
        self.finished.emit()

class Backend(QObject):
    def __init__(self):
        super().__init__()
        self.thread = None
        self.worker = None

    @Slot()
    def start_task(self):
        self.thread = QThread()
        self.worker = Worker()
        self.worker.moveToThread(self.thread)

        self.thread.started.connect(self.worker.run_task)
        self.worker.progressChanged.connect(self.on_progress)
        self.worker.finished.connect(self.on_finished)
        self.worker.finished.connect(self.thread.quit)
        self.worker.finished.connect(self.worker.deleteLater)
        self.thread.finished.connect(self.thread.deleteLater)

        self.thread.start()

    progressChanged = Signal(int)
    taskFinished = Signal()

    def on_progress(self, value):
        self.progressChanged.emit(value)

    def on_finished(self):
        self.taskFinished.emit()

app = QGuiApplication(sys.argv)
engine = QQmlApplicationEngine()

backend = Backend()
engine.rootContext().setContextProperty("backend", backend)

engine.load("main.qml")
if not engine.rootObjects():
    sys.exit(-1)

sys.exit(app.exec())
```

## 🎨 Step 2: QML UI with ProgressBar

qml

```
// main.qml
import QtQuick
import QtQuick.Controls

ApplicationWindow {
    visible: true
    width: 400
    height: 200
    title: "Background Task Example"

    Column {
        anchors.centerIn: parent
        spacing: 15

        ProgressBar {
            id: progressBar
            width: parent.width * 0.8
            from: 0
            to: 100
            value: 0
        }

        Button {
            text: "Start Task"
            onClicked: backend.start_task()
        }

        Label {
            id: statusLabel
            text: "Idle"
        }
    }

    Connections {
        target: backend
```

## 🧠 Key Concepts Learned

- **QThread**: Runs tasks in a separate thread to keep UI responsive.
    
- **Worker objects**: Encapsulate background logic.
    
- **Signals**: Report progress and completion back to QML.
    
- **ProgressBar**: Visual feedback for long tasks.
