## 📝 Step 1: Local SQLite Database (Offline Storage)



```python
# backend/database.py
import sqlite3

class LocalDatabase:
    def __init__(self, db_name="local_data.db"):
        self.conn = sqlite3.connect(db_name)
        self.create_table()

    def create_table(self):
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS institutions (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT,
                status TEXT,
                balance REAL
            )
        """)
        self.conn.commit()

    def add_institution(self, name, status, balance):
        self.conn.execute(
            "INSERT INTO institutions (name, status, balance) VALUES (?, ?, ?)",
            (name, status, balance)
        )
        self.conn.commit()

    def get_all(self):
        cursor = self.conn.execute("SELECT * FROM institutions")
        return cursor.fetchall()
```

📌 This ensures data is always stored locally, even without internet.

## 📝 Step 2: Sync Logic (When Online)



```python
# backend/sync.py
import requests

class SyncManager:
    def __init__(self, db):
        self.db = db
        self.server_url = "https://example.com/api/sync"

    def sync_to_server(self):
        # Get local data
        institutions = self.db.get_all()
        payload = [{"id": row[0], "name": row[1], "status": row[2], "balance": row[3]} for row in institutions]

        try:
            response = requests.post(self.server_url, json=payload)
            if response.status_code == 200:
                return "Sync successful"
            else:
                return f"Sync failed: {response.status_code}"
        except Exception as e:
            return f"Error: {e}"
```

📌 When internet is available, local data is pushed to the server.

## 📝 Step 3: Integrating with PySide6



```python
# main.py
import sys
from PySide6.QtCore import QObject, Slot, Signal
from PySide6.QtGui import QGuiApplication
from PySide6.QtQml import QQmlApplicationEngine
from backend.database import LocalDatabase
from backend.sync import SyncManager

class Backend(QObject):
    syncResult = Signal(str)

    def __init__(self):
        super().__init__()
        self.db = LocalDatabase()
        self.sync_manager = SyncManager(self.db)

    @Slot(str, str, float)
    def add_institution(self, name, status, balance):
        self.db.add_institution(name, status, balance)

    @Slot()
    def sync_data(self):
        result = self.sync_manager.sync_to_server()
        self.syncResult.emit(result)

app = QGuiApplication(sys.argv)
engine = QQmlApplicationEngine()

backend = Backend()
engine.rootContext().setContextProperty("backend", backend)

engine.load("ui/main.qml")
if not engine.rootObjects():
    sys.exit(-1)

sys.exit(app.exec())
```

## 🧠 Key Concepts Learned

- **SQLite local storage** → Always available offline.
    
- **Sync manager** → Pushes local data to server when online.
    
- **Integration with QML** → Buttons trigger local save and sync.
    
- **Resilience** → Users can keep working offline, then sync later.
