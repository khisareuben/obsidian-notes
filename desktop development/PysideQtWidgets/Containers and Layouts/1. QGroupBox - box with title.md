
Think of it like a box with a title and a border that groups related widgets together.
It draws a frame + title on top. Inside it you put other widgets

`main.py`
```python
import sys
from pathlib import Path
from PySide6.QtWidgets import QApplication, QMainWindow
from widget import GroupBoxWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setCentralWidget(GroupBoxWidget())

        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.showMaximized()
    sys.exit(app.exec())
```

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QGroupBox, QLabel, QLineEdit, QCheckBox, QPushButton
from PySide6.QtCore import Slot

class GroupBoxWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === GROUP 1: Customer Info ===
        self.group_customer = QGroupBox("Customer Information")
        self.group_customer.setObjectName("group_customer")
        self.group_customer.setCheckable(False) # set True for collapsible

        self.label_name = QLabel("Full Name:")
        self.label_name.setObjectName("label_small")

        self.input_name = QLineEdit()
        self.input_name.setObjectName("input")
        self.input_name.setPlaceholderText("John Doe")

        layout_customer = QVBoxLayout(self.group_customer)
        layout_customer.setSpacing(8)
        layout_customer.addWidget(self.label_name)
        layout_customer.addWidget(self.input_name)

        # === GROUP 2: Delivery Options (Checkable) ===
        self.group_delivery = QGroupBox("Delivery Options")
        self.group_delivery.setObjectName("group_delivery")
        self.group_delivery.setCheckable(True) # <-- Check to enable
        self.group_delivery.setChecked(True)

        self.check_express = QCheckBox("Express Delivery (+ KES 200)")
        self.check_express.setObjectName("check")
        self.check_insurance = QCheckBox("Add Insurance (+ KES 100)")
        self.check_insurance.setObjectName("check")

        layout_delivery = QVBoxLayout(self.group_delivery)
        layout_delivery.addWidget(self.check_express)
        layout_delivery.addWidget(self.check_insurance)

        # === GROUP 3: Payment ===
        self.group_payment = QGroupBox("Payment Method")
        self.group_payment.setObjectName("group_payment")

        self.btn_mpesa = QPushButton("M-Pesa")
        self.btn_mpesa.setObjectName("btn_pay")
        self.btn_mpesa.setCheckable(True)
        self.btn_mpesa.setChecked(True)

        self.btn_cash = QPushButton("Cash")
        self.btn_cash.setObjectName("btn_pay")
        self.btn_cash.setCheckable(True)

        layout_payment = QVBoxLayout(self.group_payment)
        layout_payment.addWidget(self.btn_mpesa)
        layout_payment.addWidget(self.btn_cash)

        # Result
        self.label_result = QLabel("GroupBox Demo - Organize related widgets")
        self.label_result.setObjectName("label_result")

        # === MAIN LAYOUT ===
        main = QVBoxLayout(self)
        main.setContentsMargins(20, 20, 20, 20)
        main.setSpacing(15)
        main.addWidget(self.group_customer)
        main.addWidget(self.group_delivery)
        main.addWidget(self.group_payment)
        main.addWidget(self.label_result)
        main.addStretch()

        self.group_delivery.toggled.connect(self.on_group_toggled)
        self.check_express.toggled.connect(self.on_option_changed)
        self.btn_mpesa.clicked.connect(lambda: self.switch_pay(0))
        self.btn_cash.clicked.connect(lambda: self.switch_pay(1))

    @Slot(bool)
    def on_group_toggled(self, checked: bool):
        self.label_result.setText(f"Delivery Options: {'Enabled' if checked else 'Disabled'}")

    @Slot(bool)
    def on_option_changed(self, checked: bool):
        self.label_result.setText(f"Express: {checked}")

    def switch_pay(self, index):
        self.btn_mpesa.setChecked(index == 0)
        self.btn_cash.setChecked(index == 1)
```

group = QGroupBox("Title")

group.setChecked(True)

group.setAlignment(Qt.AlignCenter) # title alignment

layout = QVBoxLayout(group) # set layout directly on group
layout.addWidget(QLabel("Inside group"))

# Signals
group.toggled.connect(lambda checked: print(checked)) # only if checkable
group.clicked.connect(...) # same as toggled
