```python
from pathlib import Path
from PySide6.QtGui import QIcon

ICON_ROOT = Path(__file__).parent.parent / "assets" / "icons"
FALLBACK_ROOT = Path("C:/dev/assets/icons")

def icon(name: str) -> QIcon:
    p = ICON_ROOT / f"{name}.svg"
    if p.exists():
        return QIcon(str(p))
    p2 = FALLBACK_ROOT / f"{name}.svg"
    if p2.exists():
        return QIcon(str(p2))
    print(f"[Icon Missing] {name}.svg")
    return QIcon()
```

`main.py`
```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from widget import Sidebar
from pages import HomePage, AnalyticsPage, SettingsPage

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Begining")
        self.resize(1100, 750)

        self.sidebar = Sidebar()
        self.sidebar.add_page("home", "Dashboard", HomePage(), "home")
        self.sidebar.add_page("analytics", "Analytics", AnalyticsPage(), "chart-bar")
        self.sidebar.add_page("projects", "Projects", HomePage(), "folder")
        self.sidebar.add_separator("System")
        self.sidebar.add_page("settings", "Settings", SettingsPage(), "cog", at_bottom=True)

        self.setCentralWidget(self.sidebar)

        try:
            with open("style.qss", "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```

`widget.py`
```python
from PySide6.QtWidgets import (
    QWidget, QHBoxLayout, QVBoxLayout, QListWidget,
    QListWidgetItem, QStackedWidget, QPushButton, QFrame
)
from PySide6.QtCore import Qt, QSize, Signal, QPropertyAnimation, QEasingCurve
from core.icons import icon

class Sidebar(QWidget):
    pageChanged = Signal(str)

    def __init__(self, expanded_width=240, collapsed_width=68):
        super().__init__()
        self.setAttribute(Qt.WidgetAttribute.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        self._expanded_width = expanded_width
        self._collapsed_width = collapsed_width
        self._is_collapsed = False
        self._pages = {}
        self._titles = {}

        root = QHBoxLayout(self)
        root.setContentsMargins(0,0,0,0)
        root.setSpacing(0)

        # Left container
        self.nav_container = QFrame()
        self.nav_container.setObjectName("navContainer")
        nav_layout = QVBoxLayout(self.nav_container)
        nav_layout.setContentsMargins(0,0,0,0)
        nav_layout.setSpacing(0)

        self.nav = QListWidget()
        self.nav.setObjectName("sidebar")
        self.nav.setFixedWidth(expanded_width)
        self.nav.setIconSize(QSize(20, 20))
        self.nav.setFocusPolicy(Qt.FocusPolicy.NoFocus)
        self.nav.setVerticalScrollBarPolicy(Qt.ScrollBarPolicy.ScrollBarAlwaysOff)
        self.nav.setHorizontalScrollBarPolicy(Qt.ScrollBarPolicy.ScrollBarAlwaysOff)

        self.bottom_nav = QListWidget()
        self.bottom_nav.setObjectName("sidebar")
        self.bottom_nav.setFixedHeight(60)
        self.bottom_nav.setIconSize(QSize(20, 20))
        self.bottom_nav.setFocusPolicy(Qt.FocusPolicy.NoFocus)
        self.bottom_nav.setVerticalScrollBarPolicy(Qt.ScrollBarPolicy.ScrollBarAlwaysOff)

        self.collapse_btn = QPushButton(" Collapse")
        self.collapse_btn.setObjectName("collapseBtn")
        self.collapse_btn.setIcon(icon("chevron-left"))
        self.collapse_btn.setIconSize(QSize(16, 16))
        self.collapse_btn.setFixedHeight(44)
        self.collapse_btn.setCursor(Qt.CursorShape.PointingHandCursor)
        self.collapse_btn.clicked.connect(self.toggle)

        nav_layout.addWidget(self.nav)
        nav_layout.addStretch()
        nav_layout.addWidget(self.bottom_nav)
        nav_layout.addWidget(self.collapse_btn)

        self.stack = QStackedWidget()
        self.stack.setObjectName("content")

        root.addWidget(self.nav_container)
        root.addWidget(self.stack)

        self.nav.currentRowChanged.connect(self._on_main_changed)
        self.bottom_nav.currentRowChanged.connect(self._on_bottom_changed)

        self.anim_min = QPropertyAnimation(self.nav_container, b"minimumWidth")
        self.anim_min.setDuration(250)
        self.anim_min.setEasingCurve(QEasingCurve.Type.OutCubic)
        self.anim_max = QPropertyAnimation(self.nav_container, b"maximumWidth")
        self.anim_max.setDuration(250)
        self.anim_max.setEasingCurve(QEasingCurve.Type.OutCubic)

    def add_page(self, key: str, title: str, widget: QWidget, icon_name: str, at_bottom=False):
        target = self.bottom_nav if at_bottom else self.nav
        dict_key = 1000 + target.count() if at_bottom else target.count()

        self._titles[dict_key] = title
        item = QListWidgetItem(icon(icon_name), f" {title}")
        item.setData(Qt.ItemDataRole.UserRole, key)
        item.setSizeHint(QSize(0, 44))
        item.setToolTip(title)

        target.addItem(item)
        self.stack.addWidget(widget)
        self._pages[key] = self.stack.count() - 1

        if self.stack.count() == 1:
            target.setCurrentRow(0)

    def add_separator(self, label=""):
        row = self.nav.count()
        self._titles[row] = label
        item = QListWidgetItem(f" {label.upper()}")
        item.setFlags(item.flags() & ~Qt.ItemFlag.ItemIsSelectable)
        item.setData(Qt.ItemDataRole.UserRole, None)
        item.setSizeHint(QSize(0, 36))
        self.nav.addItem(item)

    def _on_main_changed(self, row):
        if row < 0: return
        item = self.nav.item(row)
        if not item or item.data(Qt.ItemDataRole.UserRole) is None: return
        self.bottom_nav.clearSelection()
        self.stack.setCurrentIndex(self._pages[item.data(Qt.ItemDataRole.UserRole)])
        self.pageChanged.emit(item.data(Qt.ItemDataRole.UserRole))

    def _on_bottom_changed(self, row):
        if row < 0: return
        item = self.bottom_nav.item(row)
        if not item: return
        self.nav.clearSelection()
        self.stack.setCurrentIndex(self._pages[item.data(Qt.ItemDataRole.UserRole)])
        self.pageChanged.emit(item.data(Qt.ItemDataRole.UserRole))

    def toggle(self):
        self.set_collapsed(not self._is_collapsed)

    def set_collapsed(self, collapsed: bool):
        if self._is_collapsed == collapsed: return
        self._is_collapsed = collapsed
        target = self._collapsed_width if collapsed else self._expanded_width

        for r in range(self.nav.count()):
            it = self.nav.item(r)
            title = self._titles.get(r, "")
            if it.data(Qt.ItemDataRole.UserRole) is None:
                it.setText("" if collapsed else f" {title.upper()}")
            else:
                it.setText("" if collapsed else f" {title}")

        for r in range(self.bottom_nav.count()):
            it = self.bottom_nav.item(r)
            title = self._titles.get(1000 + r, "")
            it.setText("" if collapsed else f" {title}")

        self.collapse_btn.setIcon(icon("chevron-right") if collapsed else icon("chevron-left"))
        self.collapse_btn.setText("" if collapsed else " Collapse")

        self.anim_min.stop()
        self.anim_max.stop()
        self.anim_min.setStartValue(self.nav_container.width())
        self.anim_min.setEndValue(target)
        self.anim_max.setStartValue(self.nav_container.width())
        self.anim_max.setEndValue(target)
        self.anim_min.start()
        self.anim_max.start()
```

```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel
from PySide6.QtCore import Qt

class BasePage(QWidget):
    def __init__(self, text):
        super().__init__()
        self.setAttribute(Qt.WidgetAttribute.WA_StyledBackground, True)
        self.setObjectName("page")
        lay = QVBoxLayout(self)
        lay.setContentsMargins(30, 30, 30, 30)
        label = QLabel(text)
        label.setObjectName("title")
        label.setAlignment(Qt.AlignmentFlag.AlignCenter)
        lay.addWidget(label)
        lay.addStretch()

class HomePage(BasePage):
    def __init__(self): super().__init__("DASHBOARD TITLE")
class AnalyticsPage(BasePage):
    def __init__(self): super().__init__("ANALYTICS")
class SettingsPage(BasePage):
    def __init__(self): super().__init__("SETTINGS")
```
