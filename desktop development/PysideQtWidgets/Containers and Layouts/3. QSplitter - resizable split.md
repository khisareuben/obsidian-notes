```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QSplitter, QFrame, QLabel, QPushButton, QTextEdit
from PySide6.QtCore import Qt, Slot

class SplitterWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE PANELS ===

        # Left Panel
        self.panel_left = QFrame()
        self.panel_left.setObjectName("panel_left")
        l_left = QVBoxLayout(self.panel_left)
        l_left.addWidget(QLabel("📋 Orders"))
        for i in range(4):
            btn = QPushButton(f"Order #{1024+i}")
            btn.setObjectName("btn_order")
            l_left.addWidget(btn)
        l_left.addStretch()

        # Right Top
        self.panel_top = QFrame()
        self.panel_top.setObjectName("panel_top")
        l_top = QVBoxLayout(self.panel_top)
        l_top.addWidget(QLabel("🗺️ Map View"))
        lbl = QLabel("Live map area - draggable resize")
        lbl.setObjectName("content")
        l_top.addWidget(lbl)
        l_top.addStretch()

        # Right Bottom
        self.panel_bottom = QFrame()
        self.panel_bottom.setObjectName("panel_bottom")
        l_bottom = QVBoxLayout(self.panel_bottom)
        l_bottom.addWidget(QLabel("📝 Details"))
        self.text = QTextEdit("Order details here...\nDrag the splitter handles to resize")
        self.text.setObjectName("text_details")
        l_bottom.addWidget(self.text)

        # === SPLITTERS ===

        # Vertical splitter for right side (top/bottom)
        self.splitter_right = QSplitter(Qt.Orientation.Vertical)
        self.splitter_right.setObjectName("splitter")
        self.splitter_right.addWidget(self.panel_top)
        self.splitter_right.addWidget(self.panel_bottom)
        self.splitter_right.setSizes([200, 150]) # initial sizes

        # Horizontal splitter for main (left / right)
        self.splitter_main = QSplitter(Qt.Orientation.Horizontal)
        self.splitter_main.setObjectName("splitter")
        self.splitter_main.addWidget(self.panel_left)
        self.splitter_main.addWidget(self.splitter_right)
        self.splitter_main.setSizes([150, 350]) # left 150, right 350

        # Enable collapsible
        self.splitter_main.setCollapsible(0, False) # left cannot be fully collapsed
        self.splitter_main.setChildrenCollapsible(False)

        # Handle width
        self.splitter_main.setHandleWidth(6)
        self.splitter_right.setHandleWidth(6)

        self.label_result = QLabel("Drag the lines between panels to resize")
        self.label_result.setObjectName("label_result")

        # === MAIN LAYOUT ===
        main = QVBoxLayout(self)
        main.setContentsMargins(10, 10, 10, 10)
        main.setSpacing(8)
        main.addWidget(self.splitter_main, 1)
        main.addWidget(self.label_result)

        # === STYLING ===
        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel { color: #FFF; font-weight: bold; background: transparent; font-size: 13px; }
            QLabel#content { color: #888; font-weight: normal; font-size: 12px; }
            QLabel#label_result {
                color: #00FF88; font-size: 11px; background-color: #1E1E1E;
                padding: 6px; border-radius: 6px; border: 1px solid #333; font-weight: normal;
            }

            QFrame#panel_left {
                background-color: #1E1E1E;
                border: 1px solid #333;
                border-radius: 8px;
            }
            QFrame#panel_top {
                background-color: #1E1E1E;
                border: 1px solid #333;
                border-radius: 8px;
            }
            QFrame#panel_bottom {
                background-color: #1E1E1E;
                border: 1px solid #333;
                border-radius: 8px;
            }

            QPushButton#btn_order {
                background-color: #2A2A2A; color: #CCC;
                padding: 8px; border: 1px solid #333; border-radius: 6px;
                text-align: left; font-size: 12px;
            }
            QPushButton#btn_order:hover {
                background-color: #2E86AB; color: white; border: 1px solid #2E86AB;
            }

            QTextEdit#text_details {
                background-color: #2A2A2A; color: #CCC;
                border: 1px solid #333; border-radius: 6px; padding: 5px;
            }

            /* === SPLITTER HANDLE === */
            QSplitter#splitter::handle {
                background-color: #121212;
                border-radius: 2px;
            }
            QSplitter#splitter::handle:horizontal {
                width: 6px;
                margin: 20px 2px;
                background-color: #333;
                border-radius: 3px;
            }
            QSplitter#splitter::handle:vertical {
                height: 6px;
                margin: 2px 20px;
                background-color: #333;
                border-radius: 3px;
            }
            QSplitter#splitter::handle:hover {
                background-color: #2E86AB;
            }
            QSplitter#splitter::handle:pressed {
                background-color: #2E86AB;
            }
        """)

        # === CONNECT ===
        self.splitter_main.splitterMoved.connect(self.on_moved)

    @Slot(int, int)
    def on_moved(self, pos, index):
        sizes = self.splitter_main.sizes()
        self.label_result.setText(f"Sizes: Left={sizes[0]}px | Right={sizes[1]}px")
```

core api

```python
# Horizontal: left | right
splitter = QSplitter(Qt.Orientation.Horizontal)
# Vertical: top / bottom
splitter = QSplitter(Qt.Orientation.Vertical)

splitter.addWidget(widget1)
splitter.addWidget(widget2)

# Initial sizes in pixels
splitter.setSizes([150, 350])

# Get current sizes
splitter.sizes() # [150, 350]

# Settings
splitter.setHandleWidth(6) # thickness of drag handle
splitter.setCollapsible(0, False) # prevent first widget from collapsing to 0
splitter.setChildrenCollapsible(False) # none can collapse
splitter.setOpaqueResize(True) # True = live resize, False = outline while dragging

# Save/restore state
state = splitter.saveState()
splitter.restoreState(state)

# Signal
splitter.splitterMoved.connect(lambda pos, index: print(pos, index))
```
