`main_window.py`
```python
from pathlib import Path
from PySide6.QtWidgets import QMainWindow, QStatusBar, QLabel, QProgressBar
from PySide6.QtCore import Qt
from widgets import ContentWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("StatusBar - Professional")
        self.resize(1000, 700)

        self._setup_central_widget()
        self._setup_status_bar()
        self._load_stylesheet()
        self._setup_connections()

    def _setup_central_widget(self):
        self.content = ContentWidget()
        self.setCentralWidget(self.content)

    def _setup_status_bar(self):
        statusbar = QStatusBar(self)
        statusbar.setObjectName("mainStatusBar")
        self.setStatusBar(statusbar)

        # Left side: temporary message
        statusbar.showMessage("Ready", 0)

        # Right side: permanent widgets - order matters
        self.progress = QProgressBar(self)
        self.progress.setObjectName("statusProgress")
        self.progress.setFixedWidth(120)
        self.progress.setValue(70)
        self.progress.setTextVisible(False)

        self.version = QLabel("v1.0.2")
        self.version.setObjectName("versionLabel")

        self.status_pill = QLabel("Active")
        self.status_pill.setObjectName("statusPill")

        # addWidget = left side, addPermanentWidget = right side
        statusbar.addWidget(QLabel("  ")) # spacer
        statusbar.addPermanentWidget(self.progress)
        statusbar.addPermanentWidget(self.version)
        statusbar.addPermanentWidget(self.status_pill)

    def _setup_connections(self):
        self.content.action_triggered.connect(
            lambda msg: self.statusBar().showMessage(msg, 3000)
        )

    def _load_stylesheet(self):
        path = Path(__file__).parent / "style.qss"
        if path.exists():
            self.setStyleSheet(path.read_text())
```

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QPushButton
from PySide6.QtCore import Qt, Signal

class ContentWidget(QWidget):
    action_triggered = Signal(str)

    def __init__(self):
        super().__init__()
        self.setObjectName("centralWidget")
        self.setAttribute(Qt.WA_StyledBackground, True)

        self.title = QLabel("DASHBOARD", self)
        self.title.setObjectName("title")
        self.title.setAlignment(Qt.AlignCenter)

        self.btn = QPushButton("Do Something", self)
        self.btn.setObjectName("primaryButton")
        self.btn.clicked.connect(lambda: self.action_triggered.emit("Action completed"))

        layout = QVBoxLayout(self)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.setSpacing(15)
        layout.addWidget(self.title)
        layout.addStretch()
        layout.addWidget(self.btn)
```
