`main_window.py`
```python
from pathlib import Path
from PySide6.QtWidgets import QMainWindow, QMenuBar, QMenu
from PySide6.QtGui import QAction, QKeySequence
from widgets import ContentWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("MenuBar - Professional")
        self.resize(1000, 700)

        self._setup_central_widget()
        self._setup_menu_bar()
        self._load_stylesheet()

    def _setup_central_widget(self):
        self.content = ContentWidget()
        self.setCentralWidget(self.content)

    def _setup_menu_bar(self):
        menubar = QMenuBar(self)
        menubar.setObjectName("mainMenuBar")
        self.setMenuBar(menubar)

        # File
        file_menu = QMenu("&File", self)
        self.act_new = QAction("&New", self)
        self.act_new.setShortcut(QKeySequence.New)
        self.act_new.triggered.connect(lambda: print("New"))

        self.act_open = QAction("&Open...", self)
        self.act_open.setShortcut(QKeySequence.Open)

        self.act_exit = QAction("E&xit", self)
        self.act_exit.setShortcut(QKeySequence.Quit)
        self.act_exit.triggered.connect(self.close)

        file_menu.addAction(self.act_new)
        file_menu.addAction(self.act_open)
        file_menu.addSeparator()
        file_menu.addAction(self.act_exit)

        # Edit
        edit_menu = QMenu("&Edit", self)
        edit_menu.addAction("Undo").setShortcut(QKeySequence.Undo)
        edit_menu.addAction("Redo").setShortcut(QKeySequence.Redo)

        menubar.addMenu(file_menu)
        menubar.addMenu(edit_menu)
        menubar.addMenu("&Help")

    def _load_stylesheet(self):
        path = Path(__file__).parent / "style.qss"
        if path.exists():
            self.setStyleSheet(path.read_text())
```

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel
from PySide6.QtCore import Qt

class ContentWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setObjectName("centralWidget")
        self.setAttribute(Qt.WA_StyledBackground, True)
        label = QLabel("Check the MenuBar above", self)
        label.setObjectName("title")
        label.setAlignment(Qt.AlignCenter)
        layout = QVBoxLayout(self)
        layout.addWidget(label)
```
