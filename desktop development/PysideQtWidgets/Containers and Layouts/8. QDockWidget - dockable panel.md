```python
from PySide6.QtWidgets import QMainWindow, QWidget, QVBoxLayout, QLabel, QDockWidget, QListWidget, QPushButton, QTextEdit
from PySide6.QtCore import Qt, Slot
from PySide6.QtGui import QAction

class DockWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("29. QDockWidget - VS Code Style")
        self.setFixedSize(800, 500)

        # === CENTRAL ===
        central = QWidget()
        central.setObjectName("central")
        l = QVBoxLayout(central)
        l.addWidget(QLabel("Central Editor\nDocks can be dragged, floated, tabbed, closed"))
        self.text_main = QTextEdit("Main content area...\n\nTry:\n- Drag docks to edges\n- Double-click title to float\n- Drag one dock onto another to tab them")
        self.text_main.setObjectName("text_main")
        l.addWidget(self.text_main)
        self.setCentralWidget(central)

        # === DOCK 1: Orders (Left) ===
        self.dock_orders = QDockWidget("📦 Orders", self)
        self.dock_orders.setObjectName("dock_orders")
        self.dock_orders.setAllowedAreas(Qt.DockWidgetArea.LeftDockWidgetArea | Qt.DockWidgetArea.RightDockWidgetArea)

        orders_widget = QWidget()
        ol = QVBoxLayout(orders_widget)
        self.list_orders = QListWidget()
        self.list_orders.setObjectName("list")
        self.list_orders.addItems([f"Order #{1024+i}" for i in range(10)])
        ol.addWidget(self.list_orders)

        self.dock_orders.setWidget(orders_widget)

        # === DOCK 2: Map (Right Top) ===
        self.dock_map = QDockWidget("🗺️ Map", self)
        self.dock_map.setObjectName("dock_map")
        map_widget = QWidget()
        ml = QVBoxLayout(map_widget)
        ml.addWidget(QLabel("Live Tracking"))
        ml.addWidget(QLabel("Driver: Ali - 2km away"))
        map_widget.setStyleSheet("QLabel { color: #888; }")
        self.dock_map.setWidget(map_widget)

        # === DOCK 3: Logs (Bottom) ===
        self.dock_logs = QDockWidget("📝 Logs", self)
        self.dock_logs.setObjectName("dock_logs")
        logs_widget = QWidget()
        ll = QVBoxLayout(logs_widget)
        self.text_logs = QTextEdit("12:00 Order created\n12:05 Driver assigned\n12:10 Picked up")
        self.text_logs.setObjectName("text_logs")
        ll.addWidget(self.text_logs)
        self.dock_logs.setWidget(logs_widget)

        # === ADD DOCKS TO MAIN WINDOW ===
        self.addDockWidget(Qt.DockWidgetArea.LeftDockWidgetArea, self.dock_orders)
        self.addDockWidget(Qt.DockWidgetArea.RightDockWidgetArea, self.dock_map)
        self.addDockWidget(Qt.DockWidgetArea.BottomDockWidgetArea, self.dock_logs)

        # Tabify: stack map and another dock as tabs
        # self.tabifyDockWidget(self.dock_map, self.dock_other)

        # Split: place one dock beside another
        self.splitDockWidget(self.dock_orders, self.dock_logs, Qt.Orientation.Horizontal)

        # Features
        for dock in [self.dock_orders, self.dock_map, self.dock_logs]:
            dock.setFeatures(QDockWidget.DockWidgetFeature.DockWidgetMovable |
                             QDockWidget.DockWidgetFeature.DockWidgetFloatable |
                             QDockWidget.DockWidgetFeature.DockWidgetClosable)

        # === VIEW MENU TO TOGGLE DOCKS ===
        view_menu = self.menuBar().addMenu("View")
        view_menu.addAction(self.dock_orders.toggleViewAction())
        view_menu.addAction(self.dock_map.toggleViewAction())
        view_menu.addAction(self.dock_logs.toggleViewAction())

        # === STYLING ===
        self.setStyleSheet("""
            QMainWindow { background-color: #121212; }
            QWidget#central { background-color: #121212; }
            QLabel { color: #FFF; font-weight: bold; font-size: 13px; }
            QTextEdit#text_main {
                background-color: #1E1E1E; color: #CCC;
                border: 1px solid #333; border-radius: 8px; padding: 10px;
            }
            QTextEdit#text_logs {
                background-color: #1A1A1A; color: #00FF88;
                border: 1px solid #333; border-radius: 6px; font-size: 11px;
            }
            QListWidget#list {
                background-color: #1E1E1E; color: #CCC;
                border: 1px solid #333; border-radius: 6px;
            }
            QListWidget#list::item { padding: 8px; border-radius: 4px; }
            QListWidget#list::item:selected { background-color: #2E86AB; color: white; }
            QListWidget#list::item:hover { background-color: #2A2A2A; }

            /* === DOCK WIDGET === */
            QDockWidget {
                background-color: #1E1E1E;
                color: #FFF;
                titlebar-close-icon: url(none);
                titlebar-normal-icon: url(none);
            }
            QDockWidget::title {
                background-color: #1E1E1E;
                padding: 8px;
                border: 1px solid #333;
                border-radius: 6px;
                text-align: left;
            }
            QDockWidget::close-button, QDockWidget::float-button {
                background-color: #2A2A2A;
                border: 1px solid #333;
                border-radius: 4px;
                padding: 4px;
            }
            QDockWidget::close-button:hover, QDockWidget::float-button:hover {
                background-color: #E74C3C;
            }
        """)

        self.list_orders.currentRowChanged.connect(self.on_order_select)

    @Slot(int)
    def on_order_select(self, row):
        self.text_main.append(f"Selected Order #{1024+row}")
        self.text_logs.append(f"12:{15+row} - Viewing Order #{1024+row}")
```

```python
dock = QDockWidget("Title", parent)

widget = QWidget() # your content
dock.setWidget(widget)

# Where can it be docked?
dock.setAllowedAreas(Qt.LeftDockWidgetArea | Qt.RightDockWidgetArea)
# All areas: Left, Right, Top, Bottom, AllDockWidgetAreas

# Add to main window
self.addDockWidget(Qt.LeftDockWidgetArea, dock)

# Features - what user can do
dock.setFeatures(
    QDockWidget.DockWidgetMovable | # can drag
    QDockWidget.DockWidgetFloatable | # can float as window
    QDockWidget.DockWidgetClosable # can close [x]
)
# To lock: setFeatures(QDockWidget.NoDockWidgetFeatures)

# Pro functions
self.tabifyDockWidget(dock1, dock2) # stack as tabs
self.splitDockWidget(dock1, dock2, Qt.Horizontal) # side by side
dock.setFloating(True) # float as separate window

# Toggle action for View menu
view_menu.addAction(dock.toggleViewAction()) # auto checkbox

# Signals
dock.visibilityChanged.connect(lambda v: print(v))
dock.topLevelChanged.connect(lambda floating: print(f"Floating: {floating}"))
dock.dockLocationChanged.connect(lambda area: print(area))

# Save/Restore layout
state = self.saveState()
self.restoreState(state)
geometry = self.saveGeometry()
```
