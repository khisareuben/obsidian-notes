
Is the simplest widget in Qt. It's just a box/line. Every thing like QLabel, QGroupBox actually inherits form QFrame
You use it for 2 things:
- Draw a line separator
- Draw a panel/border around widgets

`main.py`
```python
import sys
from pathlib import Path
from PySide6.QtWidgets import QApplication, QMainWindow
from widget import FrameWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setCentralWidget(FrameWidget())

        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.showMaximized()
    sys.exit(app.exec())
```

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout, QFrame, QLabel, QPushButton
from PySide6.QtCore import Slot

class FrameWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === FRAME 1: Card (Box + Raised) ===
        self.frame_card = QFrame()
        self.frame_card.setObjectName("frame_card")
        self.frame_card.setFrameShape(QFrame.Shape.StyledPanel) # or Box, Panel
        self.frame_card.setFrameShadow(QFrame.Shadow.Sunken) # Raised, Sunken

        label_card_title = QLabel("📦 Order Summary")
        label_card_title.setObjectName("title")
        label_card_content = QLabel("Order #1024\nMombasa CBD -> Nyali\nKES 1,200")
        label_card_content.setObjectName("content")

        layout_card = QVBoxLayout(self.frame_card)
        layout_card.addWidget(label_card_title)
        layout_card.addWidget(label_card_content)

        # === SEPARATOR 1: Horizontal Line ===
        self.line_h = QFrame()
        self.line_h.setObjectName("line_h")
        self.line_h.setFrameShape(QFrame.Shape.HLine)
        self.line_h.setFrameShadow(QFrame.Shadow.Sunken)

        # === FRAME 2: Different shapes demo ===
        self.frame_box = QFrame()
        self.frame_box.setObjectName("frame_box")
        self.frame_box.setFrameShape(QFrame.Shape.Box)
        self.frame_box.setLineWidth(2)

        lbl_box = QLabel("Box Shape - LineWidth 2")
        lbl_box.setObjectName("small")
        l_box = QHBoxLayout(self.frame_box)
        l_box.addWidget(lbl_box)

        self.frame_panel = QFrame()
        self.frame_panel.setObjectName("frame_panel")
        self.frame_panel.setFrameShape(QFrame.Shape.StyledPanel)
        self.frame_panel.setFrameShadow(QFrame.Shadow.Sunken)
        self.frame_panel.setLineWidth(2)

        lbl_panel = QLabel("Panel + Sunken")
        lbl_panel.setObjectName("small")
        l_panel = QHBoxLayout(self.frame_panel)
        l_panel.addWidget(lbl_panel)

        # === SEPARATOR 2: Vertical ===
        self.line_v = QFrame()
        self.line_v.setObjectName("line_v")
        self.line_v.setFrameShape(QFrame.Shape.VLine)
        self.line_v.setFrameShadow(QFrame.Shadow.Sunken)

        row = QHBoxLayout()
        row.addWidget(self.frame_box)
        row.addWidget(self.line_v)
        row.addWidget(self.frame_panel)

        # === FRAME 3: NoFrame - just container ===
        self.frame_container = QFrame()
        self.frame_container.setObjectName("frame_container")

        btn = QPushButton("Action in Frame")
        btn.setObjectName("btn_action")
        l_cont = QVBoxLayout(self.frame_container)
        l_cont.addWidget(QLabel("Container Frame (No border, just background)"))
        l_cont.addWidget(btn)

        # === MAIN LAYOUT ===
        main = QVBoxLayout(self)
        main.setContentsMargins(20, 20, 20, 20)
        main.setSpacing(15)
        main.addWidget(self.frame_card)
        main.addWidget(self.line_h)
        main.addLayout(row)
        main.addWidget(self.frame_container)
        main.addStretch()

    @Slot()
    def on_click(self):
        print("Clicked")
```

must know qframe shapes:

```python
frame = QFrame()

# Shape: What border type
QFrame.Shape.NoFrame # no border
QFrame.Shape.Box # rectangle
QFrame.Shape.Panel # panel (most common)
QFrame.Shape.StyledPanel # styled panel (uses style)
QFrame.Shape.HLine # horizontal line (separator)
QFrame.Shape.VLine # vertical line (separator)
QFrame.Shape.WinPanel # windows style panel

# Shadow: 3D effect
QFrame.Shadow.Plain # flat
QFrame.Shadow.Raised # raised (3D up)
QFrame.Shadow.Sunken # sunken (3D down)

# How to use
frame.setFrameShape(QFrame.Shape.Box)
frame.setFrameShadow(QFrame.Shadow.Raised)
frame.setLineWidth(2) # border thickness
frame.setMidLineWidth(1) # extra line for Box/Panel

# Separator shortcut
h_line = QFrame()
h_line.setFrameShape(QFrame.Shape.HLine)
h_line.setFrameShadow(QFrame.Shadow.Sunken)
```
