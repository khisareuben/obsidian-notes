from PySide6.QtWidgets import QWidget, QVBoxLayout, QScrollArea, QFrame, QLabel, QPushButton, QGridLayout
from PySide6.QtCore import Qt, Slot

class ScrollWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CONTENT WIDGET (what goes INSIDE scroll) ===
        self.content = QWidget()
        self.content.setObjectName("content_widget")
        self.content_layout = QGridLayout(self.content)
        self.content_layout.setSpacing(10)
        self.content_layout.setContentsMargins(10, 10, 10, 10)

        # Add many cards to force scrolling
        for i in range(20):
            card = QFrame()
            card.setObjectName("card")
            card.setFixedHeight(70)
            l = QVBoxLayout(card)
            l.addWidget(QLabel(f"📦 Order #{1024+i}"))
            sub = QLabel(f"Mombasa -> Nyali | KES {1200 + i*100}")
            sub.setObjectName("sub")
            l.addWidget(sub)

            row = i // 2
            col = i % 2
            self.content_layout.addWidget(card, row, col)

        # === SCROLL AREA ===
        self.scroll = QScrollArea()
        self.scroll.setObjectName("scroll")
        self.scroll.setWidget(self.content) # Put content inside
        self.scroll.setWidgetResizable(True) # CRITICAL - makes content expand
        self.scroll.setHorizontalScrollBarPolicy(Qt.ScrollBarPolicy.ScrollBarAlwaysOff)
        self.scroll.setVerticalScrollBarPolicy(Qt.ScrollBarPolicy.ScrollBarAsNeeded)
        self.scroll.setFrameShape(QFrame.Shape.NoFrame) # No border

        self.label_result = QLabel(f"Total widgets in scroll: 20")
        self.label_result.setObjectName("label_result")

        # === MAIN LAYOUT ===
        main = QVBoxLayout(self)
        main.setContentsMargins(15, 15, 15, 15)
        main.setSpacing(10)
        main.addWidget(QLabel("Delivery Orders - Scrollable"))
        main.addWidget(self.scroll, 1)
        main.addWidget(self.label_result)

        # === STYLING ===
        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel { color: #FFF; font-weight: bold; background: transparent; font-size: 14px; }
            QLabel#sub { color: #888; font-weight: normal; font-size: 11px; }
            QLabel#label_result {
                color: #00FF88; font-size: 11px; background-color: #1E1E1E;
                padding: 6px; border-radius: 6px; border: 1px solid #333; font-weight: normal;
            }

            QWidget#content_widget {
                background-color: #121212;
            }

            QFrame#card {
                background-color: #1E1E1E;
                border: 1px solid #333;
                border-radius: 8px;
                padding: 5px;
            }
            QFrame#card:hover {
                border: 1px solid #2E86AB;
            }

            /* Scroll Area */
            QScrollArea#scroll {
                background-color: #121212;
                border: 1px solid #333;
                border-radius: 10px;
            }

            /* Scrollbars */
            QScrollBar:vertical {
                background-color: #121212;
                width: 10px;
                border-radius: 5px;
                margin: 0px;
            }
            QScrollBar::handle:vertical {
                background-color: #333;
                border-radius: 5px;
                min-height: 30px;
                margin: 2px;
            }
            QScrollBar::handle:vertical:hover {
                background-color: #2E86AB;
            }
            QScrollBar::add-line:vertical, QScrollBar::sub-line:vertical {
                height: 0px;
                background: none;
                border: none;
            }
            QScrollBar::add-page:vertical, QScrollBar::sub-page:vertical {
                background: #121212;
            }
            QScrollBar:horizontal { height: 0px; }
        """)

    @Slot(int)
    def on_scroll(self, value):
        self.label_result.setText(f"Scroll pos: {value}")
