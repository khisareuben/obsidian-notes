import sys
from pathlib import Path
from PySide6.QtWidgets import QApplication, QMainWindow
from widget import ScrollWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setCentralWidget(ScrollWidget())

        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.showMaximized()
    sys.exit(app.exec())

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QScrollArea, QFrame, QLabel, QPushButton, QGridLayout
from PySide6.QtCore import Qt, Slot

class ScrollWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CONTENT WIDGET (what goes INSIDE scroll) ===
        self.content = QWidget()
        self.content.setObjectName("content_widget")
        self.content_layout = QGridLayout(self.content)
        self.content_layout.setSpacing(10)
        self.content_layout.setContentsMargins(10, 10, 10, 10)

        # Add many cards to force scrolling
        for i in range(20):
            card = QFrame()
            card.setObjectName("card")
            card.setFixedHeight(70)
            l = QVBoxLayout(card)
            l.addWidget(QLabel(f"📦 Order #{1024+i}"))
            sub = QLabel(f"Mombasa -> Nyali | KES {1200 + i*100}")
            sub.setObjectName("sub")
            l.addWidget(sub)

            #2 for 2columns if you want 3 col use 3 and so on
            row = i // 2
            col = i % 2
            self.content_layout.addWidget(card, row, col)

        # === SCROLL AREA ===
        scroll = QScrollArea()
        scroll.setObjectName("scroll")
        scroll.setWidget(self.content) # Put content inside
        scroll.setWidgetResizable(True) # CRITICAL - makes content expand
        scroll.setHorizontalScrollBarPolicy(Qt.ScrollBarPolicy.ScrollBarAlwaysOff)
        scroll.setVerticalScrollBarPolicy(Qt.ScrollBarPolicy.ScrollBarAsNeeded)
        scroll.setFrameShape(QFrame.Shape.NoFrame) # No border

        self.label_result = QLabel(f"Total widgets in scroll: 20")
        self.label_result.setObjectName("label_result")

        # === MAIN LAYOUT ===
        main = QVBoxLayout(self)
        main.setContentsMargins(15, 15, 15, 15)
        main.setSpacing(10)
        main.addWidget(QLabel("Delivery Orders - Scrollable"))
        main.addWidget(scroll, 1)
        main.addWidget(self.label_result)
```
