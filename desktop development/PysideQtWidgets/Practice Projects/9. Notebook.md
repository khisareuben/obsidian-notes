`database.py`
```python
import sqlite3
from pathlib import Path

DB_PATH = Path(__file__).resolve().parent / "notes.db"

def get_connection():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS notes (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            title TEXT NOT NULL,
            content TEXT NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)
    conn.commit()
    conn.close()
```

`repository.py`
```python
from database import get_connection

def create_note(title: str, content: str) -> int:
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("INSERT INTO notes (title, content) VALUES (?, ?)", (title, content))
    conn.commit()
    new_id = cursor.lastrowid
    conn.close()
    return new_id

def get_all_notes():
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("SELECT id, title, content FROM notes ORDER BY id DESC")
    rows = cursor.fetchall()
    conn.close()
    return rows

def update_note(note_id: int, title: str, content: str):
    conn = get_connection()
    cursor = conn.cursor() # is like a hand that does the actual work
    cursor.execute("UPDATE notes SET title = ?, content = ? WHERE id = ?", (title, content, note_id))
    conn.commit()
    conn.close()

def delete_note(note_id: int):
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("DELETE FROM notes WHERE id = ?", (note_id,))
    conn.commit()
    conn.close()
```

```python
from PySide6.QtWidgets import (QWidget, QVBoxLayout, QHBoxLayout, QLabel, 
                               QLineEdit, QTextEdit, QTabWidget, QPushButton)
from PySide6.QtCore import Qt, Slot
import repository

class NotebookTab(QWidget):
    """ Each tab is a single note editor """
    def __init__(self, note_id=None, title="", content=""):
        super().__init__()
        self.note_id = note_id  # None = new unsaved note

        self.input_title = QLineEdit()
        self.input_title.setObjectName("input_name")
        self.input_title.setPlaceholderText("Note Title")
        self.input_title.setText(title)

        self.text_content = QTextEdit()
        self.text_content.setObjectName("text_content")
        self.text_content.setPlaceholderText("Write your note here...")
        self.text_content.setText(content)

        layout = QVBoxLayout(self)
        layout.setContentsMargins(10, 10, 10, 10)
        layout.setSpacing(10)
        layout.addWidget(self.input_title)
        layout.addWidget(self.text_content)

class NotebookWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        self.label_info = QLabel("My Notebook - QTabWidget + QTextEdit")
        self.label_info.setObjectName("label_info")

        self.tab_widget = QTabWidget()
        self.tab_widget.setObjectName("tab_widget")
        self.tab_widget.setTabsClosable(True)
        self.tab_widget.setMovable(True)

        self.btn_new = QPushButton("+ New Tab")
        self.btn_new.setObjectName("btn_add")

        self.btn_save = QPushButton("SAVE / Update DB")
        self.btn_save.setObjectName("btn_save")

        self.btn_delete = QPushButton("DELETE Note")
        self.btn_delete.setObjectName("btn_delete")

        self.label_result = QLabel("Ready")
        self.label_result.setObjectName("label_result")

        # Layout
        main_layout = QVBoxLayout(self)
        main_layout.setSpacing(12)
        main_layout.setContentsMargins(20, 20, 20, 20)

        btn_layout = QHBoxLayout()
        btn_layout.addWidget(self.btn_new)
        btn_layout.addWidget(self.btn_save)
        btn_layout.addWidget(self.btn_delete)

        main_layout.addWidget(self.label_info)
        main_layout.addLayout(btn_layout)
        main_layout.addWidget(self.tab_widget)
        main_layout.addWidget(self.label_result)

        # Connections
        self.btn_new.clicked.connect(self.create_new_tab)
        self.btn_save.clicked.connect(self.save_current_tab)
        self.btn_delete.clicked.connect(self.delete_current_tab)
        self.tab_widget.tabCloseRequested.connect(self.close_tab)

        # Load from DB
        self.load_notes_from_db()

    def load_notes_from_db(self):
        notes = repository.get_all_notes()
        if not notes:
            self.create_new_tab()  # Start with one empty tab if DB empty
            self.label_result.setText("No saved notes - new tab created")
        else:
            for row in notes:
                tab = NotebookTab(note_id=row["id"], title=row["title"], content=row["content"])
                self.tab_widget.addTab(tab, row["title"][:15] or "Untitled")
            self.label_result.setText(f"Loaded {len(notes)} notes from DB")

    @Slot()
    def create_new_tab(self):
        tab = NotebookTab()
        index = self.tab_widget.addTab(tab, "Untitled")
        self.tab_widget.setCurrentIndex(index)
        self.label_result.setText("CREATE: New tab added - write and SAVE")

    @Slot()
    def save_current_tab(self):
        current_tab = self.tab_widget.currentWidget()
        if not current_tab:
            return

        title = current_tab.input_title.text().strip() or "Untitled"
        content = current_tab.text_content.toPlainText()

        if current_tab.note_id is None:
            # CREATE
            new_id = repository.create_note(title, content)
            current_tab.note_id = new_id
            self.label_result.setText(f"CREATED in DB ID {new_id}: {title}")
        else:
            # UPDATE
            repository.update_note(current_tab.note_id, title, content)
            self.label_result.setText(f"UPDATED ID {current_tab.note_id}: {title}")

        # Update tab text
        current_index = self.tab_widget.currentIndex()
        self.tab_widget.setTabText(current_index, title[:15])

    @Slot()
    def delete_current_tab(self):
        current_index = self.tab_widget.currentIndex()
        current_tab = self.tab_widget.currentWidget()
        if not current_tab:
            return

        if current_tab.note_id is not None:
            repository.delete_note(current_tab.note_id)
            self.label_result.setText(f"DELETED ID {current_tab.note_id} from DB")
        
        self.tab_widget.removeTab(current_index)

    @Slot(int)
    def close_tab(self, index):
        tab = self.tab_widget.widget(index)
        # If it has a DB id, we don't auto-delete from DB on close, just close UI
        # User must press DELETE to remove from DB - industry safety
        self.tab_widget.removeTab(index)
        if self.tab_widget.count() == 0:
            self.create_new_tab()
```
