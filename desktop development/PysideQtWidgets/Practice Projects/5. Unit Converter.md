`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QLineEdit, QComboBox, QPushButton
from PySide6.QtCore import Qt, Slot

class ConverterWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        # label
        self.label_info = QLabel("Enter Value To Convert:")
        self.label_info.setObjectName("label_info")

        # input
        self.input_value = QLineEdit()
        self.input_value.setObjectName("input_name")
        self.input_value.setPlaceholderText("e.g. 100")

        # combo box - the new widget
        self.label_combo = QLabel("Conversion Type:")
        self.label_combo.setObjectName("label_info")

        self.combo_type = QComboBox()
        self.combo_type.setObjectName("combo_type")
        self.combo_type.addItems([
            "KM to Miles",
            "Miles to KM",
            "Celsius to Fahrenheit",
            "KG to Pounds"
        ])

        # readonly info
        self.input_readonly = QLabel("Formula will show here")
        self.input_readonly.setObjectName("input_readonly")

        # result
        self.label_result = QLabel("Result will show here")
        self.label_result.setObjectName("label_result")

        # button
        self.btn_get = QPushButton("Convert")
        self.btn_get.setObjectName("btn_get")

        # === Layout ====
        layout = QVBoxLayout(self)
        layout.setSpacing(12)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.addWidget(self.label_info)
        layout.addWidget(self.input_value)
        layout.addWidget(self.label_combo)
        layout.addWidget(self.combo_type)
        layout.addWidget(self.input_readonly)
        layout.addWidget(self.label_result)
        layout.addWidget(self.btn_get)
        layout.addStretch()

        # connections
        self.combo_type.currentTextChanged.connect(self.on_type_changed)
        self.btn_get.clicked.connect(self.on_convert)
        
        # init
        self.on_type_changed(self.combo_type.currentText())

    @Slot(str)
    def on_type_changed(self, text):
        if text == "KM to Miles":
            self.input_readonly.setText("Formula: Miles = KM * 0.621371")
        elif text == "Miles to KM":
            self.input_readonly.setText("Formula: KM = Miles * 1.60934")
        elif text == "Celsius to Fahrenheit":
            self.input_readonly.setText("Formula: F = (C * 9/5) + 32")
        elif text == "KG to Pounds":
            self.input_readonly.setText("Formula: Pounds = KG * 2.20462")

    @Slot()
    def on_convert(self):
        try:
            value = float(self.input_value.text())
            conv_type = self.combo_type.currentText()

            if conv_type == "KM to Miles":
                result = value * 0.621371
                self.label_result.setText(f"{value} KM = {result:.2f} Miles")
            elif conv_type == "Miles to KM":
                result = value * 1.60934
                self.label_result.setText(f"{value} Miles = {result:.2f} KM")
            elif conv_type == "Celsius to Fahrenheit":
                result = (value * 9/5) + 32
                self.label_result.setText(f"{value}°C = {result:.2f}°F")
            elif conv_type == "KG to Pounds":
                result = value * 2.20462
                self.label_result.setText(f"{value} KG = {result:.2f} lbs")

        except ValueError:
            self.label_result.setText("Please enter a valid number!")
```
