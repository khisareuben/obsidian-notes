`database.py`
```python
import sqlite3
from pathlib import Path

DB_PATH = Path(__file__).resolve().parent / "settings.db"

def get_connection():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS settings (
            key TEXT PRIMARY KEY,
            value TEXT NOT NULL
        )
    """)
    # Default settings
    defaults = [
        ("notifications", "1"),
        ("auto_save", "1"),
        ("dark_mode", "0"),
        ("theme", "light"), # light, dark, blue
        ("language", "en")
    ]
    for k, v in defaults:
        cursor.execute("INSERT OR IGNORE INTO settings (key, value) VALUES (?,?)", (k, v))
    conn.commit()
    conn.close()
```

`repository.py`
```python
from database import get_connection

def get_setting(key: str, default=""):
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("SELECT value FROM settings WHERE key =?", (key,))
    row = cursor.fetchone()
    conn.close()
    return row["value"] if row else default

def set_setting(key: str, value: str):
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("INSERT OR REPLACE INTO settings (key, value) VALUES (?,?)", (key, value))
    conn.commit()
    conn.close()

def get_all_settings():
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("SELECT key, value FROM settings")
    rows = cursor.fetchall()
    conn.close()
    return {r["key"]: r["value"] for r in rows}
```


`main.py`
```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from pathlib import Path
from database import init_db
from widget import SettingsWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("13 - Settings Panel")
        self.resize(1000, 700)
        self.setCentralWidget(SettingsWidget())
        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            pass

if __name__ == "__main__":
    init_db()
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```

```python
from PySide6.QtWidgets import (QWidget, QVBoxLayout, QHBoxLayout, QLabel,
                               QCheckBox, QRadioButton, QGroupBox, QPushButton)
from PySide6.QtCore import Slot
import repository

class SettingsWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(True, True)
        self.setObjectName("centralWidget")

        self.label_info = QLabel("Settings Panel - QCheckBox + QRadioButton + QGroupBox")
        self.label_info.setObjectName("label_info")

        # === GROUP 1: CheckBoxes [True/False settings] ===
        self.group_checks = QGroupBox("Preferences")
        self.group_checks.setObjectName("group_box")

        self.check_notifications = QCheckBox("Enable Notifications")
        self.check_notifications.setObjectName("check_box")

        self.check_autosave = QCheckBox("Auto-save notes")
        self.check_autosave.setObjectName("check_box")

        self.check_dark = QCheckBox("Dark Mode [demo]")
        self.check_dark.setObjectName("check_box")

        check_layout = QVBoxLayout(self.group_checks)
        check_layout.addWidget(self.check_notifications)
        check_layout.addWidget(self.check_autosave)
        check_layout.addWidget(self.check_dark)

        # === GROUP 2: RadioButtons [One choice only] ===
        self.group_theme = QGroupBox("Theme Selection")
        self.group_theme.setObjectName("group_box")

        self.radio_light = QRadioButton("Light Theme")
        self.radio_light.setObjectName("radio_btn")
        self.radio_dark = QRadioButton("Dark Theme")
        self.radio_dark.setObjectName("radio_btn")
        self.radio_blue = QRadioButton("Blue Theme")
        self.radio_blue.setObjectName("radio_btn")

        theme_layout = QVBoxLayout(self.group_theme)
        theme_layout.addWidget(self.radio_light)
        theme_layout.addWidget(self.radio_dark)
        theme_layout.addWidget(self.radio_blue)

        # === GROUP 3: Language [Another radio group] ===
        self.group_lang = QGroupBox("Language")
        self.group_lang.setObjectName("group_box")

        self.radio_en = QRadioButton("English")
        self.radio_en.setObjectName("radio_btn")
        self.radio_sw = QRadioButton("Swahili")
        self.radio_sw.setObjectName("radio_btn")

        lang_layout = QVBoxLayout(self.group_lang)
        lang_layout.addWidget(self.radio_en)
        lang_layout.addWidget(self.radio_sw)

        # Buttons
        self.btn_save = QPushButton("SAVE Settings to DB")
        self.btn_save.setObjectName("btn_add")

        self.btn_load = QPushButton("LOAD from DB")
        self.btn_load.setObjectName("btn_clear")

        self.label_result = QLabel("Ready")
        self.label_result.setObjectName("label_result")

        # Main Layout
        main_layout = QVBoxLayout(self)
        main_layout.setSpacing(12)
        main_layout.setContentsMargins(20, 20, 20, 20)

        btn_layout = QHBoxLayout()
        btn_layout.addWidget(self.btn_save)
        btn_layout.addWidget(self.btn_load)

        main_layout.addWidget(self.label_info)
        main_layout.addWidget(self.group_checks)
        main_layout.addWidget(self.group_theme)
        main_layout.addWidget(self.group_lang)
        main_layout.addLayout(btn_layout)
        main_layout.addWidget(self.label_result)
        main_layout.addStretch()

        # Connections
        self.btn_save.clicked.connect(self.save_settings)
        self.btn_load.clicked.connect(self.load_settings)

        # Live preview connection
        self.check_notifications.toggled.connect(self.on_preview)
        self.radio_light.toggled.connect(self.on_preview)
        self.radio_dark.toggled.connect(self.on_preview)
        self.radio_blue.toggled.connect(self.on_preview)

        # Initial load from DB
        self.load_settings()

    def load_settings(self):
        """ R - READ from DB and set UI """
        settings = repository.get_all_settings()

        self.check_notifications.setChecked(settings.get("notifications") == "1")
        self.check_autosave.setChecked(settings.get("auto_save") == "1")
        self.check_dark.setChecked(settings.get("dark_mode") == "1")

        theme = settings.get("theme", "light")
        if theme == "light":
            self.radio_light.setChecked(True)
        elif theme == "dark":
            self.radio_dark.setChecked(True)
        else:
            self.radio_blue.setChecked(True)

        lang = settings.get("language", "en")
        self.radio_en.setChecked(lang == "en")
        self.radio_sw.setChecked(lang == "sw")

        self.label_result.setText(f"LOADED from DB: {settings}")

    @Slot()
    def save_settings(self):
        """ C/U - SAVE to DB """
        repository.set_setting("notifications", "1" if self.check_notifications.isChecked() else "0")
        repository.set_setting("auto_save", "1" if self.check_autosave.isChecked() else "0")
        repository.set_setting("dark_mode", "1" if self.check_dark.isChecked() else "0")

        if self.radio_light.isChecked():
            theme = "light"
        elif self.radio_dark.isChecked():
            theme = "dark"
        else:
            theme = "blue"
        repository.set_setting("theme", theme)

        lang = "en" if self.radio_en.isChecked() else "sw"
        repository.set_setting("language", lang)

        self.label_result.setText(f"SAVED: notifications={self.check_notifications.isChecked()}, theme={theme}, lang={lang}")

    @Slot()
    def on_preview(self):
        """ Live feedback without saving """
        theme = "light"
        if self.radio_dark.isChecked():
            theme = "dark"
        elif self.radio_blue.isChecked():
            theme = "blue"

        self.label_result.setText(f"Preview: Theme={theme}, Notifications={'ON' if self.check_notifications.isChecked() else 'OFF'} - Click SAVE to persist")
```
