`widget.py`
```python
import sqlite3
from pathlib import Path
from PySide6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout, QLabel, QLineEdit, QListWidget, QPushButton
from PySide6.QtCore import Qt, Slot

class TodoWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        # --- Database setup ---
        self.db_path = Path(__file__).resolve().parent / "todo.db"
        self.init_database()

        # --- UI Widgets ---
        self.label_info = QLabel("My Tasks - SQLite CRUD (Persists after close)")
        self.label_info.setObjectName("label_info")

        self.input_task = QLineEdit()
        self.input_task.setObjectName("input_name")
        self.input_task.setPlaceholderText("e.g. Learn SQLite")

        self.list_tasks = QListWidget()
        self.list_tasks.setObjectName("list_tasks")

        self.label_result = QLabel("Loading tasks from database...")
        self.label_result.setObjectName("label_result")

        self.btn_add = QPushButton("CREATE / Add")
        self.btn_add.setObjectName("btn_add")

        self.btn_update = QPushButton("UPDATE / Save")
        self.btn_update.setObjectName("btn_update")

        self.btn_delete = QPushButton("DELETE")
        self.btn_delete.setObjectName("btn_delete")

        self.btn_clear = QPushButton("Clear All DB")
        self.btn_clear.setObjectName("btn_clear")

        # === Layout ====
        main_layout = QVBoxLayout(self)
        main_layout.setSpacing(12)
        main_layout.setContentsMargins(20, 20, 20, 20)

        btn_layout = QHBoxLayout()
        btn_layout.addWidget(self.btn_add)
        btn_layout.addWidget(self.btn_update)

        btn_layout2 = QHBoxLayout()
        btn_layout2.addWidget(self.btn_delete)
        btn_layout2.addWidget(self.btn_clear)

        main_layout.addWidget(self.label_info)
        main_layout.addWidget(self.input_task)
        main_layout.addLayout(btn_layout)
        main_layout.addWidget(self.list_tasks)
        main_layout.addLayout(btn_layout2)
        main_layout.addWidget(self.label_result)

        # === Connections ===
        self.btn_add.clicked.connect(self.create_task)
        self.btn_update.clicked.connect(self.update_task)
        self.btn_delete.clicked.connect(self.delete_task)
        self.btn_clear.clicked.connect(self.clear_tasks)
        self.list_tasks.itemClicked.connect(self.load_task_to_input)

        # Initial READ
        self.read_tasks()

    # ========== DATABASE CORE ==========
    def init_database(self):
        """ Create table if not exists """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS tasks (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                text TEXT NOT NULL
            )
        """)
        conn.commit()
        conn.close()

    # ========== CRUD OPERATIONS ==========

    @Slot()
    def read_tasks(self):
        """ R - READ: Get all tasks from DB and show in QListWidget """
        self.list_tasks.clear()
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("SELECT id, text FROM tasks")
        rows = cursor.fetchall()
        conn.close()

        for task_id, text in rows:
            self.list_tasks.addItem(text)
            # Store the DB id inside the list item (hidden)
            item = self.list_tasks.item(self.list_tasks.count() - 1)
            item.setData(Qt.UserRole, task_id)

        self.label_result.setText(f"Loaded {len(rows)} tasks from {self.db_path.name}")

    @Slot()
    def create_task(self):
        """ C - CREATE: INSERT into DB """
        text = self.input_task.text().strip()
        if not text:
            self.label_result.setText("Cannot add empty task!")
            return

        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("INSERT INTO tasks (text) VALUES (?)", (text,))
        conn.commit()
        conn.close()

        self.input_task.clear()
        self.read_tasks()  # Refresh list
        self.label_result.setText(f"Created in DB: '{text}'")

    @Slot()
    def load_task_to_input(self):
        """ Helper - when you click list, load for editing """
        if self.list_tasks.currentItem() is None:
            return
        self.input_task.setText(self.list_tasks.currentItem().text())

    @Slot()
    def update_task(self):
        """ U - UPDATE: UPDATE table WHERE id = ? """
        current_item = self.list_tasks.currentItem()
        if current_item is None:
            self.label_result.setText("Select a task first to update!")
            return

        new_text = self.input_task.text().strip()
        if not new_text:
            self.label_result.setText("New text cannot be empty!")
            return

        task_id = current_item.data(Qt.UserRole)

        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("UPDATE tasks SET text = ? WHERE id = ?", (new_text, task_id))
        conn.commit()
        conn.close()

        self.read_tasks()
        self.label_result.setText(f"Updated ID {task_id} to '{new_text}'")

    @Slot()
    def delete_task(self):
        """ D - DELETE: DELETE FROM table WHERE id = ? """
        current_item = self.list_tasks.currentItem()
        if current_item is None:
            self.label_result.setText("Select a task first to delete!")
            return

        task_id = current_item.data(Qt.UserRole)

        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("DELETE FROM tasks WHERE id = ?", (task_id,))
        conn.commit()
        conn.close()

        self.input_task.clear()
        self.read_tasks()
        self.label_result.setText(f"Deleted ID {task_id}")

    @Slot()
    def clear_tasks(self):
        """ DELETE ALL """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("DELETE FROM tasks")
        conn.commit()
        conn.close()
        self.read_tasks()
        self.label_result.setText("Database cleared!")
```
