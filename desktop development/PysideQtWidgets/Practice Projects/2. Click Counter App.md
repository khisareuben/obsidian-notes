```python
import sys
from pathlib import Path
from PySide6.QtWidgets import QApplication, QMainWindow
from widget import CounterWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setCentralWidget(CounterWidget())

        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.showMaximized()
    sys.exit(app.exec())
```

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout, QLabel, QPushButton, QLineEdit
from PySide6.QtCore import Slot, Qt

class CounterWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        self.count = 0

        self.label_title = QLabel("Counter")
        self.label_title.setObjectName("label_title")
        self.label_title.setAlignment(Qt.AlignmentFlag.AlignCenter)

        self.label_subtitle = QLabel("Track your count.Simple.Light theme")
        self.label_subtitle.setObjectName("label_subtitle")
        self.label_subtitle.setAlignment(Qt.AlignmentFlag.AlignCenter)

        self.label_count = QLabel("0")
        self.label_count.setObjectName("label_count")
        self.label_count.setAlignment(Qt.AlignmentFlag.AlignCenter)


        #buttons
        self.btn_minus = QPushButton("-")
        self.btn_minus.setObjectName("btn_minus")
        self.btn_minus.setFixedSize(70,70)
        self.btn_reset = QPushButton("Reset")
        self.btn_reset.setObjectName("btn_reset")
        self.btn_reset.setFixedSize(120,55)
        self.btn_add = QPushButton("+")
        self.btn_add.setFixedSize(70,70)
        self.btn_add.setObjectName("btn_add")

        self.label_info = QLabel("Press + or - to change the value. Reset returns to zero")
        self.label_info.setObjectName("label_info")
        self.label_info.setAlignment(Qt.AlignmentFlag.AlignCenter)

        self.status = QLineEdit()
        self.status.setText("Ready.Count: 0")
        self.status.setObjectName("status")
        self.status.setReadOnly(True)
        self.status.setMaximumWidth(700)
        self.status.setMinimumHeight(50)
        self.status.setAlignment(Qt.AlignmentFlag.AlignCenter)

        #self.label_status = QLabel("Light Theme.PySide6 6.6.2")
        #self.label_status.setObjectName("status")

        main_layout = QVBoxLayout(self)
        main_layout.setSpacing(15)
        main_layout.setAlignment(Qt.AlignmentFlag.AlignCenter)

        row = QHBoxLayout()
        row.setSpacing(20)
        row.setAlignment(Qt.AlignmentFlag.AlignCenter)
        row.addWidget(self.btn_minus)
        row.addWidget(self.btn_reset)
        row.addWidget(self.btn_add)

        main_layout.addWidget(self.label_title)
        main_layout.addWidget(self.label_subtitle)
        main_layout.addStretch()
        main_layout.addWidget(self.label_count)
        main_layout.addStretch()
        main_layout.addLayout(row)
        main_layout.addStretch()
        main_layout.addWidget(self.label_info)
        main_layout.addStretch()
        main_layout.addWidget(self.status)
        main_layout.addStretch()

        self.btn_minus.clicked.connect(self.minus_count)
        self.btn_add.clicked.connect(self.add_count)
        self.btn_reset.clicked.connect(self.reset_count)


    def add_count(self):
        self.count += 1
        self.display()

    def minus_count(self):
        self.count -= 1
        self.display()

    def display(self):
        self.label_count.setText(str(self.count))
        self.status.setText(f"Ready.Count: {self.count}")

    def reset_count(self):
        self.count = 0
        self.label_count.setText(f"{self.count}")
        self.status.setText(f"Ready.Count: {self.count}")
```
