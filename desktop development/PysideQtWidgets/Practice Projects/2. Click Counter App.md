```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from pathlib import Path
from widget import CounterWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("02 - Click Counter")
        self.resize(1000, 700)
        self.setCentralWidget(CounterWidget())

        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout, QLabel, QPushButton
from PySide6.QtCore import Qt, Slot

class CounterWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        self.count = 0

        # title
        self.label_info = QLabel("CLICK COUNTER")
        self.label_info.setObjectName("label_info")
        self.label_info.setAlignment(Qt.AlignmentFlag.AlignCenter)

        # big counter display
        self.label_counter = QLabel("0")
        self.label_counter.setObjectName("label_counter")
        self.label_counter.setAlignment(Qt.AlignmentFlag.AlignCenter)

        # result/status
        self.label_result = QLabel("Start clicking!")
        self.label_result.setObjectName("label_result")
        self.label_result.setAlignment(Qt.AlignmentFlag.AlignCenter)

        # buttons
        self.btn_minus = QPushButton("-")
        self.btn_minus.setObjectName("btn_minus")

        self.btn_plus = QPushButton("+")
        self.btn_plus.setObjectName("btn_plus")

        self.btn_reset = QPushButton("Reset")
        self.btn_reset.setObjectName("btn_reset")

        # === Layout ====
        main_layout = QVBoxLayout(self)
        main_layout.setSpacing(15)
        main_layout.setContentsMargins(30, 30, 30, 30)

        btn_layout = QHBoxLayout()
        btn_layout.setSpacing(20)
        btn_layout.addWidget(self.btn_minus)
        btn_layout.addWidget(self.btn_plus)

        main_layout.addWidget(self.label_info)
        main_layout.addStretch()
        main_layout.addWidget(self.label_counter)
        main_layout.addWidget(self.label_result)
        main_layout.addStretch()
        main_layout.addLayout(btn_layout)
        main_layout.addWidget(self.btn_reset)

        # connections
        self.btn_plus.clicked.connect(self.on_plus)
        self.btn_minus.clicked.connect(self.on_minus)
        self.btn_reset.clicked.connect(self.on_reset)

    @Slot()
    def on_plus(self):
        self.count += 1
        self.update_display()

    @Slot()
    def on_minus(self):
        self.count -= 1
        self.update_display()

    @Slot()
    def on_reset(self):
        self.count = 0
        self.update_display()

    def update_display(self):
        self.label_counter.setText(str(self.count))

        if self.count > 0:
            self.label_result.setText(f"You have clicked +{self.count} times")
        elif self.count < 0:
            self.label_result.setText(f"You went negative: {self.count}")
        else:
            self.label_result.setText("Back to zero!")
```
