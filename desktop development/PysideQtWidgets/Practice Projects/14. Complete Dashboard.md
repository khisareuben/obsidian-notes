`database.py`
```python
import sqlite3
from pathlib import Path
from PySide6.QtSql import QSqlDatabase, QSqlQuery

DB_PATH = Path(__file__).resolve().parent / "dashboard.db"

def init_db():
    # For QSqlTableModel we need QSQLITE connection
    db = QSqlDatabase.addDatabase("QSQLITE")
    db.setDatabaseName(str(DB_PATH))
    if not db.open():
        print("DB open failed:", db.lastError().text())
        return False

    query = QSqlQuery()
    query.exec("""
        CREATE TABLE IF NOT EXISTS students (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            course TEXT NOT NULL,
            age INTEGER
        )
    """)
    query.exec("""
        CREATE TABLE IF NOT EXISTS notes (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            title TEXT NOT NULL,
            content TEXT NOT NULL
        )
    """)
    query.exec("""
        CREATE TABLE IF NOT EXISTS settings (
            key TEXT PRIMARY KEY,
            value TEXT
        )
    """)
    query.exec("INSERT OR IGNORE INTO settings (key,value) VALUES ('theme','light')")
    query.exec("INSERT OR IGNORE INTO settings (key,value) VALUES ('notifications','1')")

    # Seed students if empty
    query.exec("SELECT COUNT(*) FROM students")
    query.next()
    if query.value(0) == 0:
        query.exec("INSERT INTO students (name,course,age) VALUES ('Ali','Dashboard Project',22)")
        query.exec("INSERT INTO students (name,course,age) VALUES ('Sara','PySide6 Mastery',24)")

    # Also init sqlite3 for repository functions
    return True
```

`repository.py`
```python
from PySide6.QtSql import QSqlQuery
from database import DB_PATH
import sqlite3
from pathlib import Path

def get_conn():
    conn = sqlite3.connect(str(DB_PATH))
    conn.row_factory = sqlite3.Row
    return conn

# NOTES CRUD
def create_note(title, content):
    conn = get_conn()
    cur = conn.cursor()
    cur.execute("INSERT INTO notes (title,content) VALUES (?,?)", (title, content))
    conn.commit()
    nid = cur.lastrowid
    conn.close()
    return nid

def get_notes():
    conn = get_conn()
    rows = conn.execute("SELECT id,title,content FROM notes ORDER BY id DESC").fetchall()
    conn.close()
    return rows

def delete_note(nid):
    conn = get_conn()
    conn.execute("DELETE FROM notes WHERE id=?", (nid,))
    conn.commit()
    conn.close()

# SETTINGS CRUD
def get_setting(key):
    conn = get_conn()
    row = conn.execute("SELECT value FROM settings WHERE key=?", (key,)).fetchone()
    conn.close()
    return row["value"] if row else ""

def set_setting(key, value):
    conn = get_conn()
    conn.execute("INSERT OR REPLACE INTO settings (key,value) VALUES (?,?)", (key, value))
    conn.commit()
    conn.close()
```

```python
import sys

        self.setCentralWidget(DashboardWidget())
        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            pass

if __name__ == "__main__":
    app = QApplication(sys.argv)
    if not init_db():
        sys.exit(-1)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```

`widget.py`
```python
from pathlib import Path
from PySide6.QtWidgets import (QWidget, QVBoxLayout, QHBoxLayout, QLabel, QLineEdit,
                               QTextEdit, QTabWidget, QPushButton, QTableView, QListView,
                               QCheckBox, QRadioButton, QGroupBox, QFileDialog, QHeaderView, QDoubleSpinBox, QComboBox)
from PySide6.QtCore import Qt, Slot, QDir
from PySide6.QtGui import QFileSystemModel, QPixmap
from PySide6.QtSql import QSqlTableModel
import repository

class DashboardWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setObjectName("centralWidget")
        self.current_pixmap = None
        self.scale_factor = 1.0

        self.tabs = QTabWidget()
        self.tabs.setObjectName("main_tabs")

        # TAB 1: Students - #11 QTableView + QSqlTableModel
        self.tabs.addTab(self.create_students_tab(), "🎓 Students [QTableView]")

        # TAB 2: Notebook - #9 QTabWidget + QTextEdit
        self.tabs.addTab(self.create_notes_tab(), "📓 Notes [QTextEdit]")

        # TAB 3: File Explorer - #10 QListView + QFileSystemModel
        self.tabs.addTab(self.create_explorer_tab(), "📁 Explorer [QListView]")

        # TAB 4: Image Viewer - #12 QLabel + QPixmap
        self.tabs.addTab(self.create_image_tab(), "🖼 Viewer [QPixmap]")

        # TAB 5: Calculator - #7 QSpinBox + QDoubleSpinBox
        self.tabs.addTab(self.create_calc_tab(), "🧮 Calculator")

        # TAB 6: Settings - #13 QCheckBox + QRadioButton
        self.tabs.addTab(self.create_settings_tab(), "⚙️ Settings")

        self.label_result = QLabel("Dashboard Ready - All 1-13 Combined")
        self.label_result.setObjectName("label_result")

        main_layout = QVBoxLayout(self)
        main_layout.setContentsMargins(10,10,10,10)
        main_layout.addWidget(self.tabs)
        main_layout.addWidget(self.label_result)

    # ===== TAB 1: STUDENTS =====
    def create_students_tab(self):
        w = QWidget()
        self.student_model = QSqlTableModel(w)
        self.student_model.setTable("students")
        self.student_model.setEditStrategy(QSqlTableModel.OnFieldChange)
        self.student_model.select()

        table = QTableView()
        table.setModel(self.student_model)
        table.horizontalHeader().setSectionResizeMode(QHeaderView.Stretch)
        table.setObjectName("table_view")

        self.input_s_name = QLineEdit(placeholderText="Name")
        self.input_s_name.setObjectName("input_name")
        self.input_s_course = QLineEdit(placeholderText="Course")
        self.input_s_course.setObjectName("input_name")

        btn_add = QPushButton("CREATE Student")
        btn_add.setObjectName("btn_add")
        btn_add.clicked.connect(self.add_student)

        layout = QVBoxLayout(w)
        input_row = QHBoxLayout()
        input_row.addWidget(self.input_s_name)
        input_row.addWidget(self.input_s_course)
        input_row.addWidget(btn_add)
        layout.addLayout(input_row)
        layout.addWidget(table)
        return w

    @Slot()
    def add_student(self):
        row = self.student_model.rowCount()
        self.student_model.insertRow(row)
        self.student_model.setData(self.student_model.index(row,1), self.input_s_name.text())
        self.student_model.setData(self.student_model.index(row,2), self.input_s_course.text())
        self.student_model.setData(self.student_model.index(row,3), 20)
        self.student_model.submitAll()
        self.label_result.setText(f"Student CREATED: {self.input_s_name.text()}")

    # ===== TAB 2: NOTES =====
    def create_notes_tab(self):
        w = QWidget()
        self.note_title = QLineEdit(placeholderText="Title")
        self.note_title.setObjectName("input_name")
        self.note_content = QTextEdit(placeholderText="Write note...")
        self.note_content.setObjectName("text_content")

        btn_save_note = QPushButton("SAVE Note")
        btn_save_note.setObjectName("btn_save")
        btn_save_note.clicked.connect(self.save_note)

        layout = QVBoxLayout(w)
        layout.addWidget(self.note_title)
        layout.addWidget(self.note_content)
        layout.addWidget(btn_save_note)
        return w

    @Slot()
    def save_note(self):
        repository.create_note(self.note_title.text(), self.note_content.toPlainText())
        self.label_result.setText(f"Note SAVED: {self.note_title.text()}")
        self.note_title.clear()
        self.note_content.clear()

    # ===== TAB 3: EXPLORER =====
    def create_explorer_tab(self):
        w = QWidget()
        self.fs_model = QFileSystemModel()
        self.fs_model.setRootPath(QDir.rootPath())

        self.explorer_view = QListView()
        self.explorer_view.setObjectName("list_tasks")
        self.explorer_view.setModel(self.fs_model)
        self.explorer_view.setRootIndex(self.fs_model.index(str(Path.home())))
        self.explorer_view.doubleClicked.connect(lambda idx: self.explorer_view.setRootIndex(self.fs_model.index(self.fs_model.filePath(idx))) if Path(self.fs_model.filePath(idx)).is_dir() else None)

        layout = QVBoxLayout(w)
        layout.addWidget(self.explorer_view)
        return w

    # ===== TAB 4: IMAGE VIEWER =====
    def create_image_tab(self):
        w = QWidget()
        self.img_label = QLabel("No Image")
        self.img_label.setAlignment(Qt.AlignCenter)
        self.img_label.setObjectName("image_label")

        btn_open = QPushButton("Open Image")
        btn_open.setObjectName("btn_add")
        btn_open.clicked.connect(self.open_image)

        layout = QVBoxLayout(w)
        layout.addWidget(btn_open)
        layout.addWidget(self.img_label)
        return w

    @Slot()
    def open_image(self):
        path, _ = QFileDialog.getOpenFileName(self, "Open", str(Path.home()), "Images (*.png *.jpg *.jpeg)")
        if path:
            pixmap = QPixmap(path)
            self.img_label.setPixmap(pixmap.scaled(800, 500, Qt.KeepAspectRatio, Qt.SmoothTransformation))
            self.label_result.setText(f"Image Loaded: {Path(path).name}")

    # ===== TAB 5: CALCULATOR =====
    def create_calc_tab(self):
        w = QWidget()
        self.spin_a = QDoubleSpinBox()
        self.spin_a.setObjectName("spin_number")
        self.spin_a.setValue(10)
        self.spin_b = QDoubleSpinBox()
        self.spin_b.setObjectName("spin_number")
        self.spin_b.setValue(5)
        self.combo_op = QComboBox()
        self.combo_op.setObjectName("combo_op")
        self.combo_op.addItems(["+","-","×","÷"])
        self.calc_result = QLabel("Result = 15")
        self.calc_result.setObjectName("label_result")
        self.calc_result.setAlignment(Qt.AlignCenter)

        def calc():
            a = self.spin_a.value(); b = self.spin_b.value(); op = self.combo_op.currentText()
            if op == "+": r = a+b
            elif op == "-": r = a-b
            elif op == "×": r = a*b
            else: r = a/b if b!=0 else 0
            self.calc_result.setText(f"Result = {r}")

        self.spin_a.valueChanged.connect(calc)
        self.spin_b.valueChanged.connect(calc)
        self.combo_op.currentTextChanged.connect(calc)

        layout = QVBoxLayout(w)
        row = QHBoxLayout()
        row.addWidget(self.spin_a); row.addWidget(self.combo_op); row.addWidget(self.spin_b)
        layout.addLayout(row)
        layout.addWidget(self.calc_result)
        layout.addStretch()
        return w

    # ===== TAB 6: SETTINGS =====
    def create_settings_tab(self):
        w = QWidget()
        group = QGroupBox("App Settings")
        group.setObjectName("group_box")

        self.check_notify = QCheckBox("Enable Notifications")
        self.check_notify.setObjectName("check_box")
        self.check_notify.setChecked(repository.get_setting("notifications") == "1")

        self.radio_light = QRadioButton("Light Theme")
        self.radio_light.setObjectName("radio_btn")
        self.radio_dark = QRadioButton("Dark Theme")
        self.radio_dark.setObjectName("radio_btn")
        theme = repository.get_setting("theme")
        self.radio_light.setChecked(theme == "light")
        self.radio_dark.setChecked(theme!= "light")

        btn_save_set = QPushButton("SAVE Settings")
        btn_save_set.setObjectName("btn_save")
        btn_save_set.clicked.connect(lambda: [repository.set_setting("notifications","1" if self.check_notify.isChecked() else "0"), repository.set_setting("theme","light" if self.radio_light.isChecked() else "dark"), self.label_result.setText("Settings SAVED to DB")])

        glayout = QVBoxLayout(group)
        glayout.addWidget(self.check_notify)
        glayout.addWidget(self.radio_light)
        glayout.addWidget(self.radio_dark)

        layout = QVBoxLayout(w)
        layout.addWidget(group)
        layout.addWidget(btn_save_set)
        layout.addStretch()
        return w
```
