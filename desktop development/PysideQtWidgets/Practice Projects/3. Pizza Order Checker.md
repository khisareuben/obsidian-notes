            path = Path(__file__).resolve().parent / "style.qss"

        except FileNotFoundError:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())

from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QCheckBox, QPushButton
from PySide6.QtCore import Qt, Slot

class PizzaWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        self.base_price = 8

        # title
        self.label_info = QLabel("Build Your Pizza - Base $8")
        self.label_info.setObjectName("label_info")

        # checkboxes - price stored in property
        self.cb_cheese = QCheckBox("Extra Cheese (+$1.5)")
        self.cb_cheese.setObjectName("topping_check")
        self.cb_cheese.setProperty("price", 1.5)

        self.cb_pepperoni = QCheckBox("Pepperoni (+$2.0)")
        self.cb_pepperoni.setObjectName("topping_check")
        self.cb_pepperoni.setProperty("price", 2.0)

        self.cb_mushroom = QCheckBox("Mushroom (+$1.0)")
        self.cb_mushroom.setObjectName("topping_check")
        self.cb_mushroom.setProperty("price", 1.0)

        self.cb_olives = QCheckBox("Olives (+$0.5)")
        self.cb_olives.setObjectName("topping_check")
        self.cb_olives.setProperty("price", 0.5)

        # readonly summary
        self.input_readonly = QLabel(f"Base pizza selected")
        self.input_readonly.setObjectName("input_readonly")

        # result / total
        self.label_result = QLabel(f"Total: ${self.base_price:.2f}")
        self.label_result.setObjectName("label_result")

        # button
        self.btn_get = QPushButton("Place Order")
        self.btn_get.setObjectName("btn_get")

        # === Layout ====
        layout = QVBoxLayout(self)
        layout.setSpacing(12)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.addWidget(self.label_info)
        layout.addWidget(self.cb_cheese)
        layout.addWidget(self.cb_pepperoni)
        layout.addWidget(self.cb_mushroom)
        layout.addWidget(self.cb_olives)
        layout.addWidget(self.input_readonly)
        layout.addWidget(self.label_result)
        layout.addWidget(self.btn_get)
        layout.addStretch()

        # connections - live update on every check
        self.cb_cheese.stateChanged.connect(self.update_total)
        self.cb_pepperoni.stateChanged.connect(self.update_total)
        self.cb_mushroom.stateChanged.connect(self.update_total)
        self.cb_olives.stateChanged.connect(self.update_total)
        self.btn_get.clicked.connect(self.on_order)

    @Slot()
    def update_total(self):
        total = self.base_price
        toppings = []

        checks = [self.cb_cheese, self.cb_pepperoni, self.cb_mushroom, self.cb_olives]
        for cb in checks:
            if cb.isChecked():
                total += cb.property("price")
                # remove price text for clean display
                toppings.append(cb.text().split(" (+")[0])

        self.label_result.setText(f"Total: ${total:.2f}")

        if toppings:
            self.input_readonly.setText(f"Toppings: {', '.join(toppings)}")
        else:
            self.input_readonly.setText("Base pizza selected")

    @Slot()
    def on_order(self):
        total_text = self.label_result.text()
        toppings_text = self.input_readonly.text()
        self.label_result.setText(f"Ordered! {total_text} - {toppings_text}")
