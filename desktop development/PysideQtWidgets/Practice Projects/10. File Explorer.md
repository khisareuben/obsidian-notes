`database.py`
```python
import sqlite3
from pathlib import Path

DB_PATH = Path(__file__).resolve().parent / "explorer.db"

def get_connection():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS history (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            path TEXT NOT NULL UNIQUE,
            visited_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)
    conn.commit()
    conn.close()
```

`repository.py`
```python
from database import get_connection

def add_to_history(path: str):
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("INSERT OR IGNORE INTO history (path) VALUES (?)", (path,))
    cursor.execute("UPDATE history SET visited_at = CURRENT_TIMESTAMP WHERE path = ?", (path,))
    conn.commit()
    conn.close()

def get_history():
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("SELECT path FROM history ORDER BY visited_at DESC LIMIT 10")
    rows = cursor.fetchall()
    conn.close()
    return [r["path"] for r in rows]

def clear_history():
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("DELETE FROM history")
    conn.commit()
    conn.close()
```


`main.py`
```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from pathlib import Path
from database import init_db
from widget import ExplorerWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("10 - File Explorer")
        self.resize(1100, 750)
        self.setCentralWidget(ExplorerWidget())
        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            pass

if __name__ == "__main__":
    init_db()
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```

`widget.py`
```python
from pathlib import Path
from PySide6.QtWidgets import (QWidget, QVBoxLayout, QHBoxLayout, QLabel, 
                               QLineEdit, QListView, QPushButton, QComboBox)
from PySide6.QtCore import Qt, Slot, QDir
from PySide6.QtGui import QFileSystemModel
import repository

class ExplorerWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        self.label_info = QLabel("File Explorer - QListView + QFileSystemModel")
        self.label_info.setObjectName("label_info")

        # Path bar
        self.input_path = QLineEdit()
        self.input_path.setObjectName("input_name")
        self.input_path.setText(str(Path.home()))

        # History combo [from DB]
        self.combo_history = QComboBox()
        self.combo_history.setObjectName("combo_history")
        self.combo_history.addItem("Recent History (DB)")

        # THE MODEL - This is what does all the heavy lifting
        self.model = QFileSystemModel()
        self.model.setRootPath(QDir.rootPath())
        # Show files and folders, hide . and ..
        self.model.setFilter(QDir.AllEntries | QDir.NoDotAndDotDot)

        # THE VIEW
        self.list_view = QListView()
        self.list_view.setObjectName("list_tasks")
        self.list_view.setModel(self.model)
        # Set root to home initially
        self.list_view.setRootIndex(self.model.index(str(Path.home())))

        self.btn_up = QPushButton("⬆ Up")
        self.btn_up.setObjectName("btn_up")
        self.btn_open = QPushButton("Open Folder")
        self.btn_open.setObjectName("btn_add")
        self.btn_clear_hist = QPushButton("Clear History DB")
        self.btn_clear_hist.setObjectName("btn_clear")

        self.label_result = QLabel(f"Current: {Path.home()}")
        self.label_result.setObjectName("label_result")

        # Layout
        main_layout = QVBoxLayout(self)
        main_layout.setSpacing(12)
        main_layout.setContentsMargins(20, 20, 20, 20)

        path_layout = QHBoxLayout()
        path_layout.addWidget(self.input_path, 3)
        path_layout.addWidget(self.combo_history, 2)

        btn_layout = QHBoxLayout()
        btn_layout.addWidget(self.btn_up)
        btn_layout.addWidget(self.btn_open)
        btn_layout.addWidget(self.btn_clear_hist)

        main_layout.addWidget(self.label_info)
        main_layout.addLayout(path_layout)
        main_layout.addLayout(btn_layout)
        main_layout.addWidget(self.list_view)
        main_layout.addWidget(self.label_result)

        # Connections
        self.btn_up.clicked.connect(self.go_up)
        self.btn_open.clicked.connect(self.open_path)
        self.btn_clear_hist.clicked.connect(self.clear_history)
        self.list_view.doubleClicked.connect(self.on_double_click)
        self.combo_history.activated.connect(self.on_history_selected)

        self.refresh_history()

    def refresh_history(self):
        self.combo_history.clear()
        history = repository.get_history()
        if not history:
            self.combo_history.addItem("No history yet")
        else:
            for p in history:
                self.combo_history.addItem(p)

    def set_path(self, path_str: str):
        path = Path(path_str).expanduser()
        if not path.exists():
            self.label_result.setText(f"Path not found: {path_str}")
            return
        if path.is_file():
            path = path.parent

        # Update model root
        self.list_view.setRootIndex(self.model.index(str(path)))
        self.input_path.setText(str(path))
        self.label_result.setText(f"Current: {path} | Items: {self.model.rowCount(self.list_view.rootIndex())}")
        
        # SAVE to DB [CREATE]
        repository.add_to_history(str(path))
        self.refresh_history()

    @Slot()
    def go_up(self):
        current = Path(self.input_path.text())
        self.set_path(str(current.parent))

    @Slot()
    def open_path(self):
        self.set_path(self.input_path.text())

    @Slot()
    def on_double_click(self, index):
        # QFileSystemModel gives us file info
        file_path = self.model.filePath(index)
        p = Path(file_path)
        if p.is_dir():
            self.set_path(file_path)
        else:
            self.label_result.setText(f"File: {p.name} | Size: {p.stat().st_size} bytes")

    @Slot(int)
    def on_history_selected(self, idx):
        path = self.combo_history.currentText()
        if path and path != "No history yet" and "Recent History" not in path:
            self.set_path(path)

    @Slot()
    def clear_history(self):
        repository.clear_history()
        self.refresh_history()
        self.label_result.setText("History DB cleared")
```
