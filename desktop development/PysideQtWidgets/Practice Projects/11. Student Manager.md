`database.py`
```python
from pathlib import Path
from PySide6.QtSql import QSqlDatabase, QSqlQuery

DB_PATH = Path(__file__).resolve().parent / "students.db"

def init_db():
    """ Create QSQLITE connection - this is GLOBAL for the whole app """
    db = QSqlDatabase.addDatabase("QSQLITE")
    db.setDatabaseName(str(DB_PATH))

    if not db.open():
        print("Failed to open database:", db.lastError().text())
        return False

    query = QSqlQuery()
    query.exec("""
        CREATE TABLE IF NOT EXISTS students (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            course TEXT NOT NULL,
            age INTEGER NOT NULL
        )
    """)
    
    # Seed data if empty
    query.exec("SELECT COUNT(*) FROM students")
    query.next()
    if query.value(0) == 0:
        query.exec("INSERT INTO students (name, course, age) VALUES ('Ali', 'PySide6', 22)")
        query.exec("INSERT INTO students (name, course, age) VALUES ('Sara', 'Python', 24)")
        query.exec("INSERT INTO students (name, course, age) VALUES ('John', 'Database', 21)")

    return True
```


`main.py`
```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from pathlib import Path
from database import init_db
from widget import StudentWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("11 - Student Manager - QSqlTableModel")
        self.resize(1100, 700)
        self.setCentralWidget(StudentWidget())
        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            pass

if __name__ == "__main__":
    app = QApplication(sys.argv)
    if not init_db():
        sys.exit(-1)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```

from PySide6.QtWidgets import (QWidget, QVBoxLayout, QHBoxLayout, QLabel, 
                               QTableView, QPushButton, QLineEdit, QHeaderView)
from PySide6.QtCore import Slot
from PySide6.QtSql import QSqlTableModel

class StudentWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(True, True)
        self.setObjectName("centralWidget")

        self.label_info = QLabel("Student Manager - QTableView + QSqlTableModel [Industry Qt Model]")
        self.label_info.setObjectName("label_info")

        # THE MODEL - This IS your CRUD
        self.model = QSqlTableModel(self)
        self.model.setTable("students")
        self.model.setEditStrategy(QSqlTableModel.OnFieldChange)  # Auto save when you edit cell
        self.model.setHeaderData(0, 1, "ID")
        self.model.setHeaderData(1, 1, "Name")
        self.model.setHeaderData(2, 1, "Course")
        self.model.setHeaderData(3, 1, "Age")
        self.model.select()  # READ from DB

        # THE VIEW
        self.table_view = QTableView()
        self.table_view.setObjectName("table_view")
        self.table_view.setModel(self.model)
        self.table_view.horizontalHeader().setSectionResizeMode(QHeaderView.Stretch)
        self.table_view.setSelectionBehavior(QTableView.SelectRows)
        self.table_view.setAlternatingRowColors(True)
        self.table_view.hideColumn(0)  # Hide ID column

        # Inputs for CREATE
        self.input_name = QLineEdit()
        self.input_name.setObjectName("input_name")
        self.input_name.setPlaceholderText("Name")

        self.input_course = QLineEdit()
        self.input_course.setObjectName("input_name")
        self.input_course.setPlaceholderText("Course")

        self.input_age = QLineEdit()
        self.input_age.setObjectName("input_name")
        self.input_age.setPlaceholderText("Age")

        self.btn_add = QPushButton("CREATE / Add Student")
        self.btn_add.setObjectName("btn_add")

        self.btn_delete = QPushButton("DELETE Selected")
        self.btn_delete.setObjectName("btn_delete")

        self.btn_refresh = QPushButton("Refresh")
        self.btn_refresh.setObjectName("btn_clear")

        self.label_result = QLabel(f"Loaded {self.model.rowCount()} students")
        self.label_result.setObjectName("label_result")

        # Layout
        main_layout = QVBoxLayout(self)
        main_layout.setSpacing(12)
        main_layout.setContentsMargins(20, 20, 20, 20)

        input_layout = QHBoxLayout()
        input_layout.addWidget(self.input_name)
        input_layout.addWidget(self.input_course)
        input_layout.addWidget(self.input_age)

        btn_layout = QHBoxLayout()
        btn_layout.addWidget(self.btn_add)
        btn_layout.addWidget(self.btn_delete)
        btn_layout.addWidget(self.btn_refresh)

        main_layout.addWidget(self.label_info)
        main_layout.addLayout(input_layout)
        main_layout.addLayout(btn_layout)
        main_layout.addWidget(self.table_view)
        main_layout.addWidget(self.label_result)

        # Connections
        self.btn_add.clicked.connect(self.create_student)
        self.btn_delete.clicked.connect(self.delete_student)
        self.btn_refresh.clicked.connect(self.refresh)

    @Slot()
    def create_student(self):
        """ C - CREATE: Insert new row via model """
        name = self.input_name.text().strip()
        course = self.input_course.text().strip()
        age_text = self.input_age.text().strip()

        if not name or not course or not age_text:
            self.label_result.setText("Fill all fields!")
            return

        try:
            age = int(age_text)
        except ValueError:
            self.label_result.setText("Age must be number!")
            return

        row = self.model.rowCount()
        self.model.insertRow(row)
        self.model.setData(self.model.index(row, 1), name)
        self.model.setData(self.model.index(row, 2), course)
        self.model.setData(self.model.index(row, 3), age)
        
        if self.model.submitAll():
            self.label_result.setText(f"CREATED: {name} | Total: {self.model.rowCount()}")
            self.input_name.clear()
            self.input_course.clear()
            self.input_age.clear()
        else:
            self.label_result.setText(f"Error: {self.model.lastError().text()}")

    @Slot()
    def delete_student(self):
        """ D - DELETE: Delete selected row """
        selected = self.table_view.currentIndex()
        if not selected.isValid():
            self.label_result.setText("Select a row to DELETE!")
            return

        self.model.removeRow(selected.row())
        self.model.submitAll()
        self.model.select()
        self.label_result.setText(f"DELETED row {selected.row()} | Left: {self.model.rowCount()}")

    @Slot()
    def refresh(self):
        """ R - READ: Re-select """
        self.model.select()
        self.label_result.setText(f"Refreshed | {self.model.rowCount()} students")
