import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from pathlib import Path
from widget import ThemeWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("04 - Theme Selector")
        self.resize(1000, 700)
        self.setCentralWidget(ThemeWidget())

        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())

from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QRadioButton, QButtonGroup, QPushButton
from PySide6.QtCore import Qt, Slot

class ThemeWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        # title
        self.label_info = QLabel("Choose App Theme:")
        self.label_info.setObjectName("label_info")

        # radio buttons
        self.rb_wheat = QRadioButton("Wheat (Default)")
        self.rb_wheat.setObjectName("theme_radio")
        self.rb_wheat.setChecked(True)

        self.rb_light = QRadioButton("Light Mode")
        self.rb_light.setObjectName("theme_radio")

        self.rb_dark = QRadioButton("Dark Mode")
        self.rb_dark.setObjectName("theme_radio")

        # group them - so only 1 can be selected
        self.theme_group = QButtonGroup(self)
        self.theme_group.addButton(self.rb_wheat)
        self.theme_group.addButton(self.rb_light)
        self.theme_group.addButton(self.rb_dark)
        self.theme_group.setExclusive(True)

        # readonly preview
        self.input_readonly = QLabel("Current: Wheat Theme")
        self.input_readonly.setObjectName("input_readonly")
        self.input_readonly.setAlignment(Qt.AlignmentFlag.AlignCenter)

        # result
        self.label_result = QLabel("Wheat theme is active")
        self.label_result.setObjectName("label_result")
        self.label_result.setAlignment(Qt.AlignmentFlag.AlignCenter)

        # button
        self.btn_get = QPushButton("Apply Theme")
        self.btn_get.setObjectName("btn_get")

        # === Layout ====
        layout = QVBoxLayout(self)
        layout.setSpacing(12)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.addWidget(self.label_info)
        layout.addWidget(self.rb_wheat)
        layout.addWidget(self.rb_light)
        layout.addWidget(self.rb_dark)
        layout.addWidget(self.input_readonly)
        layout.addWidget(self.label_result)
        layout.addWidget(self.btn_get)
        layout.addStretch()

        # live change
        self.rb_wheat.toggled.connect(self.on_theme_changed)
        self.rb_light.toggled.connect(self.on_theme_changed)
        self.rb_dark.toggled.connect(self.on_theme_changed)
        self.btn_get.clicked.connect(self.on_apply)

    @Slot()
    def on_theme_changed(self):
        # only react when a radio becomes checked
        if self.rb_wheat.isChecked():
            self.setStyleSheet("#centralWidget { background-color: wheat; }")
            self.input_readonly.setText("Current: Wheat Theme")
            self.label_result.setText("Preview: Wheat theme")
            self.label_result.setProperty("theme", "wheat")

        elif self.rb_light.isChecked():
            self.setStyleSheet("#centralWidget { background-color: #F5F5F5; }")
            self.input_readonly.setText("Current: Light Theme")
            self.label_result.setText("Preview: Light theme")
            self.label_result.setProperty("theme", "light")

        elif self.rb_dark.isChecked():
            self.setStyleSheet("#centralWidget { background-color: #121212; }")
            self.input_readonly.setText("Current: Dark Theme")
            self.label_result.setText("Preview: Dark theme")
            self.label_result.setProperty("theme", "dark")

    @Slot()
    def on_apply(self):
        current = self.input_readonly.text()
        self.label_result.setText(f"{current} Applied Successfully!")
