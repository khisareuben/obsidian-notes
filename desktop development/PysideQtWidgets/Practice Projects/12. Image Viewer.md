`database.py`
```python
import sqlite3
from pathlib import Path

DB_PATH = Path(__file__).resolve().parent / "viewer.db"

def get_connection():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS history (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            path TEXT NOT NULL UNIQUE,
            viewed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)
    conn.commit()
    conn.close()
```

`repository.py`
```python
from database import get_connection

def add_to_history(path: str):
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("INSERT OR REPLACE INTO history (path, viewed_at) VALUES (?, CURRENT_TIMESTAMP)", (path,))
    conn.commit()
    conn.close()

def get_history():
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("SELECT path FROM history ORDER BY viewed_at DESC LIMIT 10")
    rows = cursor.fetchall()
    conn.close()
    return [r["path"] for r in rows]
```

```python
from pathlib import Path
from PySide6.QtWidgets import (QWidget, QVBoxLayout, QHBoxLayout, QLabel,
                               QPushButton, QFileDialog, QComboBox, QScrollArea)
from PySide6.QtCore import Qt, Slot
from PySide6.QtGui import QPixmap
import repository

class ImageViewerWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")
        self.current_pixmap = None
        self.scale_factor = 1.0
        self.rotation = 0

        self.label_info = QLabel("Image Viewer - QLabel + QPixmap + QFileDialog")
        self.label_info.setObjectName("label_info")

        # Image display - QLabel holds QPixmap
        self.image_label = QLabel("No Image - Click Open")
        self.image_label.setObjectName("image_label")
        self.image_label.setAlignment(Qt.AlignCenter)
        self.image_label.setStyleSheet("background-color: #1e1e1e; border: 1px solid #333333; border-radius: 8px;")

        self.scroll = QScrollArea()
        self.scroll.setWidget(self.image_label)
        self.scroll.setWidgetResizable(True)

        self.combo_history = QComboBox()
        self.combo_history.setObjectName("combo_history")

        self.btn_open = QPushButton("Open Image")
        self.btn_open.setObjectName("btn_add")

        self.btn_zoom_in = QPushButton("Zoom +")
        self.btn_zoom_in.setObjectName("btn_up")

        self.btn_zoom_out = QPushButton("Zoom -")
        self.btn_zoom_out.setObjectName("btn_up")

        self.btn_rotate = QPushButton("Rotate 90°")
        self.btn_rotate.setObjectName("btn_clear")

        self.btn_fit = QPushButton("Fit to Window")
        self.btn_fit.setObjectName("btn_clear")

        self.label_result = QLabel("Ready - Open an image")
        self.label_result.setObjectName("label_result")

        # Layout
        main_layout = QVBoxLayout(self)
        main_layout.setSpacing(12)
        main_layout.setContentsMargins(20, 20, 20, 20)

        top_layout = QHBoxLayout()
        top_layout.addWidget(self.btn_open)
        top_layout.addWidget(self.combo_history, 1)

        zoom_layout = QHBoxLayout()
        zoom_layout.addWidget(self.btn_zoom_in)
        zoom_layout.addWidget(self.btn_zoom_out)
        zoom_layout.addWidget(self.btn_rotate)
        zoom_layout.addWidget(self.btn_fit)

        main_layout.addWidget(self.label_info)
        main_layout.addLayout(top_layout)
        main_layout.addLayout(zoom_layout)
        main_layout.addWidget(self.scroll)
        main_layout.addWidget(self.label_result)

        # Connections
        self.btn_open.clicked.connect(self.open_image)
        self.btn_zoom_in.clicked.connect(self.zoom_in)
        self.btn_zoom_out.clicked.connect(self.zoom_out)
        self.btn_rotate.clicked.connect(self.rotate)
        self.btn_fit.clicked.connect(self.fit_to_window)
        self.combo_history.activated.connect(self.on_history_selected)

        self.refresh_history()

    def refresh_history(self):
        self.combo_history.clear()
        history = repository.get_history()
        if not history:
            self.combo_history.addItem("Recent Images (DB)")
        else:
            for p in history:
                self.combo_history.addItem(p)

    def load_pixmap(self, path_str: str):
        path = Path(path_str)
        if not path.exists():
            return

        pixmap = QPixmap(str(path))
        if pixmap.isNull():
            self.label_result.setText(f"Failed to load: {path.name}")
            return

        self.current_pixmap = QPixmap(str(path)) # Keep original
        self.scale_factor = 1.0
        self.rotation = 0
        self.display_pixmap()
        self.label_result.setText(f"Loaded: {path.name} | {pixmap.width()}x{pixmap.height()}")
        repository.add_to_history(str(path))
        self.refresh_history()

    def display_pixmap(self):
        if not self.current_pixmap:
            return

        # Apply rotation and scale
        transformed = self.current_pixmap.transformed(
            self.current_pixmap.transformed(
                self.current_pixmap
            )
        )
        # For simplicity, use scaled + rotated manually
        from PySide6.QtGui import QTransform
        transform = QTransform()
        transform.rotate(self.rotation)
        rotated = self.current_pixmap.transformed(transform, Qt.SmoothTransformation)
        scaled = rotated.scaled(
            int(rotated.width() * self.scale_factor),
            int(rotated.height() * self.scale_factor),
            Qt.KeepAspectRatio,
            Qt.SmoothTransformation
        )
        self.image_label.setPixmap(scaled)
        self.image_label.resize(scaled.size())

    @Slot()
    def open_image(self):
        file_path, _ = QFileDialog.getOpenFileName(
            self, "Open Image", str(Path.home()),
            "Images (*.png *.jpg *.jpeg *.bmp *.gif);;All Files (*)"
        )
        if file_path:
            self.load_pixmap(file_path)

    @Slot()
    def zoom_in(self):
        if not self.current_pixmap: return
        self.scale_factor *= 1.25
        self.display_pixmap()
        self.label_result.setText(f"Zoom: {self.scale_factor*100:.0f}%")

    @Slot()
    def zoom_out(self):
        if not self.current_pixmap: return
        self.scale_factor *= 0.8
        self.display_pixmap()
        self.label_result.setText(f"Zoom: {self.scale_factor*100:.0f}%")

    @Slot()
    def rotate(self):
        if not self.current_pixmap: return
        self.rotation = (self.rotation + 90) % 360
        self.display_pixmap()
        self.label_result.setText(f"Rotated: {self.rotation}°")

    @Slot()
    def fit_to_window(self):
        if not self.current_pixmap: return
        self.image_label.setPixmap(
            self.current_pixmap.scaled(
                self.scroll.viewport().size(),
                Qt.KeepAspectRatio,
                Qt.SmoothTransformation
            )
        )

    @Slot(int)
    def on_history_selected(self, idx):
        path = self.combo_history.currentText()
        if path and "Recent" not in path and "No history" not in path:
            self.load_pixmap(path)
```
