`database.py`
```python
import sqlite3
from pathlib import Path

DB_PATH = Path(__file__).resolve().parent / "todo.db"

def get_connection():
    """ Single place to get DB connection. If we switch to MySQL/Postgres later, we change ONLY here. """
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row  # Allows accessing columns by name
    return conn

def init_db():
    """ Create tables - called once at startup. In real companies, this is called migrations. """
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS tasks (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            text TEXT NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)
    conn.commit()
    conn.close()
```

`repository.py`
```python
from database import get_connection

# C - CREATE
def create_task(text: str) -> int:
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("INSERT INTO tasks (text) VALUES (?)", (text,))
    conn.commit()
    new_id = cursor.lastrowid
    conn.close()
    return new_id

# R - READ
def get_all_tasks():
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("SELECT id, text FROM tasks ORDER BY id DESC")
    rows = cursor.fetchall()
    conn.close()
    return rows  # List of Row objects

# U - UPDATE
def update_task(task_id: int, new_text: str) -> bool:
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("UPDATE tasks SET text = ? WHERE id = ?", (new_text, task_id))
    conn.commit()
    updated = cursor.rowcount > 0
    conn.close()
    return updated

# D - DELETE
def delete_task(task_id: int) -> bool:
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("DELETE FROM tasks WHERE id = ?", (task_id,))
    conn.commit()
    deleted = cursor.rowcount > 0
    conn.close()
    return deleted

def delete_all_tasks():
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("DELETE FROM tasks")
    conn.commit()
    conn.close()
```

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout, QLabel, QLineEdit, QListWidget, QPushButton
from PySide6.QtCore import Qt, Slot
import repository  # Our data layer

class TodoWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        self.label_info = QLabel("My Tasks - Industry Standard CRUD")
        self.label_info.setObjectName("label_info")

        self.input_task = QLineEdit()
        self.input_task.setObjectName("input_name")
        self.input_task.setPlaceholderText("e.g. Learn Repository Pattern")

        self.list_tasks = QListWidget()
        self.list_tasks.setObjectName("list_tasks")

        self.label_result = QLabel("Ready")
        self.label_result.setObjectName("label_result")

        self.btn_add = QPushButton("CREATE")
        self.btn_add.setObjectName("btn_add")
        self.btn_update = QPushButton("UPDATE")
        self.btn_update.setObjectName("btn_update")
        self.btn_delete = QPushButton("DELETE")
        self.btn_delete.setObjectName("btn_delete")
        self.btn_clear = QPushButton("Clear All")
        self.btn_clear.setObjectName("btn_clear")

        # Layout
        main_layout = QVBoxLayout(self)
        main_layout.setSpacing(12)
        main_layout.setContentsMargins(20, 20, 20, 20)

        btn_row1 = QHBoxLayout()
        btn_row1.addWidget(self.btn_add)
        btn_row1.addWidget(self.btn_update)

        btn_row2 = QHBoxLayout()
        btn_row2.addWidget(self.btn_delete)
        btn_row2.addWidget(self.btn_clear)

        main_layout.addWidget(self.label_info)
        main_layout.addWidget(self.input_task)
        main_layout.addLayout(btn_row1)
        main_layout.addWidget(self.list_tasks)
        main_layout.addLayout(btn_row2)
        main_layout.addWidget(self.label_result)

        # Connections
        self.btn_add.clicked.connect(self.on_create)
        self.btn_update.clicked.connect(self.on_update)
        self.btn_delete.clicked.connect(self.on_delete)
        self.btn_clear.clicked.connect(self.on_clear)
        self.list_tasks.itemDoubleClicked.connect(self.on_item_selected)

        self.refresh_list()

    def refresh_list(self):
        """ R - READ: Single source of truth for displaying data """
        self.list_tasks.clear()
        tasks = repository.get_all_tasks()
        for row in tasks:
            self.list_tasks.addItem(row["text"])
            item = self.list_tasks.item(self.list_tasks.count() - 1)
            item.setData(Qt.ItemDataRole.UserRole, row["id"])  # Hide DB id in UI item
        
        self.label_result.setText(f"Loaded {len(tasks)} tasks from DB")

    @Slot()
    def on_create(self):
        text = self.input_task.text().strip()
        if not text:
            self.label_result.setText("Cannot create empty task!")
            return
        new_id = repository.create_task(text)
        self.input_task.clear()
        self.refresh_list()
        self.label_result.setText(f"CREATED ID {new_id}: {text}")

    @Slot()
    def on_item_selected(self):
        item = self.list_tasks.currentItem()
        if item:
            self.input_task.setText(item.text())

    @Slot()
    def on_update(self):
        item = self.list_tasks.currentItem()
        if not item:
            self.label_result.setText("Select a task to UPDATE!")
            return
        new_text = self.input_task.text().strip()
        if not new_text:
            return
        task_id = item.data(Qt.ItemDataRole.UserRole)
        repository.update_task(task_id, new_text)
        self.refresh_list()
        self.label_result.setText(f"UPDATED ID {task_id}")

    @Slot()
    def on_delete(self):
        item = self.list_tasks.currentItem()
        if not item:
            self.label_result.setText("Select a task to DELETE!")
            return
        task_id = item.data(Qt.ItemDataRole.UserRole)
        repository.delete_task(task_id)
        self.input_task.clear()
        self.refresh_list()
        self.label_result.setText(f"DELETED ID {task_id}")

    @Slot()
    def on_clear(self):
        repository.delete_all_tasks()
        self.refresh_list()
        self.label_result.setText("All tasks deleted from DB")
```
