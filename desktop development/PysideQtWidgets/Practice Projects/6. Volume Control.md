`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QSlider, QProgressBar, QPushButton
from PySide6.QtCore import Qt, Slot

class VolumeWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        # label
        self.label_info = QLabel("Adjust Volume")
        self.label_info.setObjectName("label_info")
        self.label_info.setAlignment(Qt.AlignmentFlag.AlignCenter)

        # slider - the new widget
        self.slider_volume = QSlider(Qt.Orientation.Horizontal)
        self.slider_volume.setObjectName("slider_volume")
        self.slider_volume.setMinimum(0)
        self.slider_volume.setMaximum(100)
        self.slider_volume.setValue(50)
        self.slider_volume.setTickPosition(QSlider.TickPosition.TicksBelow)
        self.slider_volume.setTickInterval(10)

        # progress bar
        self.progress_volume = QProgressBar()
        self.progress_volume.setObjectName("progress_volume")
        self.progress_volume.setMinimum(0)
        self.progress_volume.setMaximum(100)
        self.progress_volume.setValue(50)
        self.progress_volume.setTextVisible(False)

        # readonly status
        self.input_readonly = QLabel("Volume: 50% - Medium")
        self.input_readonly.setObjectName("input_readonly")
        self.input_readonly.setAlignment(Qt.AlignmentFlag.AlignCenter)

        # result
        self.label_result = QLabel("Volume is at 50%")
        self.label_result.setObjectName("label_result")
        self.label_result.setAlignment(Qt.AlignmentFlag.AlignCenter)

        # button
        self.btn_get = QPushButton("Mute")
        self.btn_get.setObjectName("btn_get")

        # === Layout ====
        layout = QVBoxLayout(self)
        layout.setSpacing(15)
        layout.setContentsMargins(30, 30, 30, 30)
        layout.addWidget(self.label_info)
        layout.addWidget(self.slider_volume)
        layout.addWidget(self.progress_volume)
        layout.addWidget(self.input_readonly)
        layout.addWidget(self.label_result)
        layout.addWidget(self.btn_get)
        layout.addStretch()

        # connections
        self.slider_volume.valueChanged.connect(self.on_volume_changed)
        self.btn_get.clicked.connect(self.on_mute)

        # init
        self.on_volume_changed(50)

    @Slot(int)
    def on_volume_changed(self, value):
        # link slider to progress bar
        self.progress_volume.setValue(value)

        # update labels
        if value == 0:
            status = "Muted"
            self.label_result.setText("🔇 Muted")
        elif value < 30:
            status = "Low"
            self.label_result.setText(f"🔈 Volume is at {value}% - Low")
        elif value < 70:
            status = "Medium"
            self.label_result.setText(f"🔉 Volume is at {value}% - Medium")
        else:
            status = "High"
            self.label_result.setText(f"🔊 Volume is at {value}% - High")

        self.input_readonly.setText(f"Volume: {value}% - {status}")

        # update mute button text
        if value == 0:
            self.btn_get.setText("Unmute")
        else:
            self.btn_get.setText("Mute")

    @Slot()
    def on_mute(self):
        if self.slider_volume.value() == 0:
            self.slider_volume.setValue(50)
        else:
            self.slider_volume.setValue(0)
```
