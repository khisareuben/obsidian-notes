```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from pathlib import Path
from widget import CalculatorWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("07 - Simple Calculator")
        self.resize(1000, 700)
        self.setCentralWidget(CalculatorWidget())

        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout, QLabel, QComboBox, QSpinBox, QDoubleSpinBox, QPushButton
from PySide6.QtCore import Qt, Slot

class CalculatorWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        # title
        self.label_info = QLabel("Simple Calculator")
        self.label_info.setObjectName("label_info")
        self.label_info.setAlignment(Qt.AlignmentFlag.AlignCenter)

        # first number - QDoubleSpinBox allows decimals
        self.spin_first = QDoubleSpinBox()
        self.spin_first.setObjectName("spin_number")
        self.spin_first.setRange(-1000, 1000)
        self.spin_first.setValue(10)
        self.spin_first.setDecimals(2)
        self.spin_first.setSingleStep(1)

        # operation combo
        self.combo_op = QComboBox()
        self.combo_op.setObjectName("combo_op")
        self.combo_op.addItems(["+", "-", "×", "÷"])

        # second number
        self.spin_second = QDoubleSpinBox()
        self.spin_second.setObjectName("spin_number")
        self.spin_second.setRange(-1000, 1000)
        self.spin_second.setValue(5)
        self.spin_second.setDecimals(2)
        self.spin_second.setSingleStep(1)

        # readonly formula preview
        self.input_readonly = QLabel("Formula: 10 + 5")
        self.input_readonly.setObjectName("input_readonly")
        self.input_readonly.setAlignment(Qt.AlignmentFlag.AlignCenter)

        # result
        self.label_result = QLabel("Result = 15")
        self.label_result.setObjectName("label_result")
        self.label_result.setAlignment(Qt.AlignmentFlag.AlignCenter)

        # button
        self.btn_get = QPushButton("Calculate")
        self.btn_get.setObjectName("btn_get")

        # === Layout ====
        main_layout = QVBoxLayout(self)
        main_layout.setSpacing(15)
        main_layout.setContentsMargins(20, 20, 20, 20)

        input_layout = QHBoxLayout()
        input_layout.setSpacing(10)
        input_layout.addWidget(self.spin_first, 2)
        input_layout.addWidget(self.combo_op, 1)
        input_layout.addWidget(self.spin_second, 2)

        main_layout.addWidget(self.label_info)
        main_layout.addLayout(input_layout)
        main_layout.addWidget(self.input_readonly)
        main_layout.addWidget(self.label_result)
        main_layout.addWidget(self.btn_get)
        main_layout.addStretch()

        # connections - live update
        self.spin_first.valueChanged.connect(self.update_formula)
        self.spin_second.valueChanged.connect(self.update_formula)
        self.combo_op.currentTextChanged.connect(self.update_formula)
        self.btn_get.clicked.connect(self.on_calculate)

        self.update_formula()

    @Slot()
    def update_formula(self):
        a = self.spin_first.value()
        b = self.spin_second.value()
        op = self.combo_op.currentText()
        self.input_readonly.setText(f"Formula: {a} {op} {b}")

    @Slot()
    def on_calculate(self):
        a = self.spin_first.value()
        b = self.spin_second.value()
        op = self.combo_op.currentText()

        if op == "+":
            result = a + b
        elif op == "-":
            result = a - b
        elif op == "×":
            result = a * b
        elif op == "÷":
            if b == 0:
                self.label_result.setText("Error: Cannot divide by zero!")
                return
            result = a / b

        self.label_result.setText(f"Result = {result:.2f}")
```
