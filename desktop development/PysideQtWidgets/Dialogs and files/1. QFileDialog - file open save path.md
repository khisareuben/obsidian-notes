```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QPushButton, QLabel, QFileDialog, QTextEdit
from PySide6.QtCore import Slot, QDir
import os

class FileDialogWidget(QWidget):
    def __init__(self):
        super().__init__()

        self.btn_open_file = QPushButton("Open File")
        self.btn_open_file.setObjectName("btn_primary")

        self.btn_open_files = QPushButton("Open Multiple Files")
        self.btn_open_files.setObjectName("btn")

        self.btn_save_file = QPushButton("Save File As...")
        self.btn_save_file.setObjectName("btn")

        self.btn_open_dir = QPushButton("Open Folder")
        self.btn_open_dir.setObjectName("btn")

        self.btn_color = QPushButton("Custom Filter (Images Only)")
        self.btn_color.setObjectName("btn")

        self.label_path = QLabel("No file selected")
        self.label_path.setObjectName("path")

        self.text_details = QTextEdit()
        self.text_details.setObjectName("details")
        self.text_details.setReadOnly(True)
        self.text_details.setPlaceholderText("File details will appear here...")

        main = QVBoxLayout(self)
        main.setContentsMargins(20, 20, 20, 20)
        main.setSpacing(10)
        main.addWidget(self.btn_open_file)
        main.addWidget(self.btn_open_files)
        main.addWidget(self.btn_save_file)
        main.addWidget(self.btn_open_dir)
        main.addWidget(self.btn_color)
        main.addWidget(self.label_path)
        main.addWidget(self.text_details, 1)

        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#path {
                color: #00FF88; background-color: #1E1E1E;
                padding: 10px; border-radius: 6px; border: 1px solid #333; font-size: 11px;
            }
            QTextEdit#details {
                background-color: #1E1E1E; color: #CCC;
                border: 1px solid #333; border-radius: 8px; padding: 10px; font-size: 12px;
            }
            QPushButton#btn_primary {
                background-color: #2E86AB; color: white; padding: 12px;
                border: none; border-radius: 8px; font-weight: bold;
            }
            QPushButton#btn {
                background-color: #1E1E1E; color: #CCC; padding: 10px;
                border: 1px solid #333; border-radius: 8px; text-align: left;
            }
            QPushButton#btn:hover { background-color: #2A2A2A; border: 1px solid #2E86AB; color: #FFF; }
        """)

        self.btn_open_file.clicked.connect(self.open_file)
        self.btn_open_files.clicked.connect(self.open_files)
        self.btn_save_file.clicked.connect(self.save_file)
        self.btn_open_dir.clicked.connect(self.open_dir)
        self.btn_color.clicked.connect(self.open_image)

    @Slot()
    def open_file(self):
        # 1. Single file - MOST COMMON
        path, selected_filter = QFileDialog.getOpenFileName(
            self,
            "Open File", # title
            QDir.homePath(), # start dir
            "All Files (*.*);;Text Files (*.txt);;Python Files (*.py)" # filters
        )
        if path:
            self.show_info("OPEN FILE", path, selected_filter)

    @Slot()
    def open_files(self):
        # 2. Multiple files
        paths, _ = QFileDialog.getOpenFileNames(
            self,
            "Open Multiple Files",
            QDir.homePath(),
            "All Files (*.*)"
        )
        if paths:
            self.label_path.setText(f"Selected {len(paths)} files")
            self.text_details.setPlainText("\n".join(paths))

    @Slot()
    def save_file(self):
        # 3. Save as
        path, _ = QFileDialog.getSaveFileName(
            self,
            "Save File As",
            QDir.homePath() + "/untitled.txt",
            "Text Files (*.txt);;All Files (*.*)"
        )
        if path:
            self.show_info("SAVE AS", path, "")
            # Actually save:
            # with open(path, 'w') as f: f.write("hello")

    @Slot()
    def open_dir(self):
        # 4. Directory
        path = QFileDialog.getExistingDirectory(
            self,
            "Select Folder",
            QDir.homePath(),
            QFileDialog.Option.ShowDirsOnly
        )
        if path:
            self.show_info("FOLDER", path, "")
            files = os.listdir(path)[:20]
            self.text_details.append(f"\n\nContains {len(os.listdir(path))} items:\n" + "\n".join(files))

    @Slot()
    def open_image(self):
        # 5. Image filter example
        path, _ = QFileDialog.getOpenFileName(
            self,
            "Open Image",
            "",
            "Images (*.png *.jpg *.jpeg *.webp *.bmp);;PNG (*.png);;JPEG (*.jpg *.jpeg)"
        )
        if path:
            self.show_info("IMAGE", path, "")
            size = os.path.getsize(path)
            self.text_details.append(f"\nSize: {size/1024:.1f} KB")

    def show_info(self, mode, path, filt):
        self.label_path.setText(path)
        info = f"Mode: {mode}\nPath: {path}\nFilter: {filt}\n\nExists: {os.path.exists(path)}\nDir: {os.path.dirname(path)}\nFile: {os.path.basename(path)}"
        self.text_details.setPlainText(info)
```
