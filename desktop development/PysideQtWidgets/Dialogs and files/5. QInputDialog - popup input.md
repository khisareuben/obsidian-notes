from PySide6.QtWidgets import QApplication, QWidget, QVBoxLayout, QPushButton, QLabel, QInputDialog
import sys

class Test(QWidget):
    def __init__(self):
        super().__init__()
        l = QVBoxLayout(self)
        self.label = QLabel("Result: -")
        b1 = QPushButton("Get Text")
        b2 = QPushButton("Get Int")
        b3 = QPushButton("Get Double")
        b4 = QPushButton("Get Item from List")

        l.addWidget(self.label); l.addWidget(b1); l.addWidget(b2); l.addWidget(b3); l.addWidget(b4)

        b1.clicked.connect(self.get_text)
        b2.clicked.connect(self.get_int)
        b3.clicked.connect(self.get_double)
        b4.clicked.connect(self.get_item)

    def get_text(self):
        text, ok = QInputDialog.getText(self, "Name", "Customer name:")
        if ok: self.label.setText(f"Text: {text}")

    def get_int(self):
        val, ok = QInputDialog.getInt(self, "Amount", "KES:", value=1000, minValue=0, maxValue=100000, step=100)
        if ok: self.label.setText(f"Int: {val}")

    def get_double(self):
        val, ok = QInputDialog.getDouble(self, "Weight", "Kg:", value=2.5, minValue=0, maxValue=100, decimals=1)
        if ok: self.label.setText(f"Double: {val}")

    def get_item(self):
        item, ok = QInputDialog.getItem(self, "City", "Choose:", ["Mombasa","Nairobi","Kisumu"], 0, False)
        if ok: self.label.setText(f"Item: {item}")

app = QApplication(sys.argv)
w = Test()
w.show()
sys.exit(app.exec())
