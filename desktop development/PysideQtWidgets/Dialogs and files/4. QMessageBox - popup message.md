from PySide6.QtWidgets import QApplication, QWidget, QVBoxLayout, QPushButton, QMessageBox
import sys

class Test(QWidget):
    def __init__(self):
        super().__init__()
        l = QVBoxLayout(self)

        b1 = QPushButton("Information")
        b2 = QPushButton("Warning")
        b3 = QPushButton("Critical Error")
        b4 = QPushButton("Question Yes/No")

        l.addWidget(b1); l.addWidget(b2); l.addWidget(b3); l.addWidget(b4)

        b1.clicked.connect(lambda: QMessageBox.information(self, "Done", "Order #1024 delivered!"))
        b2.clicked.connect(lambda: QMessageBox.warning(self, "Late", "Driver is 10 mins late"))
        b3.clicked.connect(lambda: QMessageBox.critical(self, "Failed", "M-Pesa payment failed"))
        b4.clicked.connect(self.ask)

    def ask(self):
        ret = QMessageBox.question(self, "Confirm", "Cancel order?",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No)
        
        if ret == QMessageBox.StandardButton.Yes:
            print("User said YES")
        else:
            print("User said NO")

app = QApplication(sys.argv)
w = Test()
w.show()
sys.exit(app.exec())
