from PySide6.QtWidgets import QWidget, QVBoxLayout, QPushButton, QLabel, QFontDialog, QTextEdit
from PySide6.QtGui import QFont
from PySide6.QtCore import Slot

class FontWidget(QWidget):
    def __init__(self):
        super().__init__()

        self.current_font = QFont("Segoe UI", 12, QFont.Weight.Bold)

        # === PREVIEW ===
        self.text_preview = QTextEdit("The quick brown fox jumps over the lazy dog\nDelivery App - Mombasa KES 1,250")
        self.text_preview.setObjectName("preview")
        self.text_preview.setFixedHeight(150)
        self.text_preview.setFont(self.current_font)

        self.label_info = QLabel(f"{self.current_font.family()}, {self.current_font.pointSize()}pt")
        self.label_info.setObjectName("info")

        # === BUTTONS ===
        self.btn_pick = QPushButton("Pick Font (Modal)")
        self.btn_pick.setObjectName("btn_primary")

        self.btn_no_native = QPushButton("Custom Dialog (No Native)")
        self.btn_no_native.setObjectName("btn")

        self.btn_live = QPushButton("Live Preview Dialog")
        self.btn_live.setObjectName("btn")

        self.btn_size_only = QPushButton("Apply to All Widgets")
        self.btn_size_only.setObjectName("btn")

        main = QVBoxLayout(self)
        main.setContentsMargins(20, 20, 20, 20)
        main.setSpacing(10)
        main.addWidget(self.text_preview)
        main.addWidget(self.label_info)
        main.addWidget(self.btn_pick)
        main.addWidget(self.btn_no_native)
        main.addWidget(self.btn_live)
        main.addWidget(self.btn_size_only)
        main.addStretch()

        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QTextEdit#preview {
                background-color: #1E1E1E; color: #FFF;
                border: 1px solid #333; border-radius: 10px; padding: 10px;
            }
            QLabel#info {
                color: #00FF88; background-color: #1E1E1E;
                padding: 8px; border-radius: 6px; border: 1px solid #333; font-size: 11px;
            }
            QPushButton#btn_primary {
                background-color: #2E86AB; color: white; padding: 12px;
                border: none; border-radius: 8px; font-weight: bold;
            }
            QPushButton#btn {
                background-color: #1E1E1E; color: #CCC; padding: 10px;
                border: 1px solid #333; border-radius: 8px; text-align: left;
            }
            QPushButton#btn:hover { background-color: #2A2A2A; border: 1px solid #2E86AB; color: #FFF; }
        """)

        self.btn_pick.clicked.connect(self.pick_font)
        self.btn_no_native.clicked.connect(self.pick_no_native)
        self.btn_live.clicked.connect(self.pick_live)
        self.btn_size_only.clicked.connect(self.apply_to_all)

        self.live_dialog = None

    def update_ui(self):
        self.text_preview.setFont(self.current_font)
        self.label_info.setText(f"{self.current_font.family()}, {self.current_font.pointSize()}pt, "
                                f"{'Bold' if self.current_font.bold() else 'Normal'}, "
                                f"{'Italic' if self.current_font.italic() else ''}")

    @Slot()
    def pick_font(self):
        # 1. Static modal - MOST COMMON
        ok, font = QFontDialog.getFont(
            self.current_font,
            self,
            "Choose Font"
        )
        if ok:
            self.current_font = font
            self.update_ui()

    @Slot()
    def pick_no_native(self):
        dialog = QFontDialog(self.current_font, self)
        dialog.setOption(QFontDialog.FontDialogOption.DontUseNativeDialog, True)
        # dialog.setOption(QFontDialog.FontDialogOption.NoButtons, True) # for live
        # dialog.setOption(QFontDialog.FontDialogOption.ScalableFonts, True) # only scalable
        # dialog.setOption(QFontDialog.FontDialogOption.MonospacedFonts, True) # only mono
        if dialog.exec():
            self.current_font = dialog.currentFont()
            self.update_ui()

    @Slot()
    def pick_live(self):
        if self.live_dialog:
            self.live_dialog.close()
        self.live_dialog = QFontDialog(self.current_font, self)
        self.live_dialog.setOption(QFontDialog.FontDialogOption.NoButtons, False)
        self.live_dialog.currentFontChanged.connect(lambda f: (setattr(self, 'current_font', f), self.update_ui()))
        self.live_dialog.show()

    @Slot()
    def apply_to_all(self):
        # Apply to whole widget
        self.setStyleSheet(self.styleSheet() + f"\nQWidget {{ font-family: '{self.current_font.family()}'; }}")
        self.label_info.setText(f"Applied {self.current_font.family()} to all")
