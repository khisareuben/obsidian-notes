from PySide6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout, QPushButton, QLabel, QColorDialog
from PySide6.QtGui import QColor
from PySide6.QtCore import Slot

class ColorWidget(QWidget):
    def __init__(self):
        super().__init__()

        self.current_color = QColor("#2E86AB")

        # === PREVIEW ===
        self.label_preview = QLabel()
        self.label_preview.setObjectName("preview")
        self.label_preview.setFixedHeight(120)
        self.update_preview()

        self.label_hex = QLabel(self.current_color.name().upper())
        self.label_hex.setObjectName("hex")

        self.label_rgb = QLabel(f"RGB({self.current_color.red()}, {self.current_color.green()}, {self.current_color.blue()})")
        self.label_rgb.setObjectName("rgb")

        # === BUTTONS ===
        self.btn_open = QPushButton("Pick Color (Modal)")
        self.btn_open.setObjectName("btn_primary")

        self.btn_custom = QPushButton("Pick with Alpha")
        self.btn_custom.setObjectName("btn")

        self.btn_non_native = QPushButton("Custom Dialog (No Native)")
        self.btn_non_native.setObjectName("btn")

        self.btn_live = QPushButton("Live Dialog (Non-Modal)")
        self.btn_live.setObjectName("btn")

        main = QVBoxLayout(self)
        main.setContentsMargins(20, 20, 20, 20)
        main.setSpacing(10)
        main.addWidget(self.label_preview)
        main.addWidget(self.label_hex)
        main.addWidget(self.label_rgb)
        main.addWidget(self.btn_open)
        main.addWidget(self.btn_custom)
        main.addWidget(self.btn_non_native)
        main.addWidget(self.btn_live)
        main.addStretch()

        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#preview { border-radius: 12px; border: 1px solid #333; }
            QLabel#hex { color: #FFF; font-weight: bold; font-size: 18px; background: transparent; }
            QLabel#rgb { color: #888; font-size: 12px; background: transparent; }
            QPushButton#btn_primary {
                background-color: #2E86AB; color: white; padding: 12px;
                border: none; border-radius: 8px; font-weight: bold;
            }
            QPushButton#btn {
                background-color: #1E1E1E; color: #CCC; padding: 10px;
                border: 1px solid #333; border-radius: 8px; text-align: left;
            }
            QPushButton#btn:hover { background-color: #2A2A2A; border: 1px solid #2E86AB; color: #FFF; }
        """)

        self.btn_open.clicked.connect(self.pick_color)
        self.btn_custom.clicked.connect(self.pick_with_alpha)
        self.btn_non_native.clicked.connect(self.pick_no_native)
        self.btn_live.clicked.connect(self.pick_live)

        self.live_dialog = None

    def update_preview(self):
        self.label_preview.setStyleSheet(f"QLabel#preview {{ background-color: {self.current_color.name()}; border: 1px solid #333; border-radius: 12px; }}")
        self.label_hex.setText(self.current_color.name().upper())
        self.label_rgb.setText(f"RGB({self.current_color.red()}, {self.current_color.green()}, {self.current_color.blue()}) Alpha: {self.current_color.alpha()}")

    @Slot()
    def pick_color(self):
        # 1. Standard - MOST COMMON
        color = QColorDialog.getColor(
            self.current_color, # initial
            self,
            "Pick a Color"
        )
        if color.isValid(): # False if cancelled
            self.current_color = color
            self.update_preview()

    @Slot()
    def pick_with_alpha(self):
        # 2. With alpha channel
        color = QColorDialog.getColor(
            self.current_color,
            self,
            "Pick with Transparency",
            QColorDialog.ColorDialogOption.ShowAlphaChannel
        )
        if color.isValid():
            self.current_color = color
            self.update_preview()

    @Slot()
    def pick_no_native(self):
        # 3. Force custom Qt dialog (stylable, consistent)
        dialog = QColorDialog(self.current_color, self)
        dialog.setOption(QColorDialog.ColorDialogOption.DontUseNativeDialog, True)
        dialog.setOption(QColorDialog.ColorDialogOption.ShowAlphaChannel, True)
        if dialog.exec():
            self.current_color = dialog.currentColor()
            self.update_preview()

    @Slot()
    def pick_live(self):
        # 4. Non-modal live preview
        if self.live_dialog is not None:
            self.live_dialog.close()

        self.live_dialog = QColorDialog(self.current_color, self)
        self.live_dialog.setOption(QColorDialog.ColorDialogOption.NoButtons, False)
        self.live_dialog.currentColorChanged.connect(self.on_live_color)
        self.live_dialog.show()

    @Slot(QColor)
    def on_live_color(self, color: QColor):
        self.current_color = color
        self.update_preview()
