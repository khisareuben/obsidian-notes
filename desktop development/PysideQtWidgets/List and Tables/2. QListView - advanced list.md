```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QListView, QPushButton, QHBoxLayout
from PySide6.QtCore import QStringListModel, Slot
from PySide6.QtGui import QStandardItemModel, QStandardItem

class ListViewWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("QListView - Model Based")
        self.label_title.setObjectName("label_title")

        self.label_info = QLabel("Uses QStringListModel - better for large data")
        self.label_info.setObjectName("label_info")

        self.list_view = QListView()
        self.list_view.setObjectName("list_view")

        # === MODEL - This is the difference ===
        # Option 1: Simple QStringListModel
        self.model = QStringListModel()
        self.model.setStringList([
            "📦 Order #1024 - Mombasa CBD - KES 1,200",
            "📦 Order #1025 - Nyali Mall - KES 3,500",
            "🚚 Order #1026 - Diani Beach - KES 5,000",
            "⚡ Order #1027 - Express - KES 800",
            "📦 Order #1028 - Likoni - KES 1,500",
            "📦 Order #1029 - Bamburi - KES 2,200"
        ])

        # Connect model to view
        self.list_view.setModel(self.model)

        # Alternative Option 2: QStandardItemModel (if you need icons, colors per item)
        # self.model2 = QStandardItemModel()
        # for text in ["Item 1", "Item 2"]:
        # item = QStandardItem(text)
        # self.model2.appendRow(item)
        # self.list_view.setModel(self.model2)

        self.label_result = QLabel("Selected: None")
        self.label_result.setObjectName("label_result")

        self.btn_add = QPushButton("+ Add")
        self.btn_add.setObjectName("btn_add")
        self.btn_remove = QPushButton("Remove")
        self.btn_remove.setObjectName("btn_remove")

        # === LAYOUT ===
        btn_layout = QHBoxLayout()
        btn_layout.addWidget(self.btn_add)
        btn_layout.addWidget(self.btn_remove)

        layout = QVBoxLayout(self)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.setSpacing(12)
        layout.addWidget(self.label_title)
        layout.addWidget(self.label_info)
        layout.addWidget(self.list_view)
        layout.addLayout(btn_layout)
        layout.addWidget(self.label_result)

        # === STYLING - Same as QListWidget ===
        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#label_title { color: #FFF; font-size: 16px; font-weight: bold; background: transparent; }
            QLabel#label_info { color: #666; font-size: 11px; background: transparent; }
            QLabel#label_result {
                color: #00FF88; font-size: 13px; background-color: #1E1E1E;
                padding: 10px; border-radius: 6px; border: 1px solid #333;
            }
            QListView#list_view {
                background-color: #1E1E1E;
                color: #CCCCCC;
                border: 1px solid #333;
                border-radius: 8px;
                padding: 4px;
                outline: none;
                font-size: 13px;
            }
            QListView#list_view::item {
                padding: 12px 10px;
                border-radius: 6px;
                margin: 2px 0px;
            }
            QListView#list_view::item:hover {
                background-color: #2A2A2A;
                color: #FFFFFF;
            }
            QListView#list_view::item:selected {
                background-color: #2E86AB;
                color: #FFFFFF;
                font-weight: bold;
            }
            QScrollBar:vertical { background: #1A1A1A; width: 8px; border-radius: 4px; }
            QScrollBar::handle:vertical { background: #444; min-height: 30px; border-radius: 4px; }
            QScrollBar::handle:vertical:hover { background: #2E86AB; }
            QScrollBar::add-line:vertical, QScrollBar::sub-line:vertical { height: 0px; }
            QPushButton#btn_add { background-color: #2E86AB; color: white; padding: 8px; border: none; border-radius: 6px; font-weight: bold; }
            QPushButton#btn_remove { background-color: #2A2A2A; color: #FF5A5A; padding: 8px; border: 1px solid #333; border-radius: 6px; }
        """)

        # === CONNECT ===
        self.list_view.clicked.connect(self.on_clicked)
        self.btn_add.clicked.connect(self.add_item)
        self.btn_remove.clicked.connect(self.remove_item)

    @Slot()
    def on_clicked(self, index):
        text = self.model.data(index)
        self.label_result.setText(f"Selected: {text} | Row: {index.row()}")

    @Slot()
    def add_item(self):
        row = self.model.rowCount()
        self.model.insertRow(row)
        self.model.setData(self.model.index(row), f"📦 Order #{1030+row} - New")

    @Slot()
    def remove_item(self):
        indexes = self.list_view.selectedIndexes()
        if indexes:
            self.model.removeRow(indexes[0].row())
```

QListWidget vs QListView - when to use what

```python
QListWidget:
  list_widget.addItem("Text") # direct
  Simple, small lists (<1000)
  Beginner friendly

QListView + Model:
  model = QStringListModel(["A","B"])
  list_view.setModel(model)
  For large data, dynamic data, MVC pattern
  Required if you work with databases
  More performant for 10k+ items
```

```python
# 1. Simplest
from PySide6.QtCore import QStringListModel
```
