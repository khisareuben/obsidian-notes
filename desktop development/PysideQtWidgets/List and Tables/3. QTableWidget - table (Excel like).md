from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QTableWidget, QTableWidgetItem, QHeaderView, QPushButton
from PySide6.QtCore import Slot, Qt

class TableWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("Delivery Orders - Table")
        self.label_title.setObjectName("label_title")

        self.table = QTableWidget()
        self.table.setObjectName("table")
        self.table.setRowCount(5)
        self.table.setColumnCount(4)
        self.table.setHorizontalHeaderLabels(["Order ID", "Location", "Amount", "Status"])

        data = [
            ["#1024", "Mombasa CBD", "KES 1,200", "Delivered"],
            ["#1025", "Nyali Mall", "KES 3,500", "In Transit"],
            ["#1026", "Diani Beach", "KES 5,000", "Pending"],
            ["#1027", "Bamburi", "KES 800", "Delivered"],
            ["#1028", "Likoni", "KES 1,500", "Failed"],
        ]

        for row, row_data in enumerate(data):
            for col, value in enumerate(row_data):
                item = QTableWidgetItem(value)
                item.setTextAlignment(Qt.AlignmentFlag.AlignCenter)
                # Make non-editable if you want
                # item.setFlags(item.flags() & ~Qt.ItemFlag.ItemIsEditable)
                self.table.setItem(row, col, item)

        # === TABLE SETTINGS ===
        self.table.horizontalHeader().setSectionResizeMode(QHeaderView.ResizeMode.Stretch)
        self.table.verticalHeader().setVisible(False)
        self.table.setAlternatingRowColors(False)
        self.table.setSelectionBehavior(QTableWidget.SelectionBehavior.SelectRows)
        self.table.setSelectionMode(QTableWidget.SelectionMode.SingleSelection)
        self.table.setShowGrid(False)
        self.table.setEditTriggers(QTableWidget.EditTrigger.NoEditTriggers) # Read-only

        self.label_result = QLabel("Selected: None")
        self.label_result.setObjectName("label_result")

        self.btn_add = QPushButton("+ Add Row")
        self.btn_add.setObjectName("btn_add")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.setSpacing(12)
        layout.addWidget(self.label_title)
        layout.addWidget(self.table)
        layout.addWidget(self.btn_add)
        layout.addWidget(self.label_result)

        # === STYLING ===
        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#label_title { color: #FFF; font-size: 16px; font-weight: bold; background: transparent; }
            QLabel#label_result {
                color: #00FF88; font-size: 12px; background-color: #1E1E1E;
                padding: 8px; border-radius: 6px; border: 1px solid #333;
            }

            /* Table */
            QTableWidget#table {
                background-color: #1E1E1E;
                color: #CCCCCC;
                border: 1px solid #333;
                border-radius: 8px;
                gridline-color: #333;
                outline: none;
                font-size: 12px;
            }
            QTableWidget#table::item {
                padding: 10px;
                border-bottom: 1px solid #2A2A2A;
            }
            QTableWidget#table::item:selected {
                background-color: #2E86AB;
                color: #FFFFFF;
            }
            QTableWidget#table::item:hover {
                background-color: #2A2A2A;
            }

            /* Header */
            QHeaderView::section {
                background-color: #252525;
                color: #888888;
                padding: 10px;
                border: none;
                border-bottom: 1px solid #333;
                font-weight: bold;
                font-size: 11px;
                text-transform: uppercase;
            }

            QScrollBar:vertical { background: #1A1A1A; width: 8px; border-radius: 4px; }
            QScrollBar::handle:vertical { background: #444; border-radius: 4px; }
            QScrollBar::handle:vertical:hover { background: #2E86AB; }
            QScrollBar::add-line:vertical, QScrollBar::sub-line:vertical { height: 0px; }

            QPushButton#btn_add {
                background-color: #2E86AB; color: white; padding: 8px;
                border: none; border-radius: 6px; font-weight: bold;
            }
        """)

        # === CONNECT ===
        self.table.cellClicked.connect(self.on_cell_clicked)
        self.btn_add.clicked.connect(self.add_row)

    @Slot(int, int)
    def on_cell_clicked(self, row, col):
        order_id = self.table.item(row, 0).text()
        location = self.table.item(row, 1).text()
        self.label_result.setText(f"Selected: {order_id} - {location}")

    @Slot()
    def add_row(self):
        row = self.table.rowCount()
        self.table.insertRow(row)
        self.table.setItem(row, 0, QTableWidgetItem(f"#{1030+row}"))
        self.table.setItem(row, 1, QTableWidgetItem("New Location"))
        self.table.setItem(row, 2, QTableWidgetItem("KES 0"))
        self.table.setItem(row, 3, QTableWidgetItem("Pending"))
