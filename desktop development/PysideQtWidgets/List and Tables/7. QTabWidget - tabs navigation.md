from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QTabWidget, QPushButton
from PySide6.QtCore import Slot

class TabsWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE TABS ===
        self.tabs = QTabWidget()
        self.tabs.setObjectName("tabs")
        self.tabs.setDocumentMode(True) # Cleaner look

        # --- Tab 1 ---
        self.tab_orders = QWidget()
        layout1 = QVBoxLayout(self.tab_orders)
        layout1.addWidget(QLabel("📦 Orders List Here"))
        label1 = QLabel("Order #1024 - Delivered")
        label1.setObjectName("content_label")
        layout1.addWidget(label1)
        layout1.addStretch()

        # --- Tab 2 ---
        self.tab_map = QWidget()
        layout2 = QVBoxLayout(self.tab_map)
        layout2.addWidget(QLabel("🗺️ Map View"))
        label2 = QLabel("Live tracking - Mombasa CBD")
        label2.setObjectName("content_label")
        layout2.addWidget(label2)
        layout2.addStretch()

        # --- Tab 3 ---
        self.tab_settings = QWidget()
        layout3 = QVBoxLayout(self.tab_settings)
        layout3.addWidget(QLabel("⚙️ Settings"))
        self.btn = QPushButton("Save Settings")
        self.btn.setObjectName("btn_save")
        layout3.addWidget(self.btn)
        layout3.addStretch()

        # Add tabs to QTabWidget
        self.tabs.addTab(self.tab_orders, "Orders")
        self.tabs.addTab(self.tab_map, "Map")
        self.tabs.addTab(self.tab_settings, "Settings")

        # === RESULT ===
        self.label_result = QLabel("Current Tab: Orders (Index 0)")
        self.label_result.setObjectName("label_result")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setContentsMargins(10, 10, 10, 10)
        layout.setSpacing(10)
        layout.addWidget(self.tabs)
        layout.addWidget(self.label_result)

        # === STYLING ===
        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#label_result {
                color: #00FF88; font-size: 12px; background-color: #1E1E1E;
                padding: 8px; border-radius: 6px; border: 1px solid #333;
            }
            QLabel#content_label {
                color: #CCCCCC; background-color: #1E1E1E;
                padding: 12px; border-radius: 6px; border: 1px solid #333;
                margin-top: 10px;
            }

            /* === TABS === */
            QTabWidget#tabs {
                background-color: #121212;
                border: none;
            }
            QTabWidget#tabs::pane {
                background-color: #1E1E1E;
                border: 1px solid #333;
                border-radius: 8px;
                margin-top: -1px;
                padding: 10px;
            }

            /* Tab Bar */
            QTabBar::tab {
                background-color: #1E1E1E;
                color: #888888;
                padding: 10px 20px;
                border: 1px solid #333;
                border-bottom: none;
                border-top-left-radius: 8px;
                border-top-right-radius: 8px;
                margin-right: 4px;
                font-size: 13px;
                font-weight: bold;
            }
            QTabBar::tab:hover {
                background-color: #2A2A2A;
                color: #FFFFFF;
            }
            QTabBar::tab:selected {
                background-color: #2E86AB;
                color: #FFFFFF;
                border: 1px solid #2E86AB;
                border-bottom: none;
            }
            QTabBar::tab:!selected {
                margin-top: 4px; /* Make selected tab taller */
            }

            QPushButton#btn_save {
                background-color: #2E86AB; color: white;
                padding: 10px; border: none; border-radius: 6px;
                font-weight: bold; margin-top: 10px;
            }
            QPushButton#btn_save:hover { background-color: #3A9AD9; }
        """)

        # === CONNECT ===
        self.tabs.currentChanged.connect(self.on_tab_changed)
        self.btn.clicked.connect(lambda: self.label_result.setText("Settings Saved!"))

    @Slot(int)
    def on_tab_changed(self, index: int):
        name = self.tabs.tabText(index)
        self.label_result.setText(f"Current Tab: {name} (Index {index})")

import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from tabs_widget import TabsWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("22. QTabWidget - Styled")
        self.setFixedSize(500, 400)
        self.ui = TabsWidget()
        self.setCentralWidget(self.ui)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
