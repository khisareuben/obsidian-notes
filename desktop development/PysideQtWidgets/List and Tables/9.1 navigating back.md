from PySide6.QtWidgets improt QStackedWidget

class MyStack(QStackedWidget):
	def __init__(self):
		super().__init__()
		self.history = []
		
	def go_to(self, index):
		self.history.append(self.currentIndex())
		self.setCurrentIndiex(index)
		
	def got_back(self):
		if self.history:
			prev = self.history.pop()
			self.setCurrentIndex(prev)
