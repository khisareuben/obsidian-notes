```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout, QLabel, QStackedWidget, QPushButton
from PySide6.QtCore import Slot

class StackedWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === SIDEBAR BUTTONS (Custom tab bar) ===
        self.btn_orders = QPushButton("📦 Orders")
        self.btn_orders.setObjectName("btn_nav")
        self.btn_orders.setCheckable(True)
        self.btn_orders.setChecked(True)

        self.btn_map = QPushButton("🗺️ Map")
        self.btn_map.setObjectName("btn_nav")
        self.btn_map.setCheckable(True)

        self.btn_settings = QPushButton("⚙️ Settings")
        self.btn_settings.setObjectName("btn_nav")
        self.btn_settings.setCheckable(True)

        # === STACKED WIDGET ===
        self.stack = QStackedWidget()
        self.stack.setObjectName("stack")

        # Page 0
        self.page_orders = QWidget()
        l0 = QVBoxLayout(self.page_orders)
        l0.addWidget(QLabel("PAGE 1: Orders Dashboard"))
        lbl0 = QLabel("Total Orders: 24\nPending: 5\nDelivered: 19")
        lbl0.setObjectName("content")
        l0.addWidget(lbl0)
        l0.addStretch()

        # Page 1
        self.page_map = QWidget()
        l1 = QVBoxLayout(self.page_map)
        l1.addWidget(QLabel("PAGE 2: Live Map"))
        lbl1 = QLabel("Tracking 5 drivers in Mombasa...")
        lbl1.setObjectName("content")
        l1.addWidget(lbl1)
        l1.addStretch()

        # Page 2
        self.page_settings = QWidget()
        l2 = QVBoxLayout(self.page_settings)
        l2.addWidget(QLabel("PAGE 3: Settings"))
        lbl2 = QLabel("Theme: Dark\nNotifications: ON")
        lbl2.setObjectName("content")
        l2.addWidget(lbl2)
        l2.addStretch()

        # Add pages to stack
        self.stack.addWidget(self.page_orders) # index 0
        self.stack.addWidget(self.page_map) # index 1
        self.stack.addWidget(self.page_settings) # index 2

        self.label_result = QLabel("Current: Page 0 - Orders")
        self.label_result.setObjectName("label_result")

        # === LAYOUT ===
        nav_layout = QHBoxLayout()
        nav_layout.setSpacing(8)
        nav_layout.addWidget(self.btn_orders)
        nav_layout.addWidget(self.btn_map)
        nav_layout.addWidget(self.btn_settings)

        main_layout = QVBoxLayout(self)
        main_layout.setContentsMargins(20, 20, 20, 20)
        main_layout.setSpacing(12)
        main_layout.addLayout(nav_layout)
        main_layout.addWidget(self.stack)
        main_layout.addWidget(self.label_result)

        # === STYLING ===
        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#label_result {
                color: #00FF88; font-size: 12px; background-color: #1E1E1E;
                padding: 8px; border-radius: 6px; border: 1px solid #333;
            }
            QLabel#content {
                color: #CCC; background-color: #1E1E1E;
                padding: 15px; border-radius: 8px; border: 1px solid #333;
                margin-top: 10px;
            }
            QStackedWidget#stack {
                background-color: #1E1E1E;
                border: 1px solid #333;
                border-radius: 8px;
            }

            /* Nav buttons = custom tabs */
            QPushButton#btn_nav {
                background-color: #1E1E1E;
                color: #888;
                padding: 10px;
                border: 1px solid #333;
                border-radius: 8px;
                font-weight: bold;
                font-size: 12px;
            }
            QPushButton#btn_nav:hover {
                background-color: #2A2A2A;
                color: #FFFFFF;
                border: 1px solid #444;
            }
            QPushButton#btn_nav:checked {
                background-color: #2E86AB;
                color: #FFFFFF;
                border: 1px solid #2E86AB;
            }
        """)

        # === CONNECT ===
        self.btn_orders.clicked.connect(lambda: self.switch_page(0))
        self.btn_map.clicked.connect(lambda: self.switch_page(1))
        self.btn_settings.clicked.connect(lambda: self.switch_page(2))
        self.stack.currentChanged.connect(self.on_page_changed)

    @Slot(int)
    def switch_page(self, index: int):
        self.stack.setCurrentIndex(index)
        # Update checked state
        self.btn_orders.setChecked(index == 0)
        self.btn_map.setChecked(index == 1)
        self.btn_settings.setChecked(index == 2)

    @Slot(int)
    def on_page_changed(self, index: int):
        names = ["Orders", "Map", "Settings"]
        self.label_result.setText(f"Current: Page {index} - {names[index]}")
```

```python
if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
```
