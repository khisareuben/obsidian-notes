from PySide6.QtWidgets import QApplication, QWidget, QTreeWidget, QTreeWidgetItem, QVBoxLayout, QPushButton
import sys

class Window(QWidget):
    """Learn: QTreeWidget, QTreeWidgetItem, addChild, expand, itemClicked"""
    def __init__(self):
        super().__init__()
        self.setWindowTitle("08 - QTreeWidget - Project Explorer")
        self.resize(500, 400)
        
        self.tree = QTreeWidget()
        self.tree.setHeaderLabels(["Name", "Type", "Size"])
        self.tree.setAlternatingRowColors(True)
        
        # Build fake project tree
        root = QTreeWidgetItem(self.tree, ["MyProject", "Folder", ""])
        src = QTreeWidgetItem(root, ["src", "Folder", ""])
        QTreeWidgetItem(src, ["main.py", "File", "2 KB"])
        QTreeWidgetItem(src, ["utils.py", "File", "1 KB"])
        assets = QTreeWidgetItem(root, ["assets", "Folder", ""])
        QTreeWidgetItem(assets, ["logo.png", "Image", "120 KB"])
        QTreeWidgetItem(root, ["README.md", "File", "1 KB"])
        QTreeWidgetItem(root, ["requirements.txt", "File", "0.5 KB"])
        
        self.tree.expandAll()
        self.tree.itemClicked.connect(lambda item, col: print(f"Clicked: {item.text(0)}"))
        
        btn = QPushButton("Add File to Selected Folder")
        btn.clicked.connect(self.add_file)
        
        layout = QVBoxLayout()
        layout.addWidget(self.tree)
        layout.addWidget(btn)
        self.setLayout(layout)
    
    def add_file(self):
        item = self.tree.currentItem()
        if item:
            if item.text(1) == "Folder":
                QTreeWidgetItem(item, ["new_file.py", "File", "0 KB"])
                item.setExpanded(True)
            else:
                parent = item.parent()
                if parent:
                    QTreeWidgetItem(parent, ["new_file.py", "File", "0 KB"])

if __name__ == "__main__":
    app = QApplication(sys.argv)
    w = Window()
    w.show()
    sys.exit(app.exec())
