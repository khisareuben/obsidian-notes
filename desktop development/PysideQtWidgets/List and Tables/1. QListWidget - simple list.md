from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QListWidget, QListWidgetItem, QPushButton, QHBoxLayout
from PySide6.QtCore import Slot, Qt

class ListWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("Delivery Orders")
        self.label_title.setObjectName("label_title")

        self.list_orders = QListWidget()
        self.list_orders.setObjectName("list_orders")
        self.list_orders.addItem("📦 Order #1024 - Mombasa CBD")
        self.list_orders.addItem("📦 Order #1025 - Nyali Mall")
        self.list_orders.addItem("🚚 Order #1026 - Diani Beach")
        self.list_orders.addItem("⚡ Order #1027 - Express - Bamburi")
        self.list_orders.addItem("📦 Order #1028 - Likoni Ferry")
        self.list_orders.setAlternatingRowColors(False)

        # Enable selection
        self.list_orders.setSelectionMode(QListWidget.SelectionMode.SingleSelection)

        self.label_result = QLabel("Selected: None")
        self.label_result.setObjectName("label_result")

        self.btn_add = QPushButton("+ Add Order")
        self.btn_add.setObjectName("btn_add")

        self.btn_remove = QPushButton("Remove Selected")
        self.btn_remove.setObjectName("btn_remove")

        # === LAYOUT ===
        btn_layout = QHBoxLayout()
        btn_layout.setSpacing(10)
        btn_layout.addWidget(self.btn_add)
        btn_layout.addWidget(self.btn_remove)

        layout = QVBoxLayout(self)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.setSpacing(12)
        layout.addWidget(self.label_title)
        layout.addWidget(self.list_orders)
        layout.addLayout(btn_layout)
        layout.addWidget(self.label_result)

        # === STYLING ===
        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#label_title { color: #FFF; font-size: 16px; font-weight: bold; background: transparent; }
            QLabel#label_result {
                color: #00FF88; font-size: 13px; background-color: #1E1E1E;
                padding: 10px; border-radius: 6px; border: 1px solid #333;
            }

            /* === LIST WIDGET === */
            QListWidget#list_orders {
                background-color: #1E1E1E;
                color: #CCCCCC;
                border: 1px solid #333333;
                border-radius: 8px;
                padding: 4px;
                outline: none;
                font-size: 13px;
            }
            QListWidget#list_orders::item {
                padding: 12px 10px;
                border-radius: 6px;
                margin: 2px 0px;
                border: 1px solid transparent;
            }
            QListWidget#list_orders::item:hover {
                background-color: #2A2A2A;
                color: #FFFFFF;
                border: 1px solid #2A2A2A;
            }
            QListWidget#list_orders::item:selected {
                background-color: #2E86AB;
                color: #FFFFFF;
                border: 1px solid #2E86AB;
                font-weight: bold;
            }
            QListWidget#list_orders::item:selected:hover {
                background-color: #3A9AD9;
            }

            /* Scrollbar (global style reused) */
            QScrollBar:vertical {
                background: #1A1A1A; width: 8px; border-radius: 4px;
            }
            QScrollBar::handle:vertical {
                background: #444; min-height: 30px; border-radius: 4px;
            }
            QScrollBar::handle:vertical:hover { background: #2E86AB; }
            QScrollBar::add-line:vertical, QScrollBar::sub-line:vertical { height: 0px; }

            QPushButton#btn_add {
                background-color: #2E86AB; color: white; padding: 8px;
                border: none; border-radius: 6px; font-weight: bold;
            }
            QPushButton#btn_add:hover { background-color: #3A9AD9; }
            QPushButton#btn_remove {
                background-color: #2A2A2A; color: #FF5A5A; padding: 8px;
                border: 1px solid #333; border-radius: 6px;
            }
            QPushButton#btn_remove:hover { background-color: #3A3A3A; }
        """)

        # === CONNECT ===
        self.list_orders.itemClicked.connect(self.on_item_clicked)
        self.list_orders.currentTextChanged.connect(self.on_text_changed)
        self.btn_add.clicked.connect(self.add_item)
        self.btn_remove.clicked.connect(self.remove_item)

    @Slot(QListWidgetItem)
    def on_item_clicked(self, item: QListWidgetItem):
        self.label_result.setText(f"Clicked: {item.text()}")

    @Slot(str)
    def on_text_changed(self, text: str):
        if text:
            self.label_result.setText(f"Selected: {text}")

    @Slot()
    def add_item(self):
        count = self.list_orders.count() + 1029
        self.list_orders.addItem(f"📦 Order #{count} - New Location")

    @Slot()
    def remove_item(self):
        row = self.list_orders.currentRow()
        if row >= 0:
            self.list_orders.takeItem(row)
