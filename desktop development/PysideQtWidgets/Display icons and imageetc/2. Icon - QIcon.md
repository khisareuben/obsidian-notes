```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout, QLabel, QPushButton, QListWidget
from PySide6.QtGui import QIcon, QPixmap
from PySide6.QtCore import QSize, Slot, Qt

class IconWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ICONS (without files - using pixmaps) ===
        def make_icon(color_hex, text):
            pix = QPixmap(64, 64)
            pix.fill(Qt.GlobalColor.transparent)
            # We use styled buttons to show icon concept, but here's real QIcon usage:
            from PySide6.QtGui import QPainter, QColor
            p = QPixmap(48, 48)
            p.fill(QColor(color_hex))
            painter = QPainter(p)
            painter.setPen(QColor("white"))
            painter.drawText(p.rect(), Qt.AlignmentFlag.AlignCenter, text)
            painter.end()
            return QIcon(p)

        self.icon_home = make_icon("#2E86AB", "H")
        self.icon_order = make_icon("#00FF88", "O")
        self.icon_alert = make_icon("#E74C3C", "!")
        self.icon_mpesa = make_icon("#00B04F", "M")

        # === BUTTONS WITH ICONS ===
        self.btn_home = QPushButton(" Home")
        self.btn_home.setObjectName("btn")
        self.btn_home.setIcon(self.icon_home)
        self.btn_home.setIconSize(QSize(24, 24))

        self.btn_order = QPushButton(" Orders")
        self.btn_order.setObjectName("btn")
        self.btn_order.setIcon(self.icon_order)
        self.btn_order.setIconSize(QSize(24, 24))

        self.btn_alert = QPushButton(" Alerts")
        self.btn_alert.setObjectName("btn_alert")
        self.btn_alert.setIcon(self.icon_alert)
        self.btn_alert.setIconSize(QSize(24, 24))

        self.btn_mpesa = QPushButton(" M-Pesa Pay")
        self.btn_mpesa.setObjectName("btn_primary")
        self.btn_mpesa.setIcon(self.icon_mpesa)
        self.btn_mpesa.setIconSize(QSize(28, 28))

        # === LIST WITH ICONS ===
        self.list = QListWidget()
        self.list.setObjectName("list")
        self.list.setIconSize(QSize(24, 24))
        self.list.addItem(" Dashboard")
        self.list.item(0).setIcon(self.icon_home)
        self.list.addItem(" Orders")
        self.list.item(1).setIcon(self.icon_order)
        self.list.addItem(" Payments")
        self.list.item(2).setIcon(self.icon_mpesa)
        self.list.addItem(" Alerts")
        self.list.item(3).setIcon(self.icon_alert)

        self.label_result = QLabel("QIcon: pixmap + states + themes")
        self.label_result.setObjectName("result")

        row = QHBoxLayout()
        row.addWidget(self.btn_home)
        row.addWidget(self.btn_order)
        row.addWidget(self.btn_alert)

        main = QVBoxLayout(self)
        main.setContentsMargins(20, 20, 20, 20)
        main.setSpacing(12)
        main.addLayout(row)
        main.addWidget(self.btn_mpesa)
        main.addWidget(QLabel("QListWidget with Icons:"))
        main.addWidget(self.list)
        main.addWidget(self.label_result)

        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel { color: #FFF; font-weight: bold; background: transparent; font-size: 13px; }
            QLabel#result {
                color: #00FF88; font-size: 11px; background-color: #1E1E1E;
                padding: 8px; border-radius: 6px; border: 1px solid #333; font-weight: normal;
            }
            QPushButton#btn {
                background-color: #1E1E1E; color: #CCC;
                padding: 10px; border: 1px solid #333; border-radius: 8px;
                text-align: left;
            }
            QPushButton#btn:hover { background-color: #2A2A2A; border: 1px solid #2E86AB; color: #FFF; }
            QPushButton#btn_alert {
                background-color: #2A1E1E; color: #E74C3C;
                padding: 10px; border: 1px solid #5A2A2A; border-radius: 8px; text-align: left;
            }
            QPushButton#btn_primary {
                background-color: #2E86AB; color: white;
                padding: 12px; border: none; border-radius: 8px; font-weight: bold;
            }
            QListWidget#list {
                background-color: #1E1E1E; border: 1px solid #333; border-radius: 8px; padding: 5px;
            }
            QListWidget#list::item { padding: 10px; border-radius: 6px; color: #CCC; }
            QListWidget#list::item:selected { background-color: #2E86AB; color: white; }
            QListWidget#list::item:hover { background-color: #2A2A2A; }
        """)

        self.btn_home.clicked.connect(lambda: self.on_click("Home"))
        self.list.currentRowChanged.connect(self.on_list)

    @Slot(str)
    def on_click(self, name):
        self.label_result.setText(f"Clicked: {name}")

    @Slot(int)
    def on_list(self, row):
        if row >= 0:
            self.label_result.setText(f"Selected: {self.list.item(row).text()} with icon")
```

```python
from PySide6.QtGui import QIcon, QPixmap

# 1. From file (best)
icon = QIcon("icons/home.png")
icon = QIcon(":/icons/home.svg") # from qrc resource file

# 2. From QPixmap
pix = QPixmap(64,64)
pix.fill(Qt.green)
icon = QIcon(pix)

# 3. Multiple sizes/states - OS picks best
icon = QIcon()
icon.addFile("icon_16.png", QSize(16,16))
icon.addFile("icon_32.png", QSize(32,32))
icon.addFile("icon_disabled.png", QSize(32,32), QIcon.Mode.Disabled)
icon.addFile("icon_active.png", QSize(32,32), QIcon.Mode.Active, QIcon.State.On)

# Use everywhere
button.setIcon(icon)
button.setIconSize(QSize(24,24))

# Window / App icon
self.setWindowIcon(QIcon("app.png"))
QApplication.setWindowIcon(QIcon("app.png"))

# From theme (Linux)
icon = QIcon.fromTheme("document-save") # freedesktop theme names

# Check
icon.isNull()
icon.availableSizes()
icon.pixmap(QSize(32,32)) # convert back to pixmap

# Standard icons
style = self.style()
icon = style.standardIcon(QStyle.StandardPixmap.SP_DirIcon)
```
