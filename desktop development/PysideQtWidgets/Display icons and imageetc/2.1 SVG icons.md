```python
from pathlib import Path
from PySide6.QtGui import QIcon

# For development: point to your permanent offline folder
# For exe: it will fallback to local assets/icons
ICON_ROOT = Path(__file__).parent.parent / "assets" / "icons"

def icon(name: str) -> QIcon:
    """ Usage: icon('home') """
    path = ICON_ROOT / f"{name}.svg"
    if not path.exists():
        # Try permanent path
        alt = Path("C:/dev/assets/icons") / f"{name}.svg"
        if alt.exists():
            return QIcon(str(alt))
        print(f"[Icon Missing] {name}.svg")
        return QIcon()
    return QIcon(str(path))
```
