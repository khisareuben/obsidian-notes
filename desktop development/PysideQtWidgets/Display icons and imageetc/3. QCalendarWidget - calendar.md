```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout, QCalendarWidget, QLabel, QPushButton
from PySide6.QtCore import QDate, Slot, Qt
from PySide6.QtGui import QTextCharFormat, QColor

class CalendarWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CALENDAR ===
        self.calendar = QCalendarWidget()
        self.calendar.setObjectName("calendar")
        self.calendar.setGridVisible(True)
        self.calendar.setVerticalHeaderFormat(QCalendarWidget.VerticalHeaderFormat.NoVerticalHeader)
        # self.calendar.setNavigationBarVisible(False)
        self.calendar.setSelectedDate(QDate.currentDate())
        self.calendar.setMinimumDate(QDate(2024, 1, 1))
        self.calendar.setMaximumDate(QDate(2027, 12, 31))
        self.calendar.setFirstDayOfWeek(Qt.DayOfWeek.Monday)

        # Highlight delivery dates
        fmt = QTextCharFormat()
        fmt.setBackground(QColor("#2E86AB"))
        fmt.setForeground(QColor("white"))
        fmt.setFontWeight(700)

        # Mark some dates
        today = QDate.currentDate()
        self.calendar.setDateTextFormat(today.addDays(1), fmt)
        self.calendar.setDateTextFormat(today.addDays(3), fmt)

        fmt_weekend = QTextCharFormat()
        fmt_weekend.setForeground(QColor("#E74C3C"))
        self.calendar.setWeekdayTextFormat(Qt.DayOfWeek.Saturday, fmt_weekend)
        self.calendar.setWeekdayTextFormat(Qt.DayOfWeek.Sunday, fmt_weekend)

        self.label_selected = QLabel(f"Selected: {today.toString('dddd, dd MMMM yyyy')}")
        self.label_selected.setObjectName("selected")

        self.label_result = QLabel("QCalendarWidget - Delivery date picker")
        self.label_result.setObjectName("result")

        self.btn_today = QPushButton("Today")
        self.btn_today.setObjectName("btn_primary")
        self.btn_tomorrow = QPushButton("Tomorrow")
        self.btn_tomorrow.setObjectName("btn_secondary")

        row = QHBoxLayout()
        row.addWidget(self.btn_today)
        row.addWidget(self.btn_tomorrow)
        row.addStretch()

        main = QVBoxLayout(self)
        main.setContentsMargins(15, 15, 15, 15)
        main.setSpacing(10)
        main.addWidget(self.calendar)
        main.addLayout(row)
        main.addWidget(self.label_selected)
        main.addWidget(self.label_result)

        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#selected { color: #FFF; font-weight: bold; font-size: 13px; background: transparent; }
            QLabel#result {
                color: #00FF88; font-size: 11px; background-color: #1E1E1E;
                padding: 8px; border-radius: 6px; border: 1px solid #333;
            }
            QPushButton#btn_primary {
                background-color: #2E86AB; color: white; padding: 8px 15px;
                border: none; border-radius: 6px; font-weight: bold;
            }
            QPushButton#btn_secondary {
                background-color: #2A2A2A; color: #CCC; padding: 8px 15px;
                border: 1px solid #333; border-radius: 6px;
            }

            /* Calendar */
            QCalendarWidget#calendar {
                background-color: #1E1E1E;
                border: 1px solid #333;
                border-radius: 10px;
            }
            QCalendarWidget QWidget { background-color: #1E1E1E; color: #CCC; alternate-background-color: #252525; }
            QCalendarWidget QToolButton {
                background-color: #1E1E1E; color: #FFF; border: none; padding: 8px; border-radius: 6px; font-weight: bold;
            }
            QCalendarWidget QToolButton:hover { background-color: #2A2A2A; }
            QCalendarWidget QMenu { background-color: #1E1E1E; color: #CCC; border: 1px solid #333; }
            QCalendarWidget QSpinBox { background-color: #2A2A2A; color: #FFF; border: 1px solid #333; padding: 4px; border-radius: 4px; }

            QCalendarWidget QTableView {
                background-color: #1E1E1E;
                selection-background-color: #2E86AB;
                selection-color: white;
                border: none;
            }
            QCalendarWidget QTableView::item { padding: 8px; border-radius: 4px; }
            QCalendarWidget QTableView::item:selected { background-color: #2E86AB; color: white; font-weight: bold; }
        """)

        self.calendar.clicked.connect(self.on_date_clicked)
        self.calendar.selectionChanged.connect(self.on_selection)
        self.btn_today.clicked.connect(self.go_today)
        self.btn_tomorrow.clicked.connect(self.go_tomorrow)

    @Slot(QDate)
    def on_date_clicked(self, date: QDate):
        self.label_selected.setText(f"Clicked: {date.toString('dddd, dd MMMM yyyy')}")
        self.label_result.setText(f"Delivery scheduled: {date.toString('yyyy-MM-dd')}")

    @Slot()
    def on_selection(self):
        d = self.calendar.selectedDate()
        self.label_selected.setText(f"Selected: {d.toString('dddd, dd MMMM yyyy')}")

    @Slot()
    def go_today(self):
        self.calendar.setSelectedDate(QDate.currentDate())
        self.calendar.setFocus()

    @Slot()
    def go_tomorrow(self):
        self.calendar.setSelectedDate(QDate.currentDate().addDays(1))
```
