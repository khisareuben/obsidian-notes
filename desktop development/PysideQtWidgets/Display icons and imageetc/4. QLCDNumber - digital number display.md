```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout, QLCDNumber, QLabel, QPushButton, QSlider
from PySide6.QtCore import Qt, Slot, QTimer
import random

class LCDWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === LCD 1: Order Count ===
        self.lcd_orders = QLCDNumber()
        self.lcd_orders.setObjectName("lcd_orders")
        self.lcd_orders.setDigitCount(5) # number of digits
        self.lcd_orders.setMode(QLCDNumber.Mode.Dec) # Dec, Hex, Oct, Bin
        self.lcd_orders.setSegmentStyle(QLCDNumber.SegmentStyle.Filled) # Filled, Outline, Flat
        self.lcd_orders.display(1024)

        # === LCD 2: Price (Flat style) ===
        self.lcd_price = QLCDNumber()
        self.lcd_price.setObjectName("lcd_price")
        self.lcd_price.setDigitCount(7)
        self.lcd_price.setSegmentStyle(QLCDNumber.SegmentStyle.Flat)
        self.lcd_price.display("12.99")

        # === LCD 3: Timer / Clock ===
        self.lcd_timer = QLCDNumber()
        self.lcd_timer.setObjectName("lcd_timer")
        self.lcd_timer.setDigitCount(8)
        self.lcd_timer.setSegmentStyle(QLCDNumber.SegmentStyle.Filled)
        self.lcd_timer.display("00:00:00")

        # === CONTROLS ===
        self.slider = QSlider(Qt.Orientation.Horizontal)
        self.slider.setObjectName("slider")
        self.slider.setRange(0, 99999)
        self.slider.setValue(1024)

        self.btn_random = QPushButton("Random")
        self.btn_random.setObjectName("btn_primary")

        self.btn_style = QPushButton("Style: Filled")
        self.btn_style.setObjectName("btn_secondary")
        self.btn_style.setCheckable(True)

        self.label_result = QLabel("QLCDNumber - Old school LED display")
        self.label_result.setObjectName("result")

        self.seconds = 0
        self.timer = QTimer()
        self.timer.timeout.connect(self.tick)
        self.timer.start(1000)

        # Layouts
        def make_row(title, lcd):
            v = QVBoxLayout()
            lbl = QLabel(title)
            lbl.setObjectName("title")
            v.addWidget(lbl)
            v.addWidget(lcd)
            return v

        row1 = QHBoxLayout()
        row1.addLayout(make_row("ORDERS", self.lcd_orders))
        row1.addLayout(make_row("PRICE (KES)", self.lcd_price))

        row2 = QHBoxLayout()
        row2.addLayout(make_row("TIMER", self.lcd_timer))

        ctrl = QHBoxLayout()
        ctrl.addWidget(self.btn_random)
        ctrl.addWidget(self.btn_style)
        ctrl.addStretch()

        main = QVBoxLayout(self)
        main.setContentsMargins(20, 20, 20, 20)
        main.setSpacing(12)
        main.addLayout(row1)
        main.addLayout(row2)
        main.addWidget(QLabel("Adjust Orders:"))
        main.addWidget(self.slider)
        main.addLayout(ctrl)
        main.addWidget(self.label_result)
        main.addStretch()

        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#title { color: #888; font-size: 10px; font-weight: bold; background: transparent; letter-spacing: 1px; }
            QLabel { color: #FFF; font-weight: bold; background: transparent; font-size: 12px; }
            QLabel#result {
                color: #00FF88; font-size: 11px; background-color: #1E1E1E;
                padding: 8px; border-radius: 6px; border: 1px solid #333; font-weight: normal;
            }

            /* LCD STYLING */
            QLCDNumber#lcd_orders {
                background-color: #1E1E1E;
                color: #2E86AB;
                border: 1px solid #333;
                border-radius: 8px;
            }
            QLCDNumber#lcd_price {
                background-color: #1E1E1E;
                color: #00FF88;
                border: 1px solid #333;
                border-radius: 8px;
            }
            QLCDNumber#lcd_timer {
                background-color: #1A1A1A;
                color: #FFAA00;
                border: 1px solid #333;
                border-radius: 8px;
            }

            QSlider#slider::groove:horizontal {
                background-color: #2A2A2A; height: 6px; border-radius: 3px;
            }
            QSlider#slider::handle:horizontal {
                background-color: #2E86AB; width: 18px; height: 18px;
                margin: -6px 0; border-radius: 9px;
            }
            QSlider#slider::sub-page:horizontal { background-color: #2E86AB; border-radius: 3px; }

            QPushButton#btn_primary {
                background-color: #2E86AB; color: white; padding: 8px 15px;
                border: none; border-radius: 6px; font-weight: bold;
            }
            QPushButton#btn_secondary {
                background-color: #2A2A2A; color: #CCC; padding: 8px 15px;
                border: 1px solid #333; border-radius: 6px;
            }
        """)

        self.slider.valueChanged.connect(self.on_slider)
        self.btn_random.clicked.connect(self.on_random)
        self.btn_style.toggled.connect(self.on_style_toggle)

    @Slot(int)
    def on_slider(self, val):
        self.lcd_orders.display(val)
        self.lcd_price.display(f"{val*1.2:.2f}")

    @Slot()
    def on_random(self):
        v = random.randint(0, 99999)
        self.slider.setValue(v)
        self.label_result.setText(f"Random: {v}")

    @Slot(bool)
    def on_style_toggle(self, checked):
        style = QLCDNumber.SegmentStyle.Flat if checked else QLCDNumber.SegmentStyle.Filled
        self.btn_style.setText(f"Style: {'Flat' if checked else 'Filled'}")
        for lcd in [self.lcd_orders, self.lcd_price, self.lcd_timer]:
            lcd.setSegmentStyle(style)

    @Slot()
    def tick(self):
        self.seconds += 1
        m, s = divmod(self.seconds, 60)
        h, m = divmod(m, 60)
        self.lcd_timer.display(f"{h:02d}:{m:02d}:{s:02d}")
```

```python
lcd = QLCDNumber()
lcd.setDigitCount(5) # how many digits fit, e.g. 5 = 99999 max

# Modes
lcd.setMode(QLCDNumber.Dec) # Decimal (default)
lcd.setMode(QLCDNumber.Hex) # Hexadecimal
lcd.setMode(QLCDNumber.Oct) # Octal
lcd.setMode(QLCDNumber.Bin) # Binary

# Segment style
lcd.setSegmentStyle(QLCDNumber.Filled) # 3D bright (default)
lcd.setSegmentStyle(QLCDNumber.Outline) # outline only
lcd.setSegmentStyle(QLCDNumber.Flat) # flat, no 3D

# Display
lcd.display(123)
lcd.display("12.34")
lcd.display(12.34) # int, float, str all work
lcd.display("Err") # can display some letters

# For overflow
lcd.setSmallDecimalPoint(True) # small dot between digits

lcd.value() # get current numeric value
lcd.intValue()

# Signals - QLCDNumber has no valueChanged, but you update it from timer/slider:
timer = QTimer()
timer.timeout.connect(lambda: lcd.display(lcd.value()+1))
timer.start(1000)

# Check overflow
lcd.overflow.connect(lambda: print("Too many digits!"))
```
