from PySide6.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout, QLabel, QPushButton, QFileDialog
from PySide6.QtGui import QPixmap
from PySide6.QtCore import Qt, Slot
import os

class ImageWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === IMAGE LABEL ===
        self.label_image = QLabel()
        self.label_image.setObjectName("label_image")
        self.label_image.setFixedSize(350, 250)
        self.label_image.setAlignment(Qt.AlignmentFlag.AlignCenter)
        self.label_image.setText("No Image\nClick Load")
        self.label_image.setScaledContents(False) # False for proper aspect ratio

        # === CONTROLS ===
        self.btn_load = QPushButton("Load Image")
        self.btn_load.setObjectName("btn_primary")

        self.btn_clear = QPushButton("Clear")
        self.btn_clear.setObjectName("btn_secondary")

        self.btn_fit = QPushButton("Fit: ON")
        self.btn_fit.setObjectName("btn_fit")
        self.btn_fit.setCheckable(True)
        self.btn_fit.setChecked(True)

        self.label_info = QLabel("Size: - | Path: -")
        self.label_info.setObjectName("label_info")

        # Load demo pixmap (color block if no file)
        self.current_pixmap = None
        self.create_demo_pixmap()

        row = QHBoxLayout()
        row.addWidget(self.btn_load)
        row.addWidget(self.btn_clear)
        row.addWidget(self.btn_fit)
        row.addStretch()

        main = QVBoxLayout(self)
        main.setContentsMargins(20, 20, 20, 20)
        main.setSpacing(12)
        main.addWidget(self.label_image, alignment=Qt.AlignmentFlag.AlignCenter)
        main.addLayout(row)
        main.addWidget(self.label_info)
        main.addStretch()

        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#label_image {
                background-color: #1E1E1E;
                border: 2px dashed #333;
                border-radius: 12px;
                color: #555;
                font-size: 14px;
            }
            QLabel#label_info {
                background-color: #1E1E1E;
                color: #00FF88;
                padding: 8px;
                border-radius: 6px;
                border: 1px solid #333;
                font-size: 11px;
            }
            QPushButton#btn_primary {
                background-color: #2E86AB; color: white;
                padding: 8px 15px; border: none; border-radius: 6px; font-weight: bold;
            }
            QPushButton#btn_secondary {
                background-color: #2A2A2A; color: #CCC;
                padding: 8px 15px; border: 1px solid #333; border-radius: 6px;
            }
            QPushButton#btn_fit {
                background-color: #2A2A2A; color: #888;
                padding: 8px 12px; border: 1px solid #333; border-radius: 6px;
            }
            QPushButton#btn_fit:checked {
                background-color: #00FF88; color: #000; border: 1px solid #00FF88;
            }
        """)

        self.btn_load.clicked.connect(self.load_image)
        self.btn_clear.clicked.connect(self.clear_image)
        self.btn_fit.toggled.connect(self.update_display)

    def create_demo_pixmap(self):
        pix = QPixmap(350, 250)
        pix.fill(Qt.GlobalColor.transparent)
        # create a gradient-like demo using style, but show how setPixmap works
        self.current_pixmap = QPixmap(300, 180)
        self.current_pixmap.fill(Qt.GlobalColor.darkCyan)
        self.display_pixmap()

    def display_pixmap(self):
        if not self.current_pixmap:
            return
        if self.btn_fit.isChecked():
            # Keep aspect ratio, smooth
            scaled = self.current_pixmap.scaled(
                self.label_image.size(),
                Qt.AspectRatioMode.KeepAspectRatio,
                Qt.TransformationMode.SmoothTransformation
            )
            self.label_image.setPixmap(scaled)
        else:
            self.label_image.setPixmap(self.current_pixmap)

        self.label_info.setText(f"Size: {self.current_pixmap.width()}x{self.current_pixmap.height()} | Fit: {self.btn_fit.isChecked()}")

    @Slot()
    def load_image(self):
        path, _ = QFileDialog.getOpenFileName(self, "Open Image", "", "Images (*.png *.jpg *.jpeg *.webp)")
        if path:
            pix = QPixmap(path)
            if not pix.isNull():
                self.current_pixmap = pix
                self.label_image.setStyleSheet("QLabel#label_image { border: 1px solid #333; }")
                self.display_pixmap()
                self.label_info.setText(f"Size: {pix.width()}x{pix.height()} | {os.path.basename(path)}")

    @Slot()
    def clear_image(self):
        self.label_image.clear()
        self.label_image.setText("No Image\nClick Load")
        self.current_pixmap = None
        self.label_info.setText("Size: - | Path: -")

    @Slot(bool)
    def update_display(self, checked):
        self.btn_fit.setText(f"Fit: {'ON' if checked else 'OFF'}")
        self.display_pixmap()

    def resizeEvent(self, event):
        super().resizeEvent(event)
        if self.current_pixmap and self.btn_fit.isChecked():
            self.display_pixmap()

# QLabel is your image display
label = QLabel()
pixmap = QPixmap("path/to/image.png") # or .jpg
pixmap = QPixmap(100, 100) # create empty
pixmap.fill(Qt.red)

label.setPixmap(pixmap)

# CRITICAL: scaling with aspect ratio
scaled = pixmap.scaled(
    300, 200,
    Qt.KeepAspectRatio, # KeepAspectRatio | IgnoreAspectRatio | KeepAspectRatioByExpanding
    Qt.SmoothTransformation # Smooth | FastTransformation
)
label.setPixmap(scaled)

# Properties
label.setScaledContents(True) # True = stretch to fill label (distorts), False = manual scale
label.setAlignment(Qt.AlignCenter)
label.pixmap() # get current pixmap
label.clear()

# QImage vs QPixmap
# QPixmap -> for display (optimized for screen)
# QImage -> for pixel manipulation (access pixels, filters)
image = QImage("photo.jpg")
image.setPixel(10, 10, QColor("red").rgb())
pixmap = QPixmap.fromImage(image) # convert to display
