`label_widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel
from PySide6.QtCore import Qt

class LabelWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("DASHBOARD TITLE")
        self.label_normal = QLabel("Normal label for info")
        self.label_status = QLabel("Status: Active")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setSpacing(15)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.addWidget(self.label_title)
        layout.addWidget(self.label_normal)
        layout.addWidget(self.label_status)

        # === STYLING - QSS (Qt CSS) ===
        self.setStyleSheet("""
            /* --- MAIN WIDGET --- */
            QWidget {
                background-color: #121212;
            }

            /* --- TITLE LABEL --- */
            QLabel#title {
                font-size: 22px;
                font-weight: bold;
                color: #FFFFFF;
                qproperty-alignment: AlignCenter;
                padding: 10px;
                background-color: #1E1E1E;
                border-radius: 8px;
                border: 1px solid #333333;
            }

            /* --- NORMAL LABEL --- */
            QLabel#normal {
                font-size: 14px;
                color: #CCCCCC;
                padding: 8px;
            }
                           
            /* --- STATUS LABEL - green pill --- */
            QLabel#status {
                font-size: 13px;
                font-weight: bold;
                color: #00FF88;
                background-color: #00331A;
                padding: 6px 12px;
                border-radius: 12px;
                border: 1px solid #00FF88;
                qproperty-alignment: AlignCenter;
            }
        """)

        # === SET OBJECT NAME TO MATCH CSS #id ===
        self.label_title.setObjectName("title")
        self.label_normal.setObjectName("normal")
        self.label_status.setObjectName("status")

        # You can also set alignment via Qt if you want
        # self.label_title.setAlignment(Qt.AlignmentFlag.AlignCenter)
```

`main.py`
```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from label_widget import LabelWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("1. QLabel - Styled")
        self.setFixedSize(400, 280)
        self.ui = LabelWidget()
        self.setCentralWidget(self.ui)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```
