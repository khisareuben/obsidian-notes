from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QTextEdit, QPushButton
from PySide6.QtCore import Slot

class TextEditWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("Trip Notes:")
        self.label_title.setObjectName("label_title")

        self.text_notes = QTextEdit()
        self.text_notes.setObjectName("text_notes")
        self.text_notes.setPlaceholderText("Write long trip description here...\nCan be multiple lines...")

        self.text_readonly = QTextEdit()
        self.text_readonly.setObjectName("text_readonly")
        self.text_readonly.setText("Log:\n- Driver checked in\n- Package loaded\n- On the way...")
        self.text_readonly.setReadOnly(True)

        self.label_count = QLabel("Characters: 0")
        self.label_count.setObjectName("label_count")

        self.btn_clear = QPushButton("Clear Notes")
        self.btn_clear.setObjectName("btn_clear")

        self.btn_get = QPushButton("Get Text")
        self.btn_get.setObjectName("btn_get")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setSpacing(12)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.addWidget(self.label_title)
        layout.addWidget(self.text_notes)
        layout.addWidget(self.text_readonly)
        layout.addWidget(self.label_count)
        layout.addWidget(self.btn_clear)
        layout.addWidget(self.btn_get)

        # === STYLING - CLEAR #id SYSTEM ===
        self.setStyleSheet("""
            QWidget {
                background-color: #121212;
            }

            /* --- label_title --- */
            QLabel#label_title {
                color: #FFFFFF;
                font-size: 15px;
                font-weight: bold;
                background: transparent;
            }

            /* --- label_count: small counter --- */
            QLabel#label_count {
                color: #888888;
                font-size: 12px;
                background: transparent;
                qproperty-alignment: AlignRight;
            }

            /* --- text_notes: Main editable area --- */
            QTextEdit#text_notes {
                background-color: #1E1E1E;
                color: #FFFFFF;
                font-size: 14px;
                padding: 10px;
                border: 1px solid #333333;
                border-radius: 8px;
            }
            QTextEdit#text_notes:focus {
                border: 1px solid #2E86AB;
                background-color: #252525;
            }

            /* --- text_readonly: Log viewer --- */
            QTextEdit#text_readonly {
                background-color: #181818;
                color: #AAAAAA;
                font-size: 12px;
                font-family: "Consolas";
                padding: 10px;
                border: 1px solid #2A2A2A;
                border-radius: 8px;
            }

            /* --- btn_clear: Red --- */
            QPushButton#btn_clear {
                background-color: #2A2A2A;
                color: #FF5A5A;
                font-size: 13px;
                padding: 8px;
                border: 1px solid #444444;
                border-radius: 8px;
            }
            QPushButton#btn_clear:hover {
                background-color: #3A2A2A;
                border: 1px solid #FF5A5A;
            }

            /* --- btn_get: Blue --- */
            QPushButton#btn_get {
                background-color: #2E86AB;
                color: white;
                font-size: 14px;
                font-weight: bold;
                padding: 10px;
                border: none;
                border-radius: 8px;
            }
            QPushButton#btn_get:hover {
                background-color: #3A9AD9;
            }
        """)

        # === CONNECT ===
        self.btn_clear.clicked.connect(self.text_notes.clear)
        self.btn_get.clicked.connect(self.on_get_text)
        self.text_notes.textChanged.connect(self.on_text_changed)

    @Slot()
    def on_get_text(self):
        text = self.text_notes.toPlainText() # <-- Important: not text() but toPlainText()
        print(text)

    @Slot()
    def on_text_changed(self):
        count = len(self.text_notes.toPlainText())
        self.label_count.setText(f"Characters: {count}")

import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from textedit_widget import TextEditWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("3. QTextEdit - Styled")
        self.setFixedSize(450, 550)
        self.ui = TextEditWidget()
        self.setCentralWidget(self.ui)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
