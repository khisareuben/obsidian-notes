from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QSpinBox, QHBoxLayout
from PySide6.QtCore import Slot

class SpinBoxWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("Trip Package Settings")
        self.label_title.setObjectName("label_title")

        self.label_packages = QLabel("Number of Packages:")
        self.label_packages.setObjectName("label_packages")

        self.spin_packages = QSpinBox()
        self.spin_packages.setObjectName("spin_packages")
        self.spin_packages.setMinimum(1)
        self.spin_packages.setMaximum(100)
        self.spin_packages.setValue(5)
        self.spin_packages.setSingleStep(1) # +1 on each arrow click
        self.spin_packages.setPrefix("Qty: ")
        self.spin_packages.setSuffix(" pkgs")

        self.label_weight = QLabel("Weight per package (kg):")
        self.label_weight.setObjectName("label_weight")

        self.spin_weight = QSpinBox()
        self.spin_weight.setObjectName("spin_weight")
        self.spin_weight.setMinimum(1)
        self.spin_weight.setMaximum(1000)
        self.spin_weight.setValue(50)

        self.label_total = QLabel("Total: 5 pkgs x 50 kg = 250 kg")
        self.label_total.setObjectName("label_total")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setSpacing(12)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.addWidget(self.label_title)
        layout.addWidget(self.label_packages)
        layout.addWidget(self.spin_packages)
        layout.addWidget(self.label_weight)
        layout.addWidget(self.spin_weight)
        layout.addWidget(self.label_total)
        layout.addStretch()

        # === STYLING - THE TRICKY PART FOR SPINBOX ===
        self.setStyleSheet("""
            QWidget {
                background-color: #121212;
            }

            QLabel#label_title {
                color: #FFFFFF;
                font-size: 16px;
                font-weight: bold;
                background: transparent;
            }

            QLabel#label_packages, QLabel#label_weight {
                color: #CCCCCC;
                font-size: 13px;
                background: transparent;
            }

            QLabel#label_total {
                color: #00FF88;
                font-size: 14px;
                font-weight: bold;
                background-color: #1E1E1E;
                padding: 12px;
                border-radius: 8px;
                border: 1px solid #333333;
            }

            /* --- Main SpinBox --- */
            QSpinBox#spin_packages, QSpinBox#spin_weight {
                background-color: #1E1E1E;
                color: #FFFFFF;
                font-size: 14px;
                padding: 8px 10px;
                border: 1px solid #333333;
                border-radius: 8px;
                selection-background-color: #2E86AB;
            }
            QSpinBox#spin_packages:focus, QSpinBox#spin_weight:focus {
                border: 1px solid #2E86AB;
            }

            /* --- The Up/Down Buttons --- */
            QSpinBox#spin_packages::up-button, QSpinBox#spin_weight::up-button {
                background-color: #2A2A2A;
                border: none;
                border-top-right-radius: 8px;
                width: 30px;
            }
            QSpinBox#spin_packages::down-button, QSpinBox#spin_weight::down-button {
                background-color: #2A2A2A;
                border: none;
                border-bottom-right-radius: 8px;
                width: 30px;
            }
            QSpinBox#spin_packages::up-button:hover, QSpinBox#spin_weight::up-button:hover {
                background-color: #2E86AB;
            }
            QSpinBox#spin_packages::down-button:hover, QSpinBox#spin_weight::down-button:hover {
                background-color: #2E86AB;
            }

            /* --- The Arrows --- */
            QSpinBox#spin_packages::up-arrow, QSpinBox#spin_weight::up-arrow {
                width: 10px;
                height: 10px;
            }
            QSpinBox#spin_packages::down-arrow, QSpinBox#spin_weight::down-arrow {
                width: 10px;
                height: 10px;
            }
        """)

        # === CONNECT ===
        self.spin_packages.valueChanged.connect(self.on_value_changed)
        self.spin_weight.valueChanged.connect(self.on_value_changed)

    @Slot(int)
    def on_value_changed(self, value):
        qty = self.spin_packages.value()
        weight = self.spin_weight.value()
        total = qty * weight
        self.label_total.setText(f"Total: {qty} pkgs x {weight} kg = {total} kg")

import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from spinbox_widget import SpinBoxWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("5. QSpinBox - Styled")
        self.setFixedSize(400, 350)
        self.ui = SpinBoxWidget()
        self.setCentralWidget(self.ui)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
