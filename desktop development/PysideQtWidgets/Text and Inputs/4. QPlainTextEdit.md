from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QPlainTextEdit, QPushButton, QHBoxLayout
from PySide6.QtCore import Slot

class PlainTextEditWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("Driver Log Viewer (QPlainTextEdit)")
        self.label_title.setObjectName("label_title")

        self.label_sub = QLabel("Optimized for large logs and code - faster than QTextEdit")
        self.label_sub.setObjectName("label_sub")

        self.text_code = QPlainTextEdit()
        self.text_code.setObjectName("text_code")
        self.text_code.setPlaceholderText("// Paste your code or 10,000 lines of logs here...\n// This widget will NOT lag...")

        self.label_line = QLabel("Line: 1 | Col: 0")
        self.label_line.setObjectName("label_line")

        self.btn_add_log = QPushButton("Add 100 Log Lines")
        self.btn_add_log.setObjectName("btn_add_log")

        self.btn_clear = QPushButton("Clear")
        self.btn_clear.setObjectName("btn_clear")

        # === LAYOUT ===
        btn_layout = QHBoxLayout()
        btn_layout.addWidget(self.btn_add_log)
        btn_layout.addWidget(self.btn_clear)

        layout = QVBoxLayout(self)
        layout.setSpacing(10)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.addWidget(self.label_title)
        layout.addWidget(self.label_sub)
        layout.addWidget(self.text_code)
        layout.addWidget(self.label_line)
        layout.addLayout(btn_layout)

        # === STYLING ===
        self.setStyleSheet("""
            QWidget {
                background-color: #121212;
            }

            QLabel#label_title {
                color: #FFFFFF;
                font-size: 15px;
                font-weight: bold;
                background: transparent;
            }

            QLabel#label_sub {
                color: #888888;
                font-size: 12px;
                background: transparent;
            }

            QLabel#label_line {
                color: #2E86AB;
                font-size: 11px;
                font-family: "Consolas";
                background-color: #1E1E1E;
                padding: 5px 10px;
                border-radius: 4px;
            }

            /* --- text_code: VS Code style --- */
            QPlainTextEdit#text_code {
                background-color: #1E1E1E;
                color: #D4D4D4;
                font-size: 13px;
                font-family: "Consolas", "Courier New", monospace;
                padding: 10px;
                border: 1px solid #333333;
                border-radius: 8px;
                selection-background-color: #264F78;
            }
            QPlainTextEdit#text_code:focus {
                border: 1px solid #2E86AB;
            }

            QPushButton#btn_add_log {
                background-color: #2E86AB;
                color: white;
                font-size: 13px;
                font-weight: bold;
                padding: 9px;
                border: none;
                border-radius: 8px;
            }
            QPushButton#btn_add_log:hover {
                background-color: #3A9AD9;
            }

            QPushButton#btn_clear {
                background-color: #2A2A2A;
                color: #CCCCCC;
                font-size: 13px;
                padding: 9px;
                border: 1px solid #444444;
                border-radius: 8px;
            }
            QPushButton#btn_clear:hover {
                background-color: #333333;
            }
        """)

        # === CONNECT ===
        self.btn_add_log.clicked.connect(self.on_add_logs)
        self.btn_clear.clicked.connect(self.text_code.clear)
        self.text_code.cursorPositionChanged.connect(self.on_cursor_changed)

    @Slot()
    def on_add_logs(self):
        # Simulate fast log appending
        for i in range(100):
            self.text_code.appendPlainText(f"[{i+1:03d}] [INFO] Driver KM-001 delivered package at {i}:23")

    @Slot()
    def on_cursor_changed(self):
        cursor = self.text_code.textCursor()
        line = cursor.blockNumber() + 1
        col = cursor.columnNumber()
        self.label_line.setText(f"Line: {line} | Col: {col}")

import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from plaintextedit_widget import PlainTextEditWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("4. QPlainTextEdit - Styled")
        self.setFixedSize(550, 500)
        self.ui = PlainTextEditWidget()
        self.setCentralWidget(self.ui)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
