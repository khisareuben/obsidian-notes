`main.py`
```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from pathlib import Path
from widget import LineEditWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("line Edit")
        self.resize(1000, 700)
        self.setCentralWidget(LineEditWidget())

        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QLineEdit, QPushButton
from PySide6.QtCore import Qt, Slot

class LineEditWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        #label
        self.label_info = QLabel("Enter Driver Name:")
        self.label_info.setObjectName("label_info")
        self.label_info.setFixedHeight(15)

        #input name
        self.input_name = QLineEdit()
        self.input_name.setObjectName("input_name")
        self.input_name.setPlaceholderText("e.g. Alex")

        #input password
        self.label_password = QLabel("Password")
        self.label_password.setObjectName("label_info")
        self.label_password.setFixedHeight(15)
        self.input_password = QLineEdit()
        self.input_password.setObjectName("input_password")
        self.input_password.setPlaceholderText("Enter password")
        self.input_password.setEchoMode(QLineEdit.EchoMode.Password)

        #readonly
        self.input_readonly = QLineEdit("ReadOnly")
        self.input_readonly.setObjectName("input_readonly")
        self.input_readonly.setReadOnly(True)

        #label
        self.label_result = QLabel("Result will show here")
        self.label_result.setObjectName("label_result")

        #button
        self.btn_get = QPushButton("Get Text")
        self.btn_get.setObjectName("btn_get")

        # === Layout ====
        layout = QVBoxLayout(self)
        layout.setSpacing(12)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.addWidget(self.label_info)
        layout.addWidget(self.input_name)
        layout.addWidget(self.label_password)
        layout.addWidget(self.input_password)
        layout.addWidget(self.input_readonly)
        layout.addWidget(self.label_result)
        layout.addWidget(self.btn_get)

        self.btn_get.clicked.connect(self.on_get_text)

    @Slot()
    def on_get_text(self):
        text = self.input_name.text()
        self.label_result.setText(f"You typed: {text}")
        
```
