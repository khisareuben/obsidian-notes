```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QLineEdit, QPushButton
from PySide6.QtCore import Slot

class LineEditWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_info = QLabel("Enter Driver Name:")
        self.label_info.setObjectName("label_info")

        self.input_name = QLineEdit()
        self.input_name.setObjectName("input_name")
        self.input_name.setPlaceholderText("e.g. Alex")

        self.input_password = QLineEdit()
        self.input_password.setObjectName("input_password")
        self.input_password.setPlaceholderText("Enter password")
        self.input_password.setEchoMode(QLineEdit.EchoMode.Password)

        self.input_readonly = QLineEdit("KM-001 (ReadOnly)")
        self.input_readonly.setObjectName("input_readonly")
        self.input_readonly.setReadOnly(True)

        self.label_result = QLabel("Result will show here")
        self.label_result.setObjectName("label_result")

        self.btn_get = QPushButton("Get Text")
        self.btn_get.setObjectName("btn_get")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setSpacing(12)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.addWidget(self.label_info)
        layout.addWidget(self.input_name)
        layout.addWidget(self.input_password)
        layout.addWidget(self.input_readonly)
        layout.addWidget(self.label_result)
        layout.addWidget(self.btn_get)

        # === STYLING - NOW 100% CLEAR ===
        self.setStyleSheet("""
            /* --- Main background --- */
            QWidget {
                background-color: #121212;
            }

            /* --- label_info: Title for input --- */
            QLabel#label_info {
                color: #FFFFFF;
                font-size: 14px;
                font-weight: bold;
                background: transparent;
            }

            /* --- label_result: Green result box --- */
            QLabel#label_result {
                color: #00FF88;
                font-size: 13px;
                background-color: #1E1E1E;
                padding: 10px;
                border-radius: 6px;
                border: 1px solid #333333;
            }

            /* --- input_name: Normal input --- */
            QLineEdit#input_name {
                background-color: #1E1E1E;
                color: #FFFFFF;
                font-size: 14px;
                padding: 10px 12px;
                border: 1px solid #333333;
                border-radius: 8px;
            }
            QLineEdit#input_name:focus {
                border: 1px solid #2E86AB;
                background-color: #252525;
            }

            /* --- input_password: Password field (same style but you can change) --- */
            QLineEdit#input_password {
                background-color: #1E1E1E;
                color: #FFFFFF;
                font-size: 14px;
                padding: 10px 12px;
                border: 1px solid #333333;
                border-radius: 8px;
            }
            QLineEdit#input_password:focus {
                border: 1px solid #FF5A5A;
            }

            /* --- input_readonly: Disabled look --- */
            QLineEdit#input_readonly {
                background-color: #2A2A2A;
                color: #888888;
                font-size: 13px;
                padding: 10px 12px;
                border: 1px dashed #444444;
                border-radius: 8px;
            }

            /* --- btn_get --- */
            QPushButton#btn_get {
                background-color: #2E86AB;
                color: white;
                font-size: 14px;
                font-weight: bold;
                padding: 10px;
                border: none;
                border-radius: 8px;
            }
            QPushButton#btn_get:hover {
                background-color: #3A9AD9;
            }
            QPushButton#btn_get:pressed {
                background-color: #1E5A7A;
            }
        """)

        # === CONNECT ===
        self.btn_get.clicked.connect(self.on_get_text)

    @Slot()
    def on_get_text(self):
        text = self.input_name.text()
        self.label_result.setText(f"You typed: {text}")
```

```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from lineedit_widget import LineEditWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("2. QLineEdit - Styled")
        self.setFixedSize(400, 380)
        self.ui = LineEditWidget()
        self.setCentralWidget(self.ui)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```
