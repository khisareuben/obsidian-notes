```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from pathlib import Path
from widget import DoubleSpinBoxWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Text Edit")
        self.resize(1000, 700)
        self.setCentralWidget(DoubleSpinBoxWidget())

        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```

`doublespinbox_widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QDoubleSpinBox
from PySide6.QtCore import Slot, Qt

class DoubleSpinBoxWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        # === CREATE ===
        self.label_title = QLabel("Delivery Price Calculator")
        self.label_title.setObjectName("label_title")

        self.label_price = QLabel("Price per kg:")
        self.label_price.setObjectName("label_price")

        self.spin_price = QDoubleSpinBox()
        self.spin_price.setObjectName("spin_price")
        self.spin_price.setMinimum(0.50)
        self.spin_price.setMaximum(100.00)
        self.spin_price.setValue(2.50)
        self.spin_price.setSingleStep(0.10)
        self.spin_price.setDecimals(2) # <-- How many decimals
        self.spin_price.setPrefix("$ ")
        self.spin_price.setSuffix(" / kg")

        self.label_distance = QLabel("Distance:")
        self.label_distance.setObjectName("label_distance")

        self.spin_distance = QDoubleSpinBox()
        self.spin_distance.setObjectName("spin_distance")
        self.spin_distance.setMinimum(0.1)
        self.spin_distance.setMaximum(1000.0)
        self.spin_distance.setValue(12.5)
        self.spin_distance.setSingleStep(0.5)
        self.spin_distance.setDecimals(1)
        self.spin_distance.setSuffix(" km")

        self.label_total = QLabel("Total: $ 2.50 x 12.5 km = $ 31.25")
        self.label_total.setObjectName("label_total")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setSpacing(12)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.addWidget(self.label_title)
        layout.addWidget(self.label_price)
        layout.addWidget(self.spin_price)
        layout.addWidget(self.label_distance)
        layout.addWidget(self.spin_distance)
        layout.addWidget(self.label_total)
        layout.addStretch()

        self.spin_price.valueChanged.connect(self.on_value_changed)
        self.spin_distance.valueChanged.connect(self.on_value_changed)

    @Slot(float)
    def on_value_changed(self, value: float):
        price = self.spin_price.value()
        dist = self.spin_distance.value()
        total = price * dist
        self.label_total.setText(f"Total: $ {price:.2f} x {dist:.1f} km = $ {total:.2f}")
```
