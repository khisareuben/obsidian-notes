```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QProgressBar, QPushButton, QSlider
from PySide6.QtCore import Qt, QTimer, Slot

class ProgressBarWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("Delivery Progress")
        self.label_title.setObjectName("label_title")

        # 1. Normal progress bar
        self.progress_normal = QProgressBar()
        self.progress_normal.setObjectName("progress_normal")
        self.progress_normal.setMinimum(0)
        self.progress_normal.setMaximum(100)
        self.progress_normal.setValue(65)
        self.progress_normal.setTextVisible(True)
        self.progress_normal.setFormat("%p% Complete")

        # 2. Busy / Loading (indeterminate)
        self.progress_busy = QProgressBar()
        self.progress_busy.setObjectName("progress_busy")
        self.progress_busy.setMinimum(0)
        self.progress_busy.setMaximum(0) # 0,0 = busy animation
        self.progress_busy.setTextVisible(False)

        self.label_busy = QLabel("Uploading trip data...")
        self.label_busy.setObjectName("label_busy")

        # 3. Custom text progress
        self.progress_custom = QProgressBar()
        self.progress_custom.setObjectName("progress_custom")
        self.progress_custom.setMinimum(0)
        self.progress_custom.setMaximum(100)
        self.progress_custom.setValue(40)
        self.progress_custom.setFormat("%v / %m km") # %v=value, %m=max

        # Demo controls
        self.slider = QSlider(Qt.Orientation.Horizontal)
        self.slider.setObjectName("slider")
        self.slider.setMinimum(0)
        self.slider.setMaximum(100)
        self.slider.setValue(65)

        self.btn_start = QPushButton("Simulate Delivery")
        self.btn_start.setObjectName("btn_start")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.setSpacing(15)
        layout.addWidget(self.label_title)
        layout.addWidget(self.progress_normal)
        layout.addWidget(self.slider)
        layout.addWidget(self.label_busy)
        layout.addWidget(self.progress_busy)
        layout.addWidget(self.progress_custom)
        layout.addWidget(self.btn_start)
        layout.addStretch()

        # === STYLING - This is where magic happens ===
        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            
            QLabel#label_title {
                color: #FFFFFF;
                font-size: 16px;
                font-weight: bold;
                background: transparent;
            }
            QLabel#label_busy {
                color: #888888;
                font-size: 11px;
                background: transparent;
                margin-top: 10px;
            }

            /* --- progress_normal: Blue --- */
            QProgressBar#progress_normal {
                background-color: #1E1E1E;
                color: #FFFFFF;
                border: 1px solid #333333;
                border-radius: 8px;
                text-align: center;
                height: 24px;
                font-size: 12px;
                font-weight: bold;
            }
            QProgressBar#progress_normal::chunk {
                background-color: #2E86AB;
                border-radius: 7px;
                margin: 1px;
            }

            /* --- progress_busy: Green animated --- */
            QProgressBar#progress_busy {
                background-color: #1E1E1E;
                border: 1px solid #333333;
                border-radius: 8px;
                height: 8px;
            }
            QProgressBar#progress_busy::chunk {
                background-color: #00FF88;
                border-radius: 7px;
            }

            /* --- progress_custom: Gradient --- */
            QProgressBar#progress_custom {
                background-color: #1E1E1E;
                color: #CCCCCC;
                border: 1px solid #333333;
                border-radius: 8px;
                text-align: center;
                height: 24px;
                font-size: 11px;
            }
            QProgressBar#progress_custom::chunk {
                background-color: qlineargradient(x1:0, y1:0, x2:1, y2:0, stop:0 #2E86AB, stop:1 #00FF88);
                border-radius: 7px;
                margin: 1px;
            }

            /* Demo controls */
            QSlider#slider { background: transparent; }
            QSlider#slider::groove:horizontal {
                background: #2A2A2A; height: 4px; border-radius: 2px;
            }
            QSlider#slider::handle:horizontal {
                background: #FFF; width: 14px; height: 14px; margin: -5px 0; border-radius: 7px;
            }

            QPushButton#btn_start {
                background-color: #2E86AB;
                color: white;
                padding: 10px;
                border: none;
                border-radius: 8px;
                font-weight: bold;
            }
            QPushButton#btn_start:hover { background-color: #3A9AD9; }
        """)

        # === CONNECT ===
        self.slider.valueChanged.connect(self.progress_normal.setValue)
        self.slider.valueChanged.connect(self.progress_custom.setValue)
        self.btn_start.clicked.connect(self.start_simulation)

        self.timer = QTimer()
        self.timer.timeout.connect(self.update_simulation)
        self.sim_value = 0

    @Slot()
    def start_simulation(self):
        self.sim_value = 0
        self.timer.start(50)

    @Slot()
    def update_simulation(self):
        self.sim_value += 1
        if self.sim_value > 100:
            self.timer.stop()
            return
        self.progress_normal.setValue(self.sim_value)
        self.progress_custom.setValue(self.sim_value)
        self.slider.setValue(self.sim_value)
```

```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from progressbar_widget import ProgressBarWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("15. QProgressBar - Styled")
        self.setFixedSize(400, 450)
        self.ui = ProgressBarWidget()
        self.setCentralWidget(self.ui)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```
