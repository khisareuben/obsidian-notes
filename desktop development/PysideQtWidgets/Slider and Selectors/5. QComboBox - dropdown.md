```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QComboBox
from PySide6.QtCore import Slot

class ComboBoxWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("Select Delivery Option")
        self.label_title.setObjectName("label_title")

        self.label_vehicle = QLabel("Vehicle Type:")
        self.label_vehicle.setObjectName("label_small")

        self.combo_vehicle = QComboBox()
        self.combo_vehicle.setObjectName("combo_vehicle")
        self.combo_vehicle.addItem("🚲 Bike")
        self.combo_vehicle.addItem("🚗 Car")
        self.combo_vehicle.addItem("🚚 Truck")
        self.combo_vehicle.addItem("✈️ Drone")
        self.combo_vehicle.setCurrentIndex(0)

        self.label_status = QLabel("Payment Method:")
        self.label_status.setObjectName("label_small")

        self.combo_payment = QComboBox()
        self.combo_payment.setObjectName("combo_payment")
        self.combo_payment.addItems(["M-Pesa", "Cash", "Card", "Bank Transfer"])
        self.combo_payment.setEditable(False)

        self.label_result = QLabel("Selected: 🚲 Bike | M-Pesa")
        self.label_result.setObjectName("label_result")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.setSpacing(12)
        layout.addWidget(self.label_title)
        layout.addWidget(self.label_vehicle)
        layout.addWidget(self.combo_vehicle)
        layout.addWidget(self.label_status)
        layout.addWidget(self.combo_payment)
        layout.addWidget(self.label_result)
        layout.addStretch()

        # === STYLING - Important ===
        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#label_title {
                color: #FFFFFF; font-size: 16px; font-weight: bold; background: transparent;
            }
            QLabel#label_small {
                color: #888888; font-size: 12px; background: transparent; margin-top: 8px;
            }
            QLabel#label_result {
                color: #00FF88; font-size: 13px; background-color: #1E1E1E;
                padding: 10px; border-radius: 6px; border: 1px solid #333; margin-top: 10px;
            }

            /* === COMBOBOX MAIN BOX === */
            QComboBox#combo_vehicle, QComboBox#combo_payment {
                background-color: #1E1E1E;
                color: #FFFFFF;
                font-size: 14px;
                padding: 10px 12px;
                border: 1px solid #333333;
                border-radius: 8px;
                min-height: 20px;
            }
            QComboBox#combo_vehicle:hover, QComboBox#combo_payment:hover {
                border: 1px solid #2E86AB;
                background-color: #252525;
            }
            QComboBox#combo_vehicle:focus, QComboBox#combo_payment:focus {
                border: 1px solid #2E86AB;
            }
            QComboBox#combo_vehicle:on, QComboBox#combo_payment:on {
                border: 1px solid #2E86AB;
                border-bottom-left-radius: 0px;
                border-bottom-right-radius: 0px;
            }

            /* Down arrow button */
            QComboBox#combo_vehicle::drop-down, QComboBox#combo_payment::drop-down {
                subcontrol-origin: padding;
                subcontrol-position: top right;
                width: 30px;
                border: none;
                border-top-right-radius: 8px;
                border-bottom-right-radius: 8px;
                background: transparent;
            }
            QComboBox#combo_vehicle::down-arrow, QComboBox#combo_payment::down-arrow {
                width: 12px;
                height: 12px;
                image: none;
                border-left: 5px solid transparent;
                border-right: 5px solid transparent;
                border-top: 6px solid #888888;
                margin-right: 8px;
            }
            QComboBox#combo_vehicle::down-arrow:hover, QComboBox#combo_payment::down-arrow:hover {
                border-top: 6px solid #FFFFFF;
            }

            /* === DROPDOWN LIST === */
            QComboBox#combo_vehicle QAbstractItemView, QComboBox#combo_payment QAbstractItemView {
                background-color: #1E1E1E;
                color: #CCCCCC;
                border: 1px solid #2E86AB;
                border-top: none;
                border-bottom-left-radius: 8px;
                border-bottom-right-radius: 8px;
                selection-background-color: #2E86AB;
                selection-color: #FFFFFF;
                padding: 4px;
                outline: none;
            }
            QComboBox#combo_vehicle QAbstractItemView::item, QComboBox#combo_payment QAbstractItemView::item {
                padding: 10px 12px;
                border-radius: 4px;
                min-height: 20px;
            }
            QComboBox#combo_vehicle QAbstractItemView::item:hover, QComboBox#combo_payment QAbstractItemView::item:hover {
                background-color: #2A2A2A;
                color: #FFFFFF;
            }
        """)

        # === CONNECT ===
        self.combo_vehicle.currentTextChanged.connect(self.on_changed)
        self.combo_payment.currentTextChanged.connect(self.on_changed)

    @Slot(str)
    def on_changed(self, text: str):
        vehicle = self.combo_vehicle.currentText()
        payment = self.combo_payment.currentText()
        self.label_result.setText(f"Selected: {vehicle} | {payment}")
```

```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from combobox_widget import ComboBoxWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("16. QComboBox - Styled")
        self.setFixedSize(400, 400)
        self.ui = ComboBoxWidget()
        self.setCentralWidget(self.ui)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```
