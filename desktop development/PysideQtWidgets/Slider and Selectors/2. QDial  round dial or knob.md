import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from widget import DialWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Begining")
        self.resize(1000, 700)
        self.ui = DialWidget()
        self.setCentralWidget(self.ui)

        try:
            with open("style.qss", "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QDial, QHBoxLayout
from PySide6.QtCore import Slot

class DialWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("Speed Control - QDial")
        self.label_title.setObjectName("label_title")

        self.dial_speed = QDial()
        self.dial_speed.setObjectName("dial_speed")
        self.dial_speed.setMinimum(0)
        self.dial_speed.setMaximum(100)
        self.dial_speed.setValue(50)
        self.dial_speed.setNotchesVisible(True)
        self.dial_speed.setWrapping(False) # True = can rotate full circle

        self.label_value = QLabel("50%")
        self.label_value.setObjectName("label_value")

        self.label_min_max = QLabel("0% ← → 100%")
        self.label_min_max.setObjectName("label_min_max")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.setSpacing(15)
        layout.setAlignment(layout.alignment())
        layout.addWidget(self.label_title)
        layout.addWidget(self.dial_speed, stretch=1)
        layout.addWidget(self.label_value)
        layout.addWidget(self.label_min_max)
        layout.addStretch()

        # Better visual - paint with stylesheet + properties
        self.dial_speed.setStyleSheet("""
            QDial {
                background-color: #1E1E1E;
                border: 2px solid #333333;
            }
        """)

        # === CONNECT ===
        self.dial_speed.valueChanged.connect(self.on_value_changed)

    @Slot(int)
    def on_value_changed(self, v):
        self.label_value.setText(f"{v}%")
        # Change color based on value
        if v > 80:
            self.label_value.setStyleSheet("color: #FF5A5A; font-size: 24px; font-weight: bold; background-color: #1E1E1E; padding: 12px; border-radius: 8px; border: 1px solid #FF5A5A;")
        elif v > 50:
            self.label_value.setStyleSheet("color: #FFD700; font-size: 24px; font-weight: bold; background-color: #1E1E1E; padding: 12px; border-radius: 8px; border: 1px solid #333;")
        else:
            self.label_value.setStyleSheet("color: #00FF88; font-size: 24px; font-weight: bold; background-color: #1E1E1E; padding: 12px; border-radius: 8px; border: 1px solid #333;")
```
