`main.py`

```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from widget import SliderWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Begining")
        self.resize(1000, 700)
        self.ui = SliderWidget()
        self.setCentralWidget(self.ui)

        try:
            with open("style.qss", "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QSlider, QHBoxLayout
from PySide6.QtCore import Qt, Slot

class SliderWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("Delivery Settings")
        self.label_title.setObjectName("label_title")

        self.label_speed_title = QLabel("Speed Limit: 50%")
        self.label_speed_title.setObjectName("label_speed_title")

        self.slider_speed = QSlider(Qt.Orientation.Horizontal)
        self.slider_speed.setObjectName("slider_speed")
        self.slider_speed.setMinimum(0)
        self.slider_speed.setMaximum(100)
        self.slider_speed.setValue(50)
        self.slider_speed.setTickPosition(QSlider.TickPosition.TicksBelow)
        self.slider_speed.setTickInterval(10)

        self.label_ticks = QLabel("0%      25%      50%      75%      100%")
        self.label_ticks.setObjectName("label_ticks")

        # Vertical slider example
        self.label_vol_title = QLabel("Volume")
        self.label_vol_title.setObjectName("label_vol_title")

        self.slider_vol = QSlider(Qt.Orientation.Vertical)
        self.slider_vol.setObjectName("slider_vol")
        self.slider_vol.setMinimum(0)
        self.slider_vol.setMaximum(100)
        self.slider_vol.setValue(70)

        self.label_vol_value = QLabel("70")
        self.label_vol_value.setObjectName("label_vol_value")

        # === LAYOUT ===
        vol_layout = QVBoxLayout()
        vol_layout.setSpacing(8)
        vol_layout.setAlignment(Qt.AlignmentFlag.AlignHCenter)
        vol_layout.addWidget(self.label_vol_title, alignment=Qt.AlignmentFlag.AlignCenter)
        vol_layout.addWidget(self.slider_vol, alignment=Qt.AlignmentFlag.AlignCenter)
        vol_layout.addWidget(self.label_vol_value, alignment=Qt.AlignmentFlag.AlignCenter)

        main_layout = QVBoxLayout(self)
        main_layout.setContentsMargins(20, 20, 20, 20)
        main_layout.setSpacing(12)
        main_layout.addWidget(self.label_title)
        main_layout.addWidget(self.label_speed_title)
        main_layout.addWidget(self.slider_speed)
        main_layout.addWidget(self.label_ticks)

        h_layout = QHBoxLayout()
        h_layout.addStretch()
        h_layout.addLayout(vol_layout)
        h_layout.addStretch()
        
        main_layout.addLayout(h_layout)
        main_layout.addStretch()

        # === CONNECT ===
        self.slider_speed.valueChanged.connect(self.on_speed_changed)
        self.slider_vol.valueChanged.connect(self.on_vol_changed)

    @Slot(int)
    def on_speed_changed(self, v):
        self.label_speed_title.setText(f"Speed Limit: {v}%")
        if v < 30:
            self.label_speed_title.setStyleSheet("color: #FF5A5A; background-color: #1E1E1E; padding: 10px; border-radius: 8px; border: 1px solid #333;")
        elif v < 70:
            self.label_speed_title.setStyleSheet("color: #FFD700; background-color: #1E1E1E; padding: 10px; border-radius: 8px; border: 1px solid #333;")
        else:
            self.label_speed_title.setStyleSheet("color: #00FF88; background-color: #1E1E1E; padding: 10px; border-radius: 8px; border: 1px solid #333;")

    @Slot(int)
    def on_vol_changed(self, v):
        self.label_vol_value.setText(str(v))
```
