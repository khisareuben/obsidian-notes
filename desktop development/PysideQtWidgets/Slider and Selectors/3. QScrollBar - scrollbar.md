from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QScrollBar, QTextEdit
from PySide6.QtCore import Qt, Slot

class ScrollBarWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("Custom QScrollBar")
        self.label_title.setObjectName("label_title")

        # Vertical scrollbar standalone
        self.scroll_v = QScrollBar(Qt.Orientation.Vertical)
        self.scroll_v.setObjectName("scroll_v")
        self.scroll_v.setMinimum(0)
        self.scroll_v.setMaximum(100)
        self.scroll_v.setValue(20)
        self.scroll_v.setPageStep(10)

        # Horizontal scrollbar
        self.scroll_h = QScrollBar(Qt.Orientation.Horizontal)
        self.scroll_h.setObjectName("scroll_h")
        self.scroll_h.setMinimum(0)
        self.scroll_h.setMaximum(100)
        self.scroll_h.setValue(40)

        # A TextEdit to show scrollbar styling in real use
        self.text_area = QTextEdit()
        self.text_area.setObjectName("text_area")
        self.text_area.setPlainText("\n".join([f"Line {i+1} - Delivery log entry..." for i in range(50)]))

        self.label_value = QLabel("V: 20%  |  H: 40%")
        self.label_value.setObjectName("label_value")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.setSpacing(12)
        layout.addWidget(self.label_title)
        layout.addWidget(self.scroll_h)
        layout.addWidget(self.text_area)
        layout.addWidget(self.label_value)

        # === STYLING - The main part ===
        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#label_title {
                color: #FFFFFF;
                font-size: 16px;
                font-weight: bold;
                background: transparent;
            }
            QLabel#label_value {
                color: #888888;
                font-size: 12px;
                background: transparent;
            }

            /* Text area to demo internal scrollbar */
            QTextEdit#text_area {
                background-color: #1E1E1E;
                color: #CCCCCC;
                border: 1px solid #333;
                border-radius: 8px;
                padding: 8px;
            }

            /* === VERTICAL SCROLLBAR === */
            QScrollBar#scroll_v, QScrollBar:vertical {
                background-color: #1A1A1A;
                width: 12px;
                border: none;
                border-radius: 6px;
            }
            QScrollBar::handle:vertical {
                background-color: #444444;
                min-height: 30px;
                border-radius: 6px;
                margin: 2px;
            }
            QScrollBar::handle:vertical:hover {
                background-color: #2E86AB;
            }
            QScrollBar::handle:vertical:pressed {
                background-color: #00FF88;
            }
            QScrollBar::add-line:vertical, QScrollBar::sub-line:vertical {
                height: 0px;
                background: none;
                border: none;
            }
            QScrollBar::add-page:vertical, QScrollBar::sub-page:vertical {
                background: none;
            }

            /* === HORIZONTAL SCROLLBAR === */
            QScrollBar#scroll_h, QScrollBar:horizontal {
                background-color: #1A1A1A;
                height: 12px;
                border: none;
                border-radius: 6px;
            }
            QScrollBar::handle:horizontal {
                background-color: #444444;
                min-width: 30px;
                border-radius: 6px;
                margin: 2px;
            }
            QScrollBar::handle:horizontal:hover {
                background-color: #2E86AB;
            }
            QScrollBar::add-line:horizontal, QScrollBar::sub-line:horizontal {
                width: 0px;
                background: none;
            }
            QScrollBar::add-page:horizontal, QScrollBar::sub-page:horizontal {
                background: none;
            }
        """)

        # === CONNECT ===
        self.scroll_v.valueChanged.connect(self.on_changed)
        self.scroll_h.valueChanged.connect(self.on_changed)

    @Slot(int)
    def on_changed(self, v):
        self.label_value.setText(f"V: {self.scroll_v.value()}%  |  H: {self.scroll_h.value()}%")

### Pro Tip: In your real app, you don't add QScrollBar manually. You style all scrollbars globally with QScrollBar:vertical selector and every QTextEdit, QListView etc. gets it automatically

```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow, QTextEdit, QListWidget

GLOBAL_DARK_SCROLLBAR = """
/* This styles EVERY scrollbar in your entire app */

QScrollBar:vertical {
    background-color: #1A1A1A;
    width: 10px;
    border: none;
    border-radius: 5px;
    margin: 0px;
}
QScrollBar::handle:vertical {
    background-color: #3A3A3A;
    min-height: 40px;
    border-radius: 5px;
    margin: 2px;
}
QScrollBar::handle:vertical:hover {
    background-color: #2E86AB;
}
QScrollBar::handle:vertical:pressed {
    background-color: #00FF88;
}
QScrollBar::add-line:vertical, QScrollBar::sub-line:vertical {
    height: 0px; /* Hide arrow buttons - modern look */
}
QScrollBar::add-page:vertical, QScrollBar::sub-page:vertical {
    background: none;
}

/* Horizontal */
QScrollBar:horizontal {
    background-color: #1A1A1A;
    height: 10px;
    border-radius: 5px;
}
QScrollBar::handle:horizontal {
    background-color: #3A3A3A;
    min-width: 40px;
    border-radius: 5px;
    margin: 2px;
}
QScrollBar::handle:horizontal:hover {
    background-color: #2E86AB;
}
QScrollBar::add-line:horizontal, QScrollBar::sub-line:horizontal {
    width: 0px;
}
"""

if __name__ == "__main__":
    app = QApplication(sys.argv)
    app.setStyleSheet(GLOBAL_DARK_SCROLLBAR) # <-- ONE LINE, affects everything

    window = QMainWindow()
    text = QTextEdit()
    text.setPlainText("\n".join([f"Line {i}" for i in range(100)]))
    window.setCentralWidget(text)
    window.show()
    sys.exit(app.exec())
```

# LEVEL 1: Global - Best for scrollbars (do this)
app.setStyleSheet("QScrollBar:vertical { ... }")

# LEVEL 2: Window level
self.setStyleSheet("QScrollBar:vertical { ... }") # affects all children of this window

# LEVEL 3: Specific widget
self.my_text_edit.setStyleSheet("QScrollBar:vertical { ... }") # only this one
