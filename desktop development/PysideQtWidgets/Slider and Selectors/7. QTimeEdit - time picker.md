from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QTimeEdit
from PySide6.QtCore import QTime, Slot

class TimeWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("Delivery Time Picker")
        self.label_title.setObjectName("label_title")

        self.label_start = QLabel("Start Time:")
        self.label_start.setObjectName("label_small")

        self.time_start = QTimeEdit()
        self.time_start.setObjectName("time_start")
        self.time_start.setTime(QTime(9, 0, 0)) # 9:00 AM
        self.time_start.setDisplayFormat("hh:mm AP") # 12-hour
        self.time_start.setWrapping(True)

        self.label_end = QLabel("End Time (24h):")
        self.label_end.setObjectName("label_small")

        self.time_end = QTimeEdit()
        self.time_end.setObjectName("time_end")
        self.time_end.setTime(QTime.currentTime())
        self.time_end.setDisplayFormat("HH:mm:ss") # 24-hour with seconds
        self.time_end.setWrapping(True)

        self.label_result = QLabel("Working: 09:00 AM - 14:30:00")
        self.label_result.setObjectName("label_result")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.setSpacing(12)
        layout.addWidget(self.label_title)
        layout.addWidget(self.label_start)
        layout.addWidget(self.time_start)
        layout.addWidget(self.label_end)
        layout.addWidget(self.time_end)
        layout.addWidget(self.label_result)
        layout.addStretch()

        # === STYLING ===
        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#label_title { color: #FFFFFF; font-size: 16px; font-weight: bold; background: transparent; }
            QLabel#label_small { color: #888888; font-size: 12px; background: transparent; margin-top: 6px; }
            QLabel#label_result {
                color: #00FF88; font-size: 13px; background-color: #1E1E1E;
                padding: 10px; border-radius: 6px; border: 1px solid #333; margin-top: 10px;
            }

            QTimeEdit#time_start, QTimeEdit#time_end {
                background-color: #1E1E1E;
                color: #FFFFFF;
                padding: 10px 12px;
                border: 1px solid #333;
                border-radius: 8px;
                font-size: 14px;
                min-height: 20px;
            }
            QTimeEdit#time_start:hover, QTimeEdit#time_end:hover {
                border: 1px solid #2E86AB;
            }
            QTimeEdit#time_start:focus, QTimeEdit#time_end:focus {
                border: 1px solid #2E86AB;
            }

            /* Up/Down buttons */
            QTimeEdit#time_start::up-button, QTimeEdit#time_end::up-button,
            QTimeEdit#time_start::down-button, QTimeEdit#time_end::down-button {
                background-color: #2A2A2A;
                border: none;
                width: 20px;
                border-radius: 4px;
            }
            QTimeEdit#time_start::up-button:hover, QTimeEdit#time_end::up-button:hover,
            QTimeEdit#time_start::down-button:hover, QTimeEdit#time_end::down-button:hover {
                background-color: #2E86AB;
            }
            QTimeEdit#time_start::up-arrow, QTimeEdit#time_end::up-arrow {
                width: 8px; height: 8px;
                border-left: 4px solid transparent;
                border-right: 4px solid transparent;
                border-bottom: 5px solid #888;
            }
            QTimeEdit#time_start::down-arrow, QTimeEdit#time_end::down-arrow {
                width: 8px; height: 8px;
                border-left: 4px solid transparent;
                border-right: 4px solid transparent;
                border-top: 5px solid #888;
            }
        """)

        # === CONNECT ===
        self.time_start.timeChanged.connect(self.on_time_changed)
        self.time_end.timeChanged.connect(self.on_time_changed)

    @Slot(QTime)
    def on_time_changed(self, t: QTime):
        start = self.time_start.time().toString("hh:mm AP")
        end = self.time_end.time().toString("HH:mm:ss")
        self.label_result.setText(f"Working: {start} - {end}")
