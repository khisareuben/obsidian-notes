```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QDateEdit, QDateTimeEdit, QCalendarWidget
from PySide6.QtCore import QDate, QDateTime, Slot, Qt

class DateWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("Delivery Date Picker")
        self.label_title.setObjectName("label_title")

        self.label_date = QLabel("Pickup Date:")
        self.label_date.setObjectName("label_small")

        self.date_pickup = QDateEdit()
        self.date_pickup.setObjectName("date_pickup")
        self.date_pickup.setDate(QDate.currentDate())
        self.date_pickup.setCalendarPopup(True) # <-- This shows calendar!
        self.date_pickup.setDisplayFormat("dd MMM yyyy")
        self.date_pickup.setMinimumDate(QDate.currentDate())
        self.date_pickup.setMaximumDate(QDate.currentDate().addDays(30))

        self.label_datetime = QLabel("Delivery Date & Time:")
        self.label_datetime.setObjectName("label_small")

        self.datetime_delivery = QDateTimeEdit()
        self.datetime_delivery.setObjectName("datetime_delivery")
        self.datetime_delivery.setDateTime(QDateTime.currentDateTime())
        self.datetime_delivery.setCalendarPopup(True)
        self.datetime_delivery.setDisplayFormat("dd MMM yyyy - hh:mm AP")

        # Standalone Calendar (optional, big)
        self.calendar = QCalendarWidget()
        self.calendar.setObjectName("calendar")
        self.calendar.setSelectedDate(QDate.currentDate())
        self.calendar.setGridVisible(True)
        self.calendar.setVerticalHeaderFormat(QCalendarWidget.VerticalHeaderFormat.NoVerticalHeader)

        self.label_result = QLabel(f"Selected: {QDate.currentDate().toString('dd MMM yyyy')}")
        self.label_result.setObjectName("label_result")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.setSpacing(10)
        layout.addWidget(self.label_title)
        layout.addWidget(self.label_date)
        layout.addWidget(self.date_pickup)
        layout.addWidget(self.label_datetime)
        layout.addWidget(self.datetime_delivery)
        layout.addWidget(self.calendar)
        layout.addWidget(self.label_result)

        # === STYLING ===
        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#label_title { color: #FFFFFF; font-size: 16px; font-weight: bold; background: transparent; }
            QLabel#label_small { color: #888888; font-size: 12px; background: transparent; margin-top: 6px; }
            QLabel#label_result {
                color: #00FF88; font-size: 13px; background-color: #1E1E1E;
                padding: 10px; border-radius: 6px; border: 1px solid #333;
            }

            /* === QDateEdit === */
            QDateEdit#date_pickup, QDateTimeEdit#datetime_delivery {
                background-color: #1E1E1E;
                color: #FFFFFF;
                padding: 10px 12px;
                border: 1px solid #333;
                border-radius: 8px;
                font-size: 14px;
                min-height: 20px;
            }
            QDateEdit#date_pickup:hover, QDateTimeEdit#datetime_delivery:hover {
                border: 1px solid #2E86AB;
            }
            QDateEdit#date_pickup:focus, QDateTimeEdit#datetime_delivery:focus {
                border: 1px solid #2E86AB;
            }
            QDateEdit#date_pickup::drop-down, QDateTimeEdit#datetime_delivery::drop-down {
                subcontrol-origin: padding;
                subcontrol-position: top right;
                width: 30px;
                border: none;
                background: transparent;
            }
            QDateEdit#date_pickup::down-arrow, QDateTimeEdit#datetime_delivery::down-arrow {
                width: 16px; height: 16px;
                image: url(none);
                /* Simple calendar icon using borders */
            }

            /* === Calendar Popup === */
            QCalendarWidget#calendar {
                background-color: #1E1E1E;
                color: #FFFFFF;
            }
            QCalendarWidget QToolButton {
                background-color: #1E1E1E;
                color: #FFFFFF;
                border: none;
                border-radius: 4px;
                padding: 6px;
                font-weight: bold;
                font-size: 13px;
            }
            QCalendarWidget QToolButton:hover {
                background-color: #2A2A2A;
                color: #2E86AB;
            }
            QCalendarWidget QMenu {
                background-color: #1E1E1E;
                color: #FFFFFF;
                border: 1px solid #333;
            }
            QCalendarWidget QSpinBox {
                background-color: #2A2A2A;
                color: #FFFFFF;
                border: 1px solid #333;
                border-radius: 4px;
                padding: 4px;
            }
            QCalendarWidget QAbstractItemView {
                background-color: #1E1E1E;
                color: #CCCCCC;
                selection-background-color: #2E86AB;
                selection-color: #FFFFFF;
                border: none;
                outline: none;
            }
            QCalendarWidget QAbstractItemView:disabled {
                color: #555555;
            }
        """)

        # === CONNECT ===
        self.date_pickup.dateChanged.connect(self.on_date_changed)
        self.datetime_delivery.dateTimeChanged.connect(self.on_datetime_changed)
        self.calendar.clicked.connect(self.on_calendar_clicked)

    @Slot(QDate)
    def on_date_changed(self, date: QDate):
        self.label_result.setText(f"Pickup: {date.toString('dd MMM yyyy')}")
        self.calendar.setSelectedDate(date)

    @Slot(QDateTime)
    def on_datetime_changed(self, dt: QDateTime):
        self.label_result.setText(f"Delivery: {dt.toString('dd MMM yyyy - hh:mm AP')}")

    @Slot(QDate)
    def on_calendar_clicked(self, date: QDate):
        self.date_pickup.setDate(date)
        self.label_result.setText(f"Calendar: {date.toString('dd MMM yyyy')}")
```

```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from date_widget import DateWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("17. Date Picker - Styled")
        self.setFixedSize(400, 600)
        self.ui = DateWidget()
        self.setCentralWidget(self.ui)

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```
