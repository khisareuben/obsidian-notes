```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QSlider, QLabel
from PySide6.QtCore import Qt

class SliderWidget(QWidget):
    def __init__(self):
        super().__init__()
        layout = QVBoxLayout(self)
        layout.setContentsMargins(30, 30, 30, 30)
        layout.setSpacing(20)

        self.label = QLabel("Value: 50")
        self.label.setAlignment(Qt.AlignmentFlag.AlignCenter)
        self.label.setStyleSheet("font-size: 22px; font-weight: bold;")

        self.slider = QSlider(Qt.Orientation.Horizontal)
        self.slider.setRange(0, 100)
        self.slider.setValue(50)

        layout.addWidget(self.label)
        layout.addWidget(self.slider)

    def get_text(self, value):
        if value < 30:
            return f"Low: {value}%"
        elif value < 70:
            return f"Medium: {value}%"
        else:
            return f"High: {value}%"
```

`Main.py`
```python
from PySide6.QtWidgets import QApplication, QMainWindow
import sys
from widget import SliderWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Slider App")
        self.setFixedSize(400, 200)

        self.ui = SliderWidget()
        self.setCentralWidget(self.ui)

        #connect
        self.ui.slider.valueChanged.connect(self.on_change)
        self.statusBar().showMessage("Ready")

    def on_change(self, value):
        self.ui.label.setText(self.ui.get_text(value))

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```
