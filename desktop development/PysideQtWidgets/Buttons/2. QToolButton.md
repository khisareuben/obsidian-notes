import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from pathlib import Path
from widget import ToolButtonWidget


class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("BUttons")
        self.resize(1000, 700)
        self.setCentralWidget(ToolButtonWidget())

        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())

`ToolButtonWidget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QToolButton, QHBoxLayout
from PySide6.QtCore import Qt

class ToolButtonWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setObjectName("centralWidget")

        self.label_title = QLabel("Toolbar Buttons (QToolButton)")
        self.label_title.setObjectName("label_title")

        # Icon only
        self.btn_menu = QToolButton()
        self.btn_menu.setObjectName("btn_menu")
        self.btn_menu.setText("☰")

        # Text + Icon
        self.btn_add = QToolButton()
        self.btn_add.setObjectName("btn_add")
        self.btn_add.setText("＋ Add")
        self.btn_add.setToolButtonStyle(Qt.ToolButtonStyle.ToolButtonTextBesideIcon)
        self.btn_add.setAutoRaise(True)

        # Checkable tool button
        self.btn_fav = QToolButton()
        self.btn_fav.setObjectName("btn_fav")
        self.btn_fav.setText("♥")
        self.btn_fav.setCheckable(True)

        row = QHBoxLayout()
        row.addWidget(self.btn_menu)
        row.addWidget(self.btn_add)
        row.addWidget(self.btn_fav)
        row.addStretch()

        layout = QVBoxLayout(self)
        layout.setContentsMargins(20,20,20,20)
        layout.setSpacing(12)
        layout.addWidget(self.label_title)
        layout.addLayout(row)
        layout.addStretch()

```
