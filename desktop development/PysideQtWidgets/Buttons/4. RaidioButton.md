from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QRadioButton, QButtonGroup

class RadioButtonWidget(QWidget):
    def __init__(self):
        super().__init__()

        self.label_title = QLabel("Select Vehicle Type")
        self.label_title.setObjectName("label_title")

        self.radio_bike = QRadioButton("Bike (Fast & Cheap)")
        self.radio_bike.setObjectName("radio_bike")
        self.radio_bike.setChecked(True)

        self.radio_car = QRadioButton("Car (Safe)")
        self.radio_car.setObjectName("radio_car")

        self.radio_truck = QRadioButton("Truck (Heavy Load)")
        self.radio_truck.setObjectName("radio_truck")

        # Group them so only one can be selected
        self.group = QButtonGroup(self)
        self.group.addButton(self.radio_bike)
        self.group.addButton(self.radio_car)
        self.group.addButton(self.radio_truck)

        self.label_result = QLabel("Selected: Bike")
        self.label_result.setObjectName("label_result")

        layout = QVBoxLayout(self)
        layout.setContentsMargins(20,20,20,20)
        layout.setSpacing(15)
        layout.addWidget(self.label_title)
        layout.addWidget(self.radio_bike)
        layout.addWidget(self.radio_car)
        layout.addWidget(self.radio_truck)
        layout.addWidget(self.label_result)
        layout.addStretch()

        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#label_title { color: #FFF; font-size: 15px; font-weight: bold; background: transparent; }
            QLabel#label_result {
                color: #00FF88;
                font-size: 13px;
                background-color: #1E1E1E;
                padding: 10px;
                border-radius: 6px;
                margin-top: 10px;
            }

            QRadioButton {
                color: #CCCCCC;
                font-size: 14px;
                spacing: 10px;
            }
            QRadioButton::indicator {
                width: 20px;
                height: 20px;
                background-color: #1E1E1E;
                border: 1px solid #444;
                border-radius: 10px; /* makes it circle */
            }
            QRadioButton::indicator:hover {
                border: 1px solid #2E86AB;
            }
            QRadioButton::indicator:checked {
                background-color: #1E1E1E;
                border: 2px solid #2E86AB;
            }
            /* inner dot when checked */
            QRadioButton::indicator:checked {
                image: none;
            }
        """)

        self.group.buttonClicked.connect(lambda btn: self.label_result.setText(f"Selected: {btn.text()}"))
