from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QPushButton, QHBoxLayout
from PySide6.QtCore import Slot

class PushButtonWidget(QWidget):
    def __init__(self):
        super().__init__()

        self.label_title = QLabel("Button Types")
        self.label_title.setObjectName("label_title")

        # 1. Primary Button
        self.btn_primary = QPushButton("Primary Action")
        self.btn_primary.setObjectName("btn_primary")

        # 2. Secondary / Outline Button
        self.btn_secondary = QPushButton("Secondary")
        self.btn_secondary.setObjectName("btn_secondary")

        # 3. Danger Button
        self.btn_danger = QPushButton("Delete Trip")
        self.btn_danger.setObjectName("btn_danger")

        # 4. Disabled Button
        self.btn_disabled = QPushButton("Disabled")
        self.btn_disabled.setObjectName("btn_disabled")
        self.btn_disabled.setEnabled(False)

        # 5. Icon Button (text + emoji as icon)
        self.btn_icon = QPushButton("▶  Start Delivery")
        self.btn_icon.setObjectName("btn_icon")

        layout = QVBoxLayout(self)
        layout.setSpacing(12)
        layout.setContentsMargins(20,20,20,20)
        layout.addWidget(self.label_title)
        layout.addWidget(self.btn_primary)
        layout.addWidget(self.btn_secondary)
        layout.addWidget(self.btn_danger)
        layout.addWidget(self.btn_disabled)
        layout.addWidget(self.btn_icon)
        layout.addStretch()

        self.setStyleSheet("""
            QWidget { background-color: #121212; }
            QLabel#label_title {
                color: #FFFFFF;
                font-size: 16px;
                font-weight: bold;
                background: transparent;
            }

            /* --- btn_primary: Solid Blue --- */
            QPushButton#btn_primary {
                background-color: #2E86AB;
                color: white;
                font-size: 14px;
                font-weight: bold;
                padding: 12px;
                border: none;
                border-radius: 8px;
            }
            QPushButton#btn_primary:hover { background-color: #3A9AD9; }
            QPushButton#btn_primary:pressed { background-color: #1E5A7A; padding-top: 13px; padding-bottom: 11px;}

            /* --- btn_secondary: Outline --- */
            QPushButton#btn_secondary {
                background-color: transparent;
                color: #CCCCCC;
                font-size: 14px;
                padding: 11px;
                border: 1px solid #444444;
                border-radius: 8px;
            }
            QPushButton#btn_secondary:hover {
                background-color: #1E1E1E;
                color: #FFFFFF;
                border: 1px solid #888888;
            }

            /* --- btn_danger: Red --- */
            QPushButton#btn_danger {
                background-color: #2A1E1E;
                color: #FF5A5A;
                font-size: 14px;
                font-weight: bold;
                padding: 12px;
                border: 1px solid #5A2A2A;
                border-radius: 8px;
            }
            QPushButton#btn_danger:hover {
                background-color: #FF5A5A;
                color: white;
                border: 1px solid #FF5A5A;
            }

            /* --- btn_disabled --- */
            QPushButton#btn_disabled {
                background-color: #1E1E1E;
                color: #555555;

                padding: 12px;
                border: none;

            }
            QPushButton#btn_disabled:disabled {
                background-color: #1A1A1A;
                color: #444444;
            }

            /* --- btn_icon: Gradient --- */
            QPushButton#btn_icon {
                background-color: qlineargradient(x1:0, y1:0, x2:1, y2:0, stop:0 #2E86AB, stop:1 #00FF88);
