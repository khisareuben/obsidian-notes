import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from pathlib import Path
from widget import CheckBoxWidget


class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("BUttons")
        self.resize(1000, 700)
        self.setCentralWidget(CheckBoxWidget())

        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())

`CheckBoxWidget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QCheckBox

class CheckBoxWidget(QWidget):
    def __init__(self):
        super().__init__()

        self.label_title = QLabel("Delivery Options")
        self.label_title.setObjectName("label_title")

        self.check_express = QCheckBox("Express Delivery (+$5)")
        self.check_express.setObjectName("check_express")
        self.check_express.setChecked(True)

        self.check_fragile = QCheckBox("Fragile Package")
        self.check_fragile.setObjectName("check_fragile")

        self.check_notify = QCheckBox("Notify on Arrival")
        self.check_notify.setObjectName("check_notify")

        layout = QVBoxLayout(self)
        layout.setContentsMargins(20,20,20,20)
        layout.setSpacing(15)
        layout.addWidget(self.label_title)
        layout.addWidget(self.check_express)
        layout.addWidget(self.check_fragile)
        layout.addWidget(self.check_notify)
        layout.addStretch()

        self.check_express.stateChanged.connect(lambda s: print(f"Express: {s}"))
```
