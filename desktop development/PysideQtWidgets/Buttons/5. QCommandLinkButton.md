It's that big Windows-style button you see in wizards e.g. windows 10 setup:
```
-> Install Now 
Get the lates version with all features``
```

It's like a Qpushbutton but built to let user choose between 2-3 big option
You rarely use it inside main app. Only on choice screens/ welcome / empty states. Use it when you have 2-4 mutually exclusive choices

`main.py`
```python
import sys
from PySide6.QtWidgets import QApplication, QMainWindow
from pathlib import Path
from widget import CommandLinkWidget


class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("BUttons")
        self.resize(1000, 700)
        self.setCentralWidget(CommandLinkWidget())

        try:
            path = Path(__file__).resolve().parent / "style.qss"
            with open(path, "r") as f:
                self.setStyleSheet(f.read())
        except FileNotFoundError:
            print("style.qss not found")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```

`CommandLinkWidget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QCommandLinkButton

class CommandLinkWidget(QWidget):
    def __init__(self):
        super().__init__()

        # === CREATE ===
        self.label_title = QLabel("How do you want to proceed?")
        self.label_title.setObjectName("label_title")

        self.btn_new = QCommandLinkButton("Start New Delivery", "Create a new trip with driver assignment")
        self.btn_new.setObjectName("btn_new")

        self.btn_continue = QCommandLinkButton("Continue Last Trip", "Resume trip KM-001 from last checkpoint")
        self.btn_continue.setObjectName("btn_continue")

        self.btn_settings = QCommandLinkButton("Open Settings", "Configure app preferences and driver profiles")
        self.btn_settings.setObjectName("btn_settings")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.setSpacing(12)
        layout.addWidget(self.label_title)
        layout.addWidget(self.btn_new)
        layout.addWidget(self.btn_continue)
        layout.addWidget(self.btn_settings)
        layout.addStretch()

        # Description text color is controlled by palette, but we can fake it
        # The small description is automatically styled lighter

        # === CONNECT ===
        self.btn_new.clicked.connect(lambda: print("New Delivery"))
        self.btn_continue.clicked.connect(lambda: print("Continue"))
        self.btn_settings.clicked.connect(lambda: print("Settings"))
```
