```python
from PySide6.QtWidgets import QApplication, QMainWindow
import sys
from widget import MyWidget

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("My App")
        self.setFixedSize(800, 600)

        # central widget - always this 2 lines
        self.ui = MyWidget()
        self.setCentralWidget(self.ui)

        self.statusBar().showMessage("Ready")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())
```

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QPushButton, QLabel
from PySide6.QtCore import Qt, Signal, Slot

class MyWidget(QWidget):

    #Signals (always on top)
    my_signal = Signal(str)

    def __init__(self):
        super().__init__()

        #create widgets (always here)
        self.label = QLabel("Hello")
        self.button = QPushButton("Click Me")

        # Layout (always here)
        layout = QVBoxLayout(self)
        layout.addWidget(self.label)
        layout.addWidget(self.button)

        # connect signals (always at end of __init__)
        self.button.clicked.connect(self.on_button_click)

    # Slots/logic (always outside __init__)
    @Slot()
    def on_button_click(self):
        self.label.setText("Clicked!")
        self.my_signal.emit("Button Clicked")
```
