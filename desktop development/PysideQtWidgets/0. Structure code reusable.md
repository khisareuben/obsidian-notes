`Main.py`
```python
import sys
from PySide6.QtWidgets import QApplication
from main_window import MainWindow

def main():
    app = QApplication(sys.argv)
    app.setApplicationName("Basic Structure")
    app.setStyle("Fusion")

    try:
        with open("style.qss", "r") as f:
            app.setStyleSheet(f.read())
    except FileNotFoundError:
        pass

    window = MainWindow()
    window.show()
    sys.exit(app.exec())

if __name__ == "__main__":
    main()
```

`widget.py`
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel
from PySide6.QtCore import Qt

class LabelWidget(QWidget):
    def __init__(self):
        super().__init__()
        self.setAttribute(Qt.WA_StyledBackground, True) #to use the object name "#centralWidget" in qss
        self.setObjectName("centralWidget")

        # === CREATE ===
        self.label_title = QLabel("DASHBOARD TITLE")
        self.label_normal = QLabel("Normal label for info")
        self.label_status = QLabel("Status: Active")

        # === LAYOUT ===
        layout = QVBoxLayout(self)
        layout.setSpacing(15)
        layout.setContentsMargins(20, 20, 20, 20)
        layout.addWidget(self.label_title)
        layout.addWidget(self.label_normal)
        layout.addWidget(self.label_status)
        self.label_title.setAlignment(Qt.AlignmentFlag.AlignCenter)
        self.label_normal.setAlignment(Qt.AlignmentFlag.AlignCenter)
        self.label_status.setAlignment(Qt.AlignmentFlag.AlignCenter)

        

        # === SET OBJECT NAME TO MATCH CSS #id ===
        self.label_title.setObjectName("title")
        self.label_normal.setObjectName("normal")
        self.label_status.setObjectName("status")

        # You can also set alignment via Qt if you want
        # self.label_title.setAlignment(Qt.AlignmentFlag.AlignCenter)
```
